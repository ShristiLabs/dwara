1//! dwara web console v3 (live charts, CRUD flows, AI ops) -- #226.
2//!
3//! A static SPA served from the mTLS admin listener. The SPA fetches
4//! from the admin API (same origin). No dataplane deps -- the SPA is
5//! static, embedded at compile time via `include_str!`/`include_bytes!`.
6//!
7//! ## Views
8//!
9//! - Overview: gateway status, active requests, uptime, config epoch,
10//!   route/listener counts.
11//! - Live (#226): real-time charts from /analytics/live and
12//!   /analytics/dashboard. Canvas-based latency sparklines and
13//!   request-rate bars, auto-refreshing every 2 seconds.
14//! - Routes: route table (name, path, service, methods) with CRUD
15//!   buttons (create, edit, delete) that call POST/PUT/DELETE /routes.
16//! - Upstreams: upstream/service health table (service, address,
17//!   health, requests, errors) with CRUD buttons.
18//! - Health: raw health JSON.
19//! - Analytics: Top-N analytics.
20//! - AI Ops (#226): AI credential pool status, MCP sessions/tools,
21//!   experiment overrides, and prompt-log query.
22//! - Fleet (DW-118): version skew check + fleet status (upgrade order,
23//!   concurrency cap, halt-on-failure). Consumes GET /fleet/skew and
24//!   GET /fleet/status.
25//! - Config: current config YAML dump (read-only).
26//! - Editor (DW-118): config YAML editor with validation preview
27//!   (POST /config/validate) and publish (PATCH /config). Shows
28//!   validation issues before the operator commits.
29//!
30//! ## Workspace switcher (DW-118)
31//!
32//! The top bar includes a workspace switcher that fetches the
33//! available workspaces from GET /workspaces (DW-067). When workspaces
34//! are not configured (OSS or no ent license), the switcher falls back
35//! to a single "default" option.
36//!
37//! ## Done-when
38//!
39//! An operator runs day-2 fleet operations (canary, rollback,
40//! credential rotation) from the console -- no CLI needed. CRUD on
41//! routes/services/policies/consumers shows a validation preview
42//! before publish. Fleet views show all CP-managed edges; workspace
43//! switcher scopes views to a workspace.
44//!
45//! ## Serving
46//!
47//! The console is served at `/console/` from the admin listener. The
48//! admin handler checks for `/console` paths before dispatching to the
49//! admin API handlers. The SPA fetches from the admin API endpoints
50//! (`/health`, `/stats`, `/config`, `/config_dump`, `/analytics/top`)
51//! on the same origin.

53// Embed the static files at compile time. No runtime file system
54// dependency, no external crate needed.
55const INDEX_HTML: &str = include_str!("../static/index.html");
56const STYLE_CSS: &str = include_str!("../static/style.css");
57const APP_JS: &str = include_str!("../static/app.js");
58
59/// A static file response: the body bytes and the content-type.
60pub struct StaticFile {
61    pub body: &'static [u8],
62    pub content_type: &'static str,
63}
64
65/// Resolve a console path to a static file.
66///
67/// Returns `Some(file)` if the path matches a known static file,
68/// `None` if the path is not a console path.
69///
70/// # Examples
71///
72/// ```
73/// use dwara_console::resolve;
74///
75/// // Index page.
76/// let file = resolve("/console/").unwrap();
77/// assert_eq!(file.content_type, "text/html; charset=utf-8");
78///
79/// // CSS.
80/// let file = resolve("/console/style.css").unwrap();
81/// assert_eq!(file.content_type, "text/css; charset=utf-8");
82///
83/// // Unknown path.
84/// assert!(resolve("/console/unknown").is_none());
85/// ```
86pub fn resolve(path: &str) -> Option<StaticFile> {
87    match path {
88        "/console" | "/console/" | "/console/index.html" => Some(StaticFile {
89            body: INDEX_HTML.as_bytes(),
90            content_type: "text/html; charset=utf-8",
91        }),
92        "/console/style.css" => Some(StaticFile {
93            body: STYLE_CSS.as_bytes(),
94            content_type: "text/css; charset=utf-8",
95        }),
96        "/console/app.js" => Some(StaticFile {
97            body: APP_JS.as_bytes(),
98            content_type: "application/javascript; charset=utf-8",
99        }),
100        _ => None,
101    }
102}
103
104/// Check if a path is a console path (starts with `/console`).
105pub fn is_console_path(path: &str) -> bool {
106    path == "/console" || path.starts_with("/console/")
107}
