1//! Replay time-travel debugging CLI logic (DW-102): the pure half of
2//! the `dwara replay` subcommand, kept library-shaped so tests exercise
3//! exactly what the binary runs.
4//!
5//! # Exit-code contract
6//!
7//! - 0 = no decision diffs (the candidate config behaves identically
8//!   to the baseline for every recorded request).
9//! - 1 = diffs found (useful as a CI gate: a config change that
10//!   alters routing, authz, rate-limit, transform, or upstream
11//!   decisions for recorded traffic is surfaced before deploy).
12//! - 2 = the recording or a config could not be loaded (operator
13//!   error, distinct from a clean diff).
14//!
15//! # Recording format
16//!
17//! A recording is a JSON document exported from the analytics store
18//! (or authored by hand for test fixtures):
19//!
20//! ```json
21//! {
22//!   "baseline_config": "<baseline YAML string>",
23//!   "requests": [
24//!     {
25//!       "method": "GET",
26//!       "path": "/api/foo",
27//!       "headers": [["x-plan", "pro"]],
28//!       "auth_identity": "alice",
29//!       "timestamp_ms": 1700000000000
30//!     }
31//!   ]
32//! }
33//! ```
34//!
35//! The `baseline_config` is the config the requests were captured
36//! under; `--config` is the candidate (new) config. The replay runs
37//! [`dwara_core::dataplane::replay::decide`] for each request under
38//! BOTH configs and emits a per-request [`DecisionDiff`] report.
39
40use dwara_core::config::parse_gateway;
41use dwara_core::dataplane::replay::{decide, DecisionDiff, ReplayRequest, SimulatedCounter};
42use dwara_core::snapshot::{compile, Snapshot};
43
44/// One request in a recording (the JSON shape the CLI reads).
45#[derive(Debug, Clone, serde::Deserialize, serde::Serialize)]
46pub struct RecordedRequest {
47    pub method: String,
48    pub path: String,
49    #[serde(default)]
50    pub headers: Vec<(String, String)>,
51    #[serde(default)]
52    pub auth_identity: Option<String>,
53    pub timestamp_ms: i64,
54}
55
56/// A recording: the baseline config plus the captured requests.
57#[derive(Debug, Clone, serde::Deserialize, serde::Serialize)]
58pub struct Recording {
59    pub baseline_config: String,
60    pub requests: Vec<RecordedRequest>,
61}
62
63/// The outcome of a replay run: the per-request diffs and a summary
64/// count. The CLI prints the diffs and exits 1 when `diff_count > 0`.
65#[derive(Debug, Clone)]
66pub struct ReplayReport {
67    /// One diff per request (in recording order); empty summary means
68    /// no change for that request.
69    pub diffs: Vec<DecisionDiff>,
70    /// The number of requests with at least one changed stage.
71    pub diff_count: usize,
72}
73
74impl ReplayReport {
75    /// A human-readable report (one line per changed request, plus a
76    /// summary footer). Empty output means no diffs.
77    pub fn render(&self) -> String {
78        if self.diffs.is_empty() {
79            return "no requests in recording\n".to_string();
80        }
81        let mut out = String::new();
82        for diff in &self.diffs {
83            let line = diff.summary();
84            if !line.is_empty() {
85                out.push_str(&line);
86                out.push('\n');
87            }
88        }
89        if self.diff_count == 0 {
90            out.push_str("no decision differences\n");
91        } else {
92            out.push_str(&format!(
93                "{} request(s) with decision differences\n",
94                self.diff_count
95            ));
96        }
97        out
98    }
99}

101/// Run a replay: load the recording, compile the baseline and candidate
102/// configs, run [`decide`] for each request under both, and diff.
103/// Returns the report or an error message (operator error).
104pub fn run_replay(
105    recording_text: &str,
106    candidate_config_text: &str,
107) -> Result<ReplayReport, String> {
108    let recording: Recording =
109        serde_json::from_str(recording_text).map_err(|e| format!("cannot parse recording: {e}"))?;
110    let baseline = compile_config(&recording.baseline_config)?;
111    let candidate = compile_config(candidate_config_text)?;
112    Ok(replay_against(&baseline, &candidate, &recording.requests))
113}
114
115/// Run [`decide`] for each request under both snapshots and diff.
116/// Public for testing (the pure core of [`run_replay`] without the
117/// parse/compile error paths).
118pub fn replay_against(
119    baseline: &Snapshot,
120    candidate: &Snapshot,
121    requests: &[RecordedRequest],
122) -> ReplayReport {
123    let mut diffs = Vec::with_capacity(requests.len());
124    let mut diff_count = 0usize;
125    for req in requests {
126        let replay_req = ReplayRequest {
127            method: req.method.clone(),
128            path: req.path.clone(),
129            headers: req.headers.clone(),
130            auth_identity: req.auth_identity.clone(),
131            timestamp_ms: req.timestamp_ms,
132        };
133        // Each request gets its OWN simulated counter: replay answers
134        // "what would THIS request do under THIS config?" not "what
135        // would a burst of these do?" — the per-request decision
136        // boundary is what a diff cares about, and sharing a counter
137        // across requests would make the rate-limit verdict depend on
138        // recording order.
139        let mut old_counter = SimulatedCounter::new();
140        let mut new_counter = SimulatedCounter::new();
141        let old = decide(baseline, &replay_req, &mut old_counter);
142        let new = decide(candidate, &replay_req, &mut new_counter);
143        let diff = DecisionDiff::compare(&req.path, &old, &new);
144        if diff.any() {
145            diff_count += 1;
146        }
147        diffs.push(diff);
148    }
149    ReplayReport { diffs, diff_count }
150}

152/// Compile a config YAML into a [`Snapshot`] (validate + compile). The
153/// snapshot is generation-0 (replay does not need a real generation
154/// id).
155fn compile_config(text: &str) -> Result<Snapshot, String> {
156    let gateway = parse_gateway(text).map_err(|e| format!("parse failed: {e}"))?;
157    let compiled = compile(&gateway).map_err(|e| format!("{e}"))?;
158    Ok(Snapshot::from_compiled(compiled))
159}
