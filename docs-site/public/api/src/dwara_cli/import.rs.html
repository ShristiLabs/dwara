1//! OpenAPI spec import (DW-047): `dwara import openapi` reads an
2//! OpenAPI 3.x spec (YAML or JSON) and generates a Dwara config YAML
3//! with routes derived from the spec's paths and methods.
4//!
5//! The import is a one-shot scaffolding step: it produces a config the
6//! operator edits to point at their real backend (the placeholder
7//! upstream `openapi-backend` at 127.0.0.1:9000). Each path+method
8//! becomes a route with `type: exact` matching (OpenAPI paths are
9//! exact, with `{param}` segments preserved as Dwara path params), a
10//! `proxy` action, and an `openapi` extension field carrying the
11//! operationId, summary, and tags for traceability.
12//!
13//! CFG-04 (#240): with `--mock`, the importer reads the OpenAPI
14//! `responses` examples and generates `mock` route actions with the
15//! example response body and status. This lets an operator scaffold a
16//! fully functional mock API from an OpenAPI spec without any backend.
17//!
18//! No new dependencies: `serde_yaml_ng` and `serde_json` (both already
19//! workspace dependencies) parse the spec. A minimal OpenAPI 3.x struct
20//! captures paths, methods, operationId, parameters, and requestBody
21//! schemas — enough to generate routes; the full OpenAPI surface is
22//! out of scope (this is scaffolding, not a spec validator).
23

32/// The minimal OpenAPI 3.x document shape the importer reads. Unknown
33/// fields are ignored (the spec is rich; we capture only what drives
34/// route generation).
35#[derive(Debug, Deserialize)]
36struct OpenApiDoc {
37    #[serde(default)]
38    paths: BTreeMap<String, PathItem>,
39    #[serde(default, rename = "x-openapi-version")]
40    _version: Option<String>,
41}
42
43/// One path's operations (OpenAPI 3.x: `get`, `post`, ...). Unknown
44/// fields (servers, parameters shared across operations) are ignored.
45#[derive(Debug, Deserialize)]
46struct PathItem {
47    #[serde(default)]
48    get: Option<Operation>,
49    #[serde(default)]
50    post: Option<Operation>,
51    #[serde(default)]
52    put: Option<Operation>,
53    #[serde(default)]
54    delete: Option<Operation>,
55    #[serde(default)]
56    patch: Option<Operation>,
57    #[serde(default)]
58    options: Option<Operation>,
59    #[serde(default)]
60    head: Option<Operation>,
61}
62
63/// One OpenAPI operation: operationId, summary, tags, and responses
64/// (CFG-04, #240: responses carry examples for mock mode).
65#[derive(Debug, Deserialize)]
66struct Operation {
67    #[serde(default, rename = "operationId")]
68    operation_id: Option<String>,
69    #[serde(default)]
70    summary: Option<String>,
71    #[serde(default)]
72    tags: Vec<String>,
73    /// OpenAPI responses: status code -> response object. Used by
74    /// `--mock` to pick an example response.
75    #[serde(default)]
76    responses: BTreeMap<String, Response>,
77}
78
79/// One OpenAPI response: carries `content` (media type -> media type
80/// object with examples).
81#[derive(Debug, Deserialize)]
82struct Response {
83    #[serde(default)]
84    content: BTreeMap<String, MediaType>,
85}
86
87/// One media type object: carries `example` or `examples`.
88#[derive(Debug, Deserialize)]
89struct MediaType {
90    #[serde(default)]
91    example: Option<serde_json::Value>,
92    #[serde(default)]
93    examples: BTreeMap<String, ExampleRef>,
94}
95
96/// An example reference or inline value.
97#[derive(Debug, Deserialize)]
98struct ExampleRef {
99    #[serde(default)]
100    value: Option<serde_json::Value>,
101}

103/// The HTTP methods the importer recognizes, in a stable order for
104/// deterministic route generation.
105/// Function pointer type for extracting an operation from a path item.
106type OpGetter = fn(&PathItem) -> Option<&Operation>;
107
108const METHODS: &[(&str, OpGetter)] = &[
109    ("GET", |p| p.get.as_ref()),
110    ("POST", |p| p.post.as_ref()),
111    ("PUT", |p| p.put.as_ref()),
112    ("DELETE", |p| p.delete.as_ref()),
113    ("PATCH", |p| p.patch.as_ref()),
114    ("OPTIONS", |p| p.options.as_ref()),
115    ("HEAD", |p| p.head.as_ref()),
116];

118/// The output of an import: the generated config YAML text, ready to
119/// write to disk.
120#[derive(Debug)]
121pub struct ImportResult {
122    pub yaml: String,
123    pub route_count: usize,
124}
125
126/// Import an OpenAPI spec (YAML or JSON, detected by file extension or
127/// content) and produce a Dwara config YAML. Returns an error string
128/// on parse failure (the caller prints it and exits 1).
129///
130/// When `mock` is true (CFG-04, #240), the importer reads OpenAPI
131/// `responses` examples and generates `mock` route actions instead of
132/// `proxy` actions, so the config is a fully functional mock API
133/// without any backend.
134pub fn import_openapi(spec_text: &str, is_json: bool, mock: bool) -> Result<ImportResult, String> {
135    let doc: OpenApiDoc = if is_json {
136        serde_json::from_str(spec_text).map_err(|e| format!("invalid JSON spec: {e}"))?
137    } else {
138        serde_yaml_ng::from_str(spec_text).map_err(|e| format!("invalid YAML spec: {e}"))?
139    };
140    let gateway = build_gateway(&doc, mock);
141    let route_count = gateway.routes.len();
142    let yaml = dwara_core::config::gateway_to_yaml(&gateway)
143        .map_err(|e| format!("failed to serialize generated config: {e}"))?;
144    Ok(ImportResult { yaml, route_count })
145}

147/// Detect whether the spec text is JSON (starts with `{` after
148/// trimming leading whitespace/BOM) or YAML.
149pub fn is_json_spec(text: &str) -> bool {
150    let trimmed = text.trim_start_matches('\u{feff}').trim_start();
151    trimmed.starts_with('{')
152}

154/// Build a [`Gateway`] from the parsed OpenAPI document: one route per
155/// UNIQUE PATH (Dwara's route table is path-only, so multiple methods
156/// on the same path share a single route with `match.methods` listing
157/// them all), one service, one placeholder upstream. The first
158/// operation on each path drives the route name (operationId
159/// preferred); subsequent methods on the same path are appended to
160/// `match.methods`.
161fn build_gateway(doc: &OpenApiDoc, mock: bool) -> Gateway {
162    let mut routes = Vec::new();
163    let mut used_names: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();
164
165    for (path, item) in &doc.paths {
166        let mut methods_for_path: Vec<String> = Vec::new();
167        let mut first_op: Option<&Operation> = None;
168        let mut first_method: Option<&str> = None;
169        for (method, getter) in METHODS {
170            let Some(op) = getter(item) else { continue };
171            methods_for_path.push(method.to_string());
172            if first_op.is_none() {
173                first_op = Some(op);
174                first_method = Some(method);
175            }
176        }
177        if methods_for_path.is_empty() {
178            continue;
179        }
180        let op = first_op.expect("at least one operation");
181        let method = first_method.expect("at least one method");
182        let name = derive_route_name(op, path, method, &mut used_names);
183
184        // CFG-04 (#240): in mock mode, generate a mock action with the
185        // example response from the OpenAPI spec. In proxy mode, the
186        // standard proxy action is used.
187        let action = if mock {
188            let (status, body) = extract_example(op);
189            RouteAction::Mock {
190                mock: MockAction {
191                    status,
192                    headers: BTreeMap::new(),
193                    body,
194                    body_file: None,
195                    delay_ms: None,
196                },
197            }
198        } else {
199            RouteAction::Proxy { rewrite: None }
200        };
201
202        let route = Route {
203            name,
204            service: "openapi-service".to_string(),
205            r#match: RouteMatch {
206                path: PathMatch {
207                    kind: PathMatchKind::Exact,
208                    value: path.clone(),
209                },
210                host: None,
211                methods: methods_for_path,
212                headers: BTreeMap::new(),
213                query: Vec::new(),
214                cookies: Vec::new(),
215                accept: None,
216            },
217            action,
218            policies: Vec::new(),
219            priority: None,
220            auth_required: false,
221            cors: None,
222            compression: None,
223            limits: None,
224            authorization: None,
225            security_headers_opt_out: false,
226            deprecation: None,
227            maintenance: None,
228            transforms: None,
229            security_headers: None,
230            masking: None,
231            cache: None,
232            methods: Vec::new(),
233            slo: None,
234            websocket: None,
235            waf: None,
236            graphql: None,
237            grpc_web: None,
238            translation: None,
239            request_validation: None,
240            openapi: Some(OpenApiMeta {
241                operation_id: op.operation_id.clone(),
242                summary: op.summary.clone(),
243                tags: op.tags.clone(),
244                method: method.to_string(),
245                path: path.clone(),
246            }),
247            mirror: None,
248            fault_injection: None,
249            plugins: Vec::new(),
250            filter_chain: None,
251            oidc_login: None,
252        };
253        routes.push(route);
254    }
255
256    let allow_empty_routes = routes.is_empty();
257    Gateway {
258        version: 1,
259        listeners: Vec::new(),
260        routes,
261        services: vec![Service {
262            name: "openapi-service".to_string(),
263            upstream: Some("openapi-backend".to_string()),
264            split: None,
265            sticky: None,
266            base_path: None,
267            version: None,
268            policies: Vec::new(),
269            authorization: None,
270        }],
271        upstreams: vec![Upstream {
272            name: "openapi-backend".to_string(),
273            load_balancer: dwara_core::config::LoadBalancer::RoundRobin,
274            hash_on: None,
275            protocol: dwara_core::config::UpstreamProtocol::Http1,
276            trusted_ca_file: None,
277            use_system_roots: false,
278            endpoints: vec![Endpoint {
279                address: "127.0.0.1".to_string(),
280                port: 9000,
281                weight: 1,
282                region: None,
283                zone: None,
284            }],
285            connection_cap: None,
286            slow_start_ms: None,
287            health: None,
288            active_health: None,
289            retries: None,
290            breaker: None,
291            max_pending: None,
292            timeouts: None,
293            oauth2_client_credentials: None,
294            dns_discovery: None,
295            peak_ewma: None,
296            locality: None,
297            pq: false,
298            cert_pinning: None,
299            mtls: None,
300            pool: None,
301        }],
302        consumers: Vec::new(),
303        policies: Vec::new(),
304        global_policies: Vec::new(),
305        authorization: None,
306        default_security_headers: None,
307        waf: None,
308        trusted_proxies: Vec::new(),
309        max_concurrent_requests: None,

339/// CFG-04 (#240): extract an example response from an OpenAPI
340/// operation's `responses` block. Returns `(status, body)` where
341/// `body` is the serialized example (pretty JSON) or `None` when no
342/// example is found.
343///
344/// Selection order:
345/// 1. The lowest 2xx status code with an example.
346/// 2. The `default` response with an example.
347/// 3. The first response (any status) with an example.
348/// 4. Fallback: `(200, None)` — a 200 with an empty body.
349fn extract_example(op: &Operation) -> (u16, Option<String>) {
350    // Try 2xx responses in ascending order.
351    for (status_str, response) in &op.responses {
352        if let Ok(status) = status_str.parse::<u16>() {
353            if (200..300).contains(&status) {
354                if let Some(body) = pick_example(response) {
355                    return (status, Some(body));
356                }
357            }
358        }
359    }
360    // Try `default` response.
361    if let Some(response) = op.responses.get("default") {
362        if let Some(body) = pick_example(response) {
363            return (200, Some(body));
364        }
365    }
366    // Try any response with an example.
367    for (status_str, response) in &op.responses {
368        if let Some(body) = pick_example(response) {
369            let status = status_str.parse::<u16>().unwrap_or(200);
370            return (status, Some(body));
371        }
372    }
373    // Fallback: 200 with no body.
374    (200, None)
375}

377/// Pick an example from a response's content. Tries `example` first,
378/// then the first `examples` entry's `value`.
379fn pick_example(response: &Response) -> Option<String> {
380    for media in response.content.values() {
381        if let Some(example) = &media.example {
382            return Some(serde_json::to_string_pretty(example).unwrap_or_default());
383        }
384        for ex in media.examples.values() {
385            if let Some(value) = &ex.value {
386                return Some(serde_json::to_string_pretty(value).unwrap_or_default());
387            }
388        }
389    }
390    None
391}

393/// Derive a route name from the operationId (preferred) or a
394/// path+method fallback. Ensures uniqueness by appending a counter
395/// when the derived name collides.
396fn derive_route_name(
397    op: &Operation,
398    path: &str,
399    method: &str,
400    used: &mut std::collections::BTreeSet<String>,
401) -> String {
402    let base = if let Some(id) = &op.operation_id {
403        sanitize_name(id)
404    } else {
405        // Fallback: method + path with separators replaced.
406        let path_slug = path
407            .trim_start_matches('/')
408            .replace('/', "-")
409            .replace(['{', '}', '.'], "");
410        let path_slug = path_slug.trim_matches('-');
411        if path_slug.is_empty() {
412            format!("{}-root", method.to_lowercase())
413        } else {
414            format!("{}-{}", method.to_lowercase(), path_slug)
415        }
416    };
417    // Ensure uniqueness.
418    let mut name = base.clone();
419    let mut counter = 2;
420    while !used.insert(name.clone()) {
421        name = format!("{base}-{counter}");
422        counter += 1;
423    }
424    name
425}

427/// Sanitize an operationId into a valid Dwara route name (lowercase,
428/// alphanumerics and hyphens only).
429fn sanitize_name(id: &str) -> String {
430    id.chars()
431        .map(|c| {
432            if c.is_ascii_alphanumeric() {
433                c.to_ascii_lowercase()
434            } else {
435                '-'
436            }
437        })
438        .collect::<String>()
439        .trim_matches('-')
440        .to_string()
441}
