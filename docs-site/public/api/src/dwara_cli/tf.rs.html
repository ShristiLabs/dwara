1//! Terraform-compatible state tool (DW-065): a `dwara tf` CLI subcommand
2//! that exports/imports Terraform-compatible JSON state and generates
3//! HCL, performing plan/apply round-trips directly over the admin API.
4//!
5//! ## Why a CLI tool, not a terraform-plugin-rs provider
6//!
7//! The `terraform-plugin-rs` ecosystem is MPL-2.0, which is not in the
8//! project's `deny.toml` allow list, and the compliance control must not
9//! be modified. So this module implements a CLI-based Terraform state
10//! tool instead: no gRPC plugin, no Terraform binary required, and no
11//! new external dependencies (hyper, serde_json, serde_yaml_ng, serde,
12//! clap, tokio are all already workspace deps).
13//!
14//! ## SCALE-12 (#193): True provider behavior via per-entity CRUD
15//!
16//! The original `apply` command pushes the full desired config as a
17//! single `PATCH /config` (full-document replacement). The
18//! `apply-crud` command (added in #193) uses the admin API's
19//! per-entity CRUD endpoints (`POST /routes`, `PUT /routes/<name>`,
20//! `DELETE /routes/<name>`, etc.) to manage each resource
21//! independently — the same behavior a real Terraform provider would
22//! exhibit. This makes the tool a "true provider" in behavior, even
23//! without the gRPC plugin protocol. The existing `apply` (full-document
24//! PATCH) is retained for backward compatibility and bulk-replace
25//! workflows.
26//!
27//! ## State model
28//!
29//! The tfstate JSON follows Terraform's state file structure (version,
30//! terraform_version, resources[] with each resource having mode
31//! "managed", type, name, instances[] with attributes). Dwara config
32//! entities map to resources:
33//!
34//! - `dwara_listener` (one per listener)
35//! - `dwara_route` (one per route)
36//! - `dwara_service` (one per service)
37//! - `dwara_upstream` (one per upstream, with endpoints as an attribute)
38//! - `dwara_consumer` (one per consumer)
39//!
40//! The HCL file generates `resource "dwara_route" "<name>" { ... }`
41//! blocks with the entity attributes. The state is structurally
42//! Terraform-compatible so a future real provider or `terraform import`
43//! could consume it.
44//!
45//! ## Plan/apply flow
46//!
47//! - `export`: GET /config from the admin API, parse the YAML via
48//!   `dwara_core::config::parse_gateway`, then serialize each entity to
49//!   tfstate attributes and HCL blocks.
50//! - `plan`: read local tfstate, GET /config from the gateway, compute
51//!   the diff (added/removed/changed entities), print it human-readably.
52//!   Exit 0 if no diff, 1 if diff.
53//! - `apply`: push the desired config to the gateway via PATCH /config
54//!   (body = YAML), then refresh state from the response. If `--config`
55//!   is given, use that YAML as the desired config; otherwise derive the
56//!   desired YAML from the tfstate.

75/// The Terraform state file root shape. `version` is the state format
76/// version (4 is the current Terraform state format version); the
77/// `resources` array holds one entry per managed resource instance.
78#[derive(Debug, Clone, Serialize, Deserialize)]
79pub struct TfState {
80    pub version: u32,
81    pub terraform_version: String,
82    pub resources: Vec<TfResource>,
83}

85/// One resource entry in the tfstate: a managed resource of `type` named
86/// `name` with one instance carrying its attributes.
87#[derive(Debug, Clone, Serialize, Deserialize)]
88pub struct TfResource {
89    pub mode: String,
90    #[serde(rename = "type")]
91    pub r#type: String,
92    pub name: String,
93    pub instances: Vec<TfInstance>,
94}

107/// Build a [`TfState`] from a parsed [`Gateway`]. Each listener, route,
108/// service, upstream, and consumer becomes one managed resource. The
109/// attribute set is the serde_json serialization of the entity (stable
110/// and faithful enough for export -> apply -> export round-trips).
111pub fn gateway_to_state(gateway: &Gateway) -> TfState {
112    let mut resources = Vec::new();
113
114    for l in &gateway.listeners {
115        resources.push(TfResource {
116            mode: "managed".to_string(),
117            r#type: "dwara_listener".to_string(),
118            name: l.name.clone(),
119            instances: vec![TfInstance {
120                attributes: serde_json::to_value(listener_attrs(l)).unwrap_or(Value::Null),
121            }],
122        });
123    }
124    for r in &gateway.routes {
125        resources.push(TfResource {
126            mode: "managed".to_string(),
127            r#type: "dwara_route".to_string(),
128            name: r.name.clone(),
129            instances: vec![TfInstance {
130                attributes: serde_json::to_value(route_attrs(r)).unwrap_or(Value::Null),
131            }],
132        });
133    }
134    for s in &gateway.services {
135        resources.push(TfResource {
136            mode: "managed".to_string(),
137            r#type: "dwara_service".to_string(),
138            name: s.name.clone(),
139            instances: vec![TfInstance {
140                attributes: serde_json::to_value(service_attrs(s)).unwrap_or(Value::Null),
141            }],
142        });
143    }
144    for u in &gateway.upstreams {
145        resources.push(TfResource {
146            mode: "managed".to_string(),
147            r#type: "dwara_upstream".to_string(),
148            name: u.name.clone(),
149            instances: vec![TfInstance {
150                attributes: serde_json::to_value(upstream_attrs(u)).unwrap_or(Value::Null),
151            }],
152        });
153    }
154    for c in &gateway.consumers {
155        resources.push(TfResource {
156            mode: "managed".to_string(),
157            r#type: "dwara_consumer".to_string(),
158            name: c.name.clone(),
159            instances: vec![TfInstance {
160                attributes: json!({ "name": c.name }),
161            }],
162        });
163    }
164
165    TfState {
166        version: 4,
167        terraform_version: "1.5.0".to_string(),
168        resources,
169    }
170}

183/// Reconstruct a [`Gateway`] from a [`TfState`]. This is the inverse of
184/// [`gateway_to_state`]: each resource's attributes are deserialized
185/// back into the typed config entity. Fields not captured in the state
186/// default to their schema defaults. The `allow_empty_routes` flag is
187/// set when the reconstructed route list is empty (so validation passes
188/// for a state that legitimately has no routes).
189pub fn state_to_gateway(state: &TfState) -> Result<Gateway, String> {
190    let mut listeners = Vec::new();
191    let mut routes = Vec::new();
192    let mut services = Vec::new();
193    let mut upstreams = Vec::new();
194    let mut consumers = Vec::new();
195
196    for res in &state.resources {
197        let attrs = res
198            .instances
199            .first()
200            .map(|i| &i.attributes)
201            .unwrap_or(&Value::Null);
202        match res.r#type.as_str() {
203            "dwara_listener" => {
204                listeners.push(parse_listener_attrs(attrs)?);
205            }
206            "dwara_route" => {
207                routes.push(parse_route_attrs(attrs)?);
208            }
209            "dwara_service" => {
210                services.push(parse_service_attrs(attrs)?);
211            }
212            "dwara_upstream" => {
213                upstreams.push(parse_upstream_attrs(attrs)?);
214            }
215            "dwara_consumer" => {
216                let name = attrs
217                    .get("name")
218                    .and_then(Value::as_str)
219                    .unwrap_or("")
220                    .to_string();
221                consumers.push(dwara_core::config::Consumer {
222                    name,
223                    credentials: Vec::new(),
224                    policies: Vec::new(),
225                    consumer_type: dwara_core::config::ConsumerType::User,
226                    tool_allowlist: Vec::new(),
227                    token_budget: None,
228                    priority: None,
229                    groups: Vec::new(),
230                    authorization: None,
231                    quotas: None,
232                    ai_logging: None,
233                    agent: None,
234                });
235            }
236            _ => {}
237        }
238    }
239
240    let allow_empty_routes = routes.is_empty();
241    Ok(Gateway {
242        version: 1,
243        listeners,
244        routes,
245        services,
246        upstreams,
247        consumers,
248        policies: Vec::new(),
249        global_policies: Vec::new(),
250        authorization: None,
251        default_security_headers: None,
252        waf: None,
253        trusted_proxies: Vec::new(),
254        max_concurrent_requests: None,
255        load_shed_dry_run: false,
256        jwt_providers: Vec::new(),
257        admin: None,
258        allow_empty_routes,
259        hmac_auth: None,
260        webhooks: Vec::new(),
261        analytics: None,
262        analytics_stream: None,
263        geoip: None,
264        admission_queue: None,
265        mtls_consumer_mapping: None,
266        mtls_forward_headers: None,
267        license: None,
268        oidc_providers: Vec::new(),
269        redis_rate_limiter: None,
270        redis_quotas: None,
271        redis_cache: None,
272        config_convergence: None,
273        plugins: Vec::new(),
274        filter_chain: None,
275        plugin_registry: None,
276        ai: None,
277        fleet: None,
278        lifecycle: None,
279        mesh: None,
280        ssrf_filter: None,
281    })
282}

288/// The subset of listener attributes captured in tfstate. The full
289/// Listener struct is rich; the tf tool captures the fields that
290/// round-trip through the admin API's normalized YAML (name, address,
291/// port, protocol). TLS and policy attachments are documented as
292/// follow-up scope (the dev admin round-trip is the primary target).
293fn listener_attrs(l: &Listener) -> Value {
294    json!({
295        "name": l.name,
296        "address": l.address,
297        "port": l.port,
298        "protocol": match l.protocol {
299            ListenerProtocol::Http => "http",
300            ListenerProtocol::Https => "https",
301            ListenerProtocol::H3 => "h3",
302            ListenerProtocol::Tcp => "tcp",
303            ListenerProtocol::Udp => "udp",
304        },
305    })
306}

308fn parse_listener_attrs(v: &Value) -> Result<Listener, String> {
309    let name = v

342fn route_attrs(r: &Route) -> Value {
343    let mut m = serde_json::Map::new();
344    m.insert("name".to_string(), json!(r.name));
345    m.insert("service".to_string(), json!(r.service));
346    m.insert(
347        "path_type".to_string(),
348        json!(match r.r#match.path.kind {
349            PathMatchKind::Exact => "exact",
350            PathMatchKind::Prefix => "prefix",
351            PathMatchKind::Regex => "regex",
352        }),
353    );
354    m.insert("path_value".to_string(), json!(r.r#match.path.value));
355    if let Some(h) = &r.r#match.host {
356        m.insert("host".to_string(), json!(h));
357    }
358    if !r.r#match.methods.is_empty() {
359        m.insert("methods".to_string(), json!(r.r#match.methods));
360    }
361    let action_type = match &r.action {
362        RouteAction::Proxy { .. } => "proxy",
363        RouteAction::Redirect { .. } => "redirect",
364        RouteAction::Respond { .. } => "respond",
365        RouteAction::Mock { .. } => "mock",
366        RouteAction::Ai { .. } => "ai",
367        RouteAction::NanoService { .. } => "nano_service",
368    };
369    m.insert("action_type".to_string(), json!(action_type));
370    if r.auth_required {
371        m.insert("auth_required".to_string(), json!(true));
372    }
373    if let Some(p) = r.priority {
374        m.insert("priority".to_string(), json!(p));
375    }
376    Value::Object(m)
377}

379fn parse_route_attrs(v: &Value) -> Result<Route, String> {
380    let name = v
381        .get("name")
382        .and_then(Value::as_str)
383        .ok_or("route missing name")?
384        .to_string();
385    let service = v
386        .get("service")
387        .and_then(Value::as_str)
388        .ok_or("route missing service")?
389        .to_string();
390    let path_type = v
391        .get("path_type")
392        .and_then(Value::as_str)
393        .unwrap_or("prefix");
394    let kind = match path_type {
395        "exact" => PathMatchKind::Exact,
396        "regex" => PathMatchKind::Regex,
397        _ => PathMatchKind::Prefix,
398    };
399    let path_value = v
400        .get("path_value")
401        .and_then(Value::as_str)
402        .unwrap_or("/")
403        .to_string();
404    let host = v.get("host").and_then(Value::as_str).map(String::from);
405    let methods: Vec<String> = v
406        .get("methods")
407        .and_then(Value::as_array)
408        .map(|a| {
409            a.iter()
410                .filter_map(|x| x.as_str().map(String::from))
411                .collect()
412        })
413        .unwrap_or_default();
414    let auth_required = v
415        .get("auth_required")
416        .and_then(Value::as_bool)
417        .unwrap_or(false);
418    let priority = v.get("priority").and_then(Value::as_u64).map(|p| p as u8);
419
420    Ok(Route {
421        name,
422        service,
423        r#match: RouteMatch {
424            path: PathMatch {
425                kind,
426                value: path_value,
427            },
428            host,
429            methods,

608/// Generate an HCL `.tf` file from a [`Gateway`]: one `resource` block
609/// per entity. The block body uses HCL attribute syntax with the same
610/// fields captured in the tfstate attributes.
611pub fn gateway_to_hcl(gateway: &Gateway) -> String {
612    let mut out = String::new();
613    out.push_str("# Generated by `dwara tf export`. Do not edit by hand;\n");
614    out.push_str("# regenerate with `dwara tf export --out-hcl <path>`.\n\n");
615
616    for l in &gateway.listeners {
617        out.push_str(&format!(
618            "resource \"dwara_listener\" \"{}\" {{\n  name    = \"{}\"\n  address = \"{}\"\n  port    = {}\n  protocol = \"{}\"\n}}\n\n",
619            hcl_name(&l.name),
620            l.name,
621            l.address,
622            l.port,
623            match l.protocol {
624                ListenerProtocol::Http => "http",
625                ListenerProtocol::Https => "https",
626                ListenerProtocol::H3 => "h3",
627                ListenerProtocol::Tcp => "tcp",
628                ListenerProtocol::Udp => "udp",
629            }
630        ));
631    }
632    for r in &gateway.routes {
633        out.push_str(&format!(
634            "resource \"dwara_route\" \"{}\" {{\n",
635            hcl_name(&r.name)
636        ));
637        out.push_str(&format!("  name    = \"{}\"\n", r.name));
638        out.push_str(&format!("  service = \"{}\"\n", r.service));
639        out.push_str(&format!(
640            "  path_type  = \"{}\"\n",
641            match r.r#match.path.kind {
642                PathMatchKind::Exact => "exact",
643                PathMatchKind::Prefix => "prefix",
644                PathMatchKind::Regex => "regex",
645            }
646        ));
647        out.push_str(&format!(
648            "  path_value = \"{}\"\n",
649            hcl_string(&r.r#match.path.value)
650        ));
651        if let Some(h) = &r.r#match.host {
652            out.push_str(&format!("  host = \"{}\"\n", hcl_string(h)));
653        }
654        if !r.r#match.methods.is_empty() {
655            let methods: Vec<String> = r
656                .r#match
657                .methods
658                .iter()
659                .map(|m| format!("\"{}\"", m))
660                .collect();
661            out.push_str(&format!("  methods = [{}]\n", methods.join(", ")));
662        }
663        let action_type = match &r.action {
664            RouteAction::Proxy { .. } => "proxy",
665            RouteAction::Redirect { .. } => "redirect",
666            RouteAction::Respond { .. } => "respond",
667            RouteAction::Mock { .. } => "mock",
668            RouteAction::Ai { .. } => "ai",

745/// Sanitize a name into a valid HCL resource label (lowercase
746/// alphanumerics and hyphens only; other characters become hyphens).

748    name.chars()
749        .map(|c| {

752            } else {
753                '-'
754            }
755        })
756        .collect::<String>()
757        .trim_matches('-')
758        .to_string()
759}

770/// One line of a plan diff: the kind of change and a human-readable
771/// description.
772#[derive(Debug, Clone, PartialEq, Eq)]
773pub enum DiffEntry {
774    Added {
775        r#type: String,
776        name: String,
777    },
778    Removed {
779        r#type: String,
780        name: String,
781    },
782    Changed {
783        r#type: String,
784        name: String,
785        detail: String,
786    },
787}

789/// Compute the diff between a desired state (the local tfstate) and the
790/// actual gateway config (parsed from GET /config). Returns a list of
791/// diff entries; an empty list means no drift (plan is clean).
792pub fn plan_diff(desired: &TfState, actual: &Gateway) -> Vec<DiffEntry> {
793    let actual_state = gateway_to_state(actual);
794    let desired_map = index_resources(desired);
795    let actual_map = index_resources(&actual_state);
796    let mut entries = Vec::new();
797
798    let all_keys: std::collections::BTreeSet<(String, String)> = desired_map
799        .keys()
800        .chain(actual_map.keys())
801        .cloned()
802        .collect();
803    for (rtype, name) in all_keys {
804        match (
805            desired_map.get(&(rtype.clone(), name.clone())),
806            actual_map.get(&(rtype.clone(), name.clone())),
807        ) {
808            (Some(_d), None) => entries.push(DiffEntry::Added {
809                r#type: rtype,
810                name,
811            }),
812            (None, Some(_a)) => entries.push(DiffEntry::Removed {
813                r#type: rtype,
814                name,
815            }),
816            (Some(d), Some(a)) => {
817                if d != a {
818                    entries.push(DiffEntry::Changed {
819                        r#type: rtype,
820                        name,
821                        detail: summarize_diff(d, a),
822                    });
823                }
824            }
825            (None, None) => {}
826        }
827    }
828    entries
829}

831/// Index a state's resources by (type, name) -> serialized attributes
832/// (a stable JSON string for comparison).
833fn index_resources(state: &TfState) -> BTreeMap<(String, String), String> {
834    let mut map = BTreeMap::new();
835    for res in &state.resources {
836        let attrs = res
837            .instances
838            .first()
839            .map(|i| i.attributes.clone())
840            .unwrap_or(Value::Null);
841        let key = (res.r#type.clone(), res.name.clone());
842        let serialized = serde_json::to_string(&attrs).unwrap_or_default();
843        map.insert(key, serialized);
844    }
845    map
846}

848/// Produce a short human-readable summary of what changed between two
849/// attribute JSON values (top-level key-level diff).
850fn summarize_diff(desired: &str, actual: &str) -> String {
851    let d: Value = serde_json::from_str(desired).unwrap_or(Value::Null);
852    let a: Value = serde_json::from_str(actual).unwrap_or(Value::Null);
853    let mut changes = Vec::new();
854    if let (Some(dobj), Some(aobj)) = (d.as_object(), a.as_object()) {
855        let all_keys: std::collections::BTreeSet<&String> =
856            dobj.keys().chain(aobj.keys()).collect();
857        for k in all_keys {
858            let dv = dobj.get(k);
859            let av = aobj.get(k);
860            if dv != av {
861                changes.push(format!(
862                    "{}: {} -> {}",
863                    k,
864                    av.map(|v| v.to_string())
865                        .unwrap_or_else(|| "(absent)".to_string()),
866                    dv.map(|v| v.to_string())
867                        .unwrap_or_else(|| "(absent)".to_string()),
868                ));
869            }
870        }
871    }
872    if changes.is_empty() {
873        "content differs".to_string()
874    } else {
875        changes.join("; ")
876    }
877}
878
879/// Format a list of diff entries as human-readable text (one line per
880/// entry). Returns the text and whether any diff was present.
881pub fn format_diff(entries: &[DiffEntry]) -> (String, bool) {
882    if entries.is_empty() {
883        return (
884            "No changes. Infrastructure is up-to-date.\n".to_string(),
885            false,
886        );
887    }
888    let mut out = String::new();
889    for e in entries {
890        match e {
891            DiffEntry::Added { r#type, name } => {
892                out.push_str(&format!("  + {} {} will be created\n", r#type, name));
893            }
894            DiffEntry::Removed { r#type, name } => {
895                out.push_str(&format!("  - {} {} will be destroyed\n", r#type, name));
896            }
897            DiffEntry::Changed {
898                r#type,
899                name,
900                detail,
901            } => {
902                out.push_str(&format!(
903                    "  ~ {} {} will be changed ({})\n",
904                    r#type, name, detail
905                ));
906            }
907        }
908    }
909    out.push_str(&format!("\nPlan: {} change(s).\n", entries.len()));
910    (out, true)
911}

1246/// operations instead of full-document PATCH. This is the "true
1247/// provider" behavior: each resource is managed independently via

1254/// 3. Compute the diff (added/removed/changed entities).
1255/// 4. For each added entity: POST /<entity_kind> with the entity JSON.
1256/// 5. For each changed entity: PUT /<entity_kind>/<name> with the
1257///    entity JSON.
1258/// 6. For each removed entity: DELETE /<entity_kind>/<name>.
1259///
1260/// Returns a human-readable summary of the operations performed.
1261pub async fn apply_crud(admin_url: &str, state_path: &str) -> Result<String, String> {
1262    let state_text = std::fs::read_to_string(state_path)
1263        .map_err(|e| format!("cannot read state file {state_path}: {e}"))?;
1264    let state = state_from_json(&state_text)?;
1265    let client = AdminClient::new(admin_url)?;
1266    let yaml = client.get_config().await?;
1267    let current_gateway = dwara_core::config::parse_gateway(&yaml)
1268        .map_err(|e| format!("parse gateway config: {e}"))?;
1269    let current_state = gateway_to_state(&current_gateway);
1270
1271    // Index desired and current resources by (type, name).
1272    let desired = index_state_resources(&state);
1273    let current = index_state_resources(&current_state);
1274
1275    let mut operations = Vec::new();
1276
1277    // Process each entity kind in order.
1278    for &kind in CRUD_ENTITY_KINDS {
1279        let tf_type = entity_kind_to_tf_type(kind);
1280
1281        // Added + changed: in desired but not in current, or different.
1282        for (name, desired_attrs) in desired.get(tf_type).into_iter().flatten() {
1283            match current.get(tf_type).and_then(|m| m.get(name)) {
1284                None => {
1285                    // Create: POST /<kind> with the entity JSON.
1286                    let body = attrs_to_entity_json(kind, name, desired_attrs)?;
1287                    client.create_entity(kind, &body).await?;
1288                    operations.push(format!("+ {tf_type}.{name} (created)"));
1289                }
1290                Some(current_attrs) if current_attrs != desired_attrs => {
1291                    // Replace: PUT /<kind>/<name> with the entity JSON.
1292                    let body = attrs_to_entity_json(kind, name, desired_attrs)?;
1293                    client.replace_entity(kind, name, &body).await?;
1294                    operations.push(format!("~ {tf_type}.{name} (replaced)"));
1295                }
1296                _ => {}
1297            }
1298        }
1299
1300        // Removed: in current but not in desired.
1301        for (name, _) in current.get(tf_type).into_iter().flatten() {
1302            if !desired
1303                .get(tf_type)
1304                .into_iter()
1305                .flatten()
1306                .any(|(n, _)| n == name)
1307            {
1308                client.delete_entity(kind, name).await?;
1309                operations.push(format!("- {tf_type}.{name} (deleted)"));
1310            }
1311        }
1312    }
1313
1314    if operations.is_empty() {
1315        Ok("No changes. Infrastructure is up-to-date.".to_string())
1316    } else {
1317        Ok(format!(
1318            "Applied {} change(s):\n{}",
1319            operations.len(),
1320            operations
1321                .iter()
1322                .map(|op| format!("  {op}"))
1323                .collect::<Vec<_>>()
1324                .join("\n")
1325        ))
1326    }
1327}

1329/// Map an admin API entity kind to the tfstate resource type.
1330fn entity_kind_to_tf_type(kind: &str) -> &'static str {
1331    match kind {
1332        "routes" => "dwara_route",
1333        "services" => "dwara_service",
1334        "upstreams" => "dwara_upstream",
1335        "consumers" => "dwara_consumer",
1336        "policies" => "dwara_policy",
1337        _ => "dwara_unknown",
1338    }
1339}

1341/// Index tfstate resources by (type -> name -> attributes). Returns a
1342/// map from resource type to a map from resource name to attributes.
1343fn index_state_resources(state: &TfState) -> BTreeMap<&'static str, BTreeMap<String, Value>> {
1344    let mut idx: BTreeMap<&'static str, BTreeMap<String, Value>> = BTreeMap::new();
1345    for res in &state.resources {
1346        let kind = match tf_type_to_entity_kind(&res.r#type) {
1347            Some(k) => k,
1348            None => continue,
1349        };
1350        let tf_type = entity_kind_to_tf_type(kind);
1351        for inst in &res.instances {
1352            if let Some(name) = inst.attributes.get("name").and_then(|v| v.as_str()) {
1353                idx.entry(tf_type)
1354                    .or_default()
1355                    .insert(name.to_string(), inst.attributes.clone());
1356            }
1357        }
1358    }
1359    idx
1360}

1362/// Convert tfstate attributes for an entity into the JSON body for
1363/// the admin API's POST/PUT endpoints. The admin API expects the
1364/// entity JSON (e.g., a `Route` object). The tfstate attributes are
1365/// already a subset of the entity fields; we wrap them in the
1366/// expected shape.
1367fn attrs_to_entity_json(_kind: &str, _name: &str, attrs: &Value) -> Result<String, String> {
1368    serde_json::to_string(attrs).map_err(|e| format!("serialize entity JSON: {e}"))
1369}
