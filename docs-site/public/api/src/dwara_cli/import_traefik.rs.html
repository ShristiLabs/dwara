1//! Traefik config import (CFG-09, #257): `dwara import traefik` reads
2//! a Traefik dynamic config (YAML) and generates a Dwara config YAML
3//! with routes derived from Traefik's HTTP routers and services.
4//!
5//! Traefik's dynamic config shape (YAML):
6//! ```yaml
7//! http:
8//!   routers:
9//!     my-router:
10//!       rule: "PathPrefix(`/api`)"
11//!       service: my-service
12//!   services:
13//!     my-service:
14//!       loadBalancer:
15//!         servers:
16//!           - url: http://127.0.0.1:8080
17//! ```
18//!
19//! The import maps each router to a Dwara route and each service to
20//! a Dwara upstream. Unsupported features (middlewares, TLS options,
21//! TCP/UDP routers) are emitted as YAML comments in the output.
22
23use dwara_core::config::{
24    Endpoint, Gateway, LoadBalancer, PathMatch, PathMatchKind, Route, RouteAction, RouteMatch,
25    Service, Upstream, UpstreamProtocol,
26};
27use serde::Deserialize;
28use std::collections::BTreeMap;
29
30use crate::import::ImportResult;

32/// The minimal Traefik dynamic config shape.
33#[derive(Debug, Default, Deserialize)]
34struct TraefikConfig {
35    #[serde(default)]
36    http: Option<TraefikHttp>,
37}
38
39#[derive(Debug, Default, Deserialize)]
40struct TraefikHttp {
41    #[serde(default)]
42    routers: BTreeMap<String, TraefikRouter>,
43    #[serde(default)]
44    services: BTreeMap<String, TraefikService>,
45}
46
47#[derive(Debug, Deserialize)]
48#[allow(dead_code)]
49struct TraefikRouter {
50    rule: String,
51    #[serde(default)]
52    service: Option<String>,
53    #[serde(default)]
54    priority: Option<i64>,
55    #[serde(default)]
56    entry_points: Vec<String>,
57}
58
59#[derive(Debug, Deserialize)]
60struct TraefikService {
61    #[serde(default)]
62    load_balancer: Option<TraefikLoadBalancer>,
63}
64
65#[derive(Debug, Deserialize)]
66struct TraefikLoadBalancer {
67    #[serde(default)]
68    servers: Vec<TraefikServer>,
69}
70
71#[derive(Debug, Deserialize)]
72#[allow(dead_code)]
73struct TraefikServer {
74    url: String,
75}

77/// Import a Traefik dynamic config (YAML) and produce a Dwara config
78/// YAML. Returns an error string on parse failure.
79pub fn import_traefik(text: &str) -> Result<ImportResult, String> {
80    let config: TraefikConfig =
81        serde_yaml_ng::from_str(text).map_err(|e| format!("invalid Traefik config: {e}"))?;
82
83    let mut upstreams = Vec::new();
84    let mut services: BTreeMap<String, Service> = BTreeMap::new();
85    let mut routes = Vec::new();
86    let mut warnings: Vec<String> = Vec::new();
87
88    // Build upstreams from services.
89    for (name, svc) in config
90        .http
91        .as_ref()
92        .map(|h| &h.services)
93        .into_iter()
94        .flatten()
95    {
96        let endpoints: Vec<Endpoint> = svc
97            .load_balancer
98            .as_ref()
99            .map(|lb| &lb.servers)
100            .into_iter()
101            .flatten()
102            .filter_map(|s| parse_url(&s.url))
103            .collect();
104
105        if endpoints.is_empty() {
106            warnings.push(format!(
107                "# WARNING: service '{name}' has no loadBalancer servers"
108            ));
109        }
110
111        upstreams.push(Upstream {
112            name: name.clone(),
113            load_balancer: LoadBalancer::RoundRobin,
114            protocol: UpstreamProtocol::Http1,
115            trusted_ca_file: None,
116            endpoints,
117            connection_cap: None,
118            slow_start_ms: None,
119            health: None,
120            active_health: None,
121            retries: None,
122            breaker: None,
123            max_pending: None,
124            timeouts: None,
125            oauth2_client_credentials: None,
126            dns_discovery: None,
127            peak_ewma: None,
128            locality: None,
129            pq: false,
130            cert_pinning: None,
131            mtls: None,
132            pool: None,
133            hash_on: None,
134            use_system_roots: false,
135        });
136
137        services.insert(
138            name.clone(),
139            Service {
140                name: name.clone(),
141                upstream: Some(name.clone()),
142                split: None,
143                sticky: None,
144                base_path: None,
145                version: None,
146                policies: Vec::new(),
147                authorization: None,
148            },
149        );
150    }
151
152    // Build routes from routers.
153    for (name, router) in config
154        .http
155        .as_ref()
156        .map(|h| &h.routers)
157        .into_iter()
158        .flatten()
159    {
160        let (path_match, host) = parse_rule(&router.rule, &mut warnings, name);
161
162        let service_name = router.service.clone().unwrap_or_else(|| {
163            warnings.push(format!("# WARNING: router '{name}' has no service"));
164            "placeholder".to_string()
165        });
166
167        // Create a service if it doesn't exist.
168        if !services.contains_key(&service_name) {
169            services.insert(
170                service_name.clone(),
171                Service {
172                    name: service_name.clone(),
173                    upstream: Some(service_name.clone()),
174                    split: None,
175                    sticky: None,
176                    base_path: None,
177                    version: None,
178                    policies: Vec::new(),
179                    authorization: None,
180                },
181            );
182        }
183
184        routes.push(Route {
185            name: name.clone(),
186            service: service_name,
187            r#match: RouteMatch {
188                path: path_match,
189                host,
190                methods: Vec::new(),
191                headers: BTreeMap::new(),
192                query: Vec::new(),
193                cookies: Vec::new(),
194                accept: None,
195            },
196            action: RouteAction::Proxy { rewrite: None },
197            policies: Vec::new(),
198            priority: None,
199            auth_required: false,
200            cors: None,
201            compression: None,
202            limits: None,
203            authorization: None,
204            deprecation: None,
205            maintenance: None,
206            transforms: None,
207            security_headers: None,
208            masking: None,
209            cache: None,
210            methods: Vec::new(),
211            slo: None,
212            websocket: None,
213            waf: None,
214            graphql: None,
215            grpc_web: None,
216            translation: None,
217            request_validation: None,
218            openapi: None,
219            mirror: None,
220            fault_injection: None,
221            plugins: Vec::new(),
222            oidc_login: None,
223            filter_chain: None,
224            security_headers_opt_out: false,
225        });
226    }
227
228    let allow_empty_routes = routes.is_empty();
229    let gateway = Gateway {
230        version: 1,
231        listeners: Vec::new(),
232        routes,
233        services: services.into_values().collect(),
234        upstreams,
235        consumers: Vec::new(),
236        policies: Vec::new(),
237        global_policies: Vec::new(),
238        authorization: None,
239        default_security_headers: None,
240        waf: None,
241        trusted_proxies: Vec::new(),
242        max_concurrent_requests: None,
243        load_shed_dry_run: false,
244        jwt_providers: Vec::new(),
245        admin: None,
246        allow_empty_routes,
247        hmac_auth: None,
248        webhooks: Vec::new(),
249        analytics: None,
250        analytics_stream: None,
251        geoip: None,
252        admission_queue: None,
253        mtls_consumer_mapping: None,
254        mtls_forward_headers: None,
255        license: None,
256        oidc_providers: Vec::new(),
257        redis_rate_limiter: None,
258        redis_quotas: None,
259        redis_cache: None,
260        config_convergence: None,
261        plugins: Vec::new(),
262        filter_chain: None,
263        plugin_registry: None,
264        ai: None,
265        fleet: None,
266        lifecycle: None,
267        mesh: None,
268        ssrf_filter: None,
269    };
270
271    let route_count = gateway.routes.len();
272    let mut yaml = dwara_core::config::gateway_to_yaml(&gateway)
273        .map_err(|e| format!("serialize failed: {e}"))?;
274    for w in &warnings {
275        yaml.push_str(w);
276        yaml.push('\n');
277    }
278
279    Ok(ImportResult { yaml, route_count })
280}

282/// Parse a Traefik router rule into a (PathMatch, Option<host>).
283/// Supports `PathPrefix(`/path`)`, `Path(`/path`)`, and `Host(`domain`)`.
284fn parse_rule(
285    rule: &str,
286    warnings: &mut Vec<String>,
287    router_name: &str,
288) -> (PathMatch, Option<String>) {
289    let mut path_match = PathMatch {
290        kind: PathMatchKind::Prefix,
291        value: "/".to_string(),
292    };
293    let mut host = None;
294
295    for part in rule.split("&&") {
296        let part = part.trim();
297        if let Some(p) = extract_call(part, "PathPrefix") {
298            path_match = PathMatch {
299                kind: PathMatchKind::Prefix,
300                value: p,
301            };
302        } else if let Some(p) = extract_call(part, "Path") {
303            path_match = PathMatch {
304                kind: PathMatchKind::Exact,
305                value: p,
306            };
307        } else if let Some(h) = extract_call(part, "Host") {
308            host = Some(h);
309        } else {
310            warnings.push(format!(
311                "# WARNING: router '{router_name}' rule segment '{part}' is not supported (ignored)"
312            ));
313        }
314    }
315
316    (path_match, host)
317}

319/// Extract the argument from a function-call-like rule segment,
320/// e.g. `PathPrefix(`/api`)` -> `/api`.
321fn extract_call(s: &str, func: &str) -> Option<String> {
322    let s = s.trim();
323    let prefix = format!("{func}(`");
324    let start = s.find(&prefix)?;
325    let rest = &s[start + prefix.len()..];
326    let end = rest.find("`)")?;
327    Some(rest[..end].to_string())
328}

330/// Parse a URL string into an Endpoint (host + port).
331fn parse_url(url: &str) -> Option<Endpoint> {
332    let url = url
333        .strip_prefix("http://")
334        .or_else(|| url.strip_prefix("https://"))
335        .unwrap_or(url);
336    let (host, port) = url.split_once(':')?;
337    let port: u16 = port.parse().ok()?;
338    Some(Endpoint {
339        address: host.to_string(),
340        port,
341        weight: 1,
342        region: None,
343        zone: None,
344    })
345}
