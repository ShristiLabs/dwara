1//! SCALE-12 (#192): Remote/signed plugin registry CLI.
2//!
3//! Provides `dwara plugin search` and `dwara plugin install` commands.
4//! The registry is a simple HTTP endpoint that serves a JSON manifest
5//! of available plugins and their .wasm artifacts. The CLI downloads
6//! artifacts, verifies their SHA-256 digest, and writes them to the
7//! specified output directory.
8//!
9//! This is a lean-dependency implementation: instead of embedding an
10//! HTTPS client in the CLI (which would require adding TLS
11//! dependencies), it uses `curl` as an external tool for HTTP(S)
12//! fetches. This is consistent with the project's approach in #187
13//! (Kafka REST Proxy) and #190 (external Parquet conversion). The
14//! operator is expected to have `curl` installed (available on all
15//! major platforms). The `sha2` crate (already in the workspace
16//! dependency tree) is used for digest verification.
17
18use sha2::{Digest, Sha256};
19
20/// The default plugin registry URL (the dwara community registry,
21/// served as a static site from the shristilabs/dwara-plugins repo;
22/// registry.dwara.dev will CNAME here).
23const DEFAULT_REGISTRY: &str = "https://shristilabs.github.io/dwara-plugins";
24
25/// The result of a successful `plugin install` command.
26#[derive(Debug)]
27pub struct InstallResult {
28    pub name: String,
29    pub path: String,
30    pub digest: String,
31}
32
33/// Resolve the registry URL: CLI arg > env var > default.
34/// Shared with the publish tooling (`plugin_publish`), which derives
35/// the default artifact URL from the same base, and with the CLI
36/// binary's `plugin publish` dispatch (a separate crate, hence pub).
37pub fn resolve_registry(registry: Option<&str>) -> String {
38    registry
39        .map(|s| s.to_string())
40        .or_else(|| std::env::var("DWARA_PLUGIN_REGISTRY").ok())
41        .unwrap_or_else(|| DEFAULT_REGISTRY.to_string())
42}

44/// `dwara plugin search [--registry URL] [query]`
45///
46/// Fetches the registry's manifest (a JSON array of plugin entries)
47/// and filters by the optional query (substring match on plugin
48/// names). Prints the matching plugin names, versions, and digests.
49pub fn search(registry: Option<&str>, query: Option<&str>) -> Result<String, String> {
50    let base = resolve_registry(registry);
51    let manifest_url = format!("{}/manifest.json", base.trim_end_matches('/'));
52    let body = http_get_text(&manifest_url)?;
53    let plugins: Vec<serde_json::Value> =
54        serde_json::from_str(&body).map_err(|e| format!("invalid manifest JSON: {e}"))?;
55    let mut out = String::new();
56    for p in &plugins {
57        let name = p
58            .get("name")
59            .and_then(|v| v.as_str())
60            .unwrap_or("(unknown)");
61        if let Some(q) = query {
62            if !name.to_lowercase().contains(&q.to_lowercase()) {
63                continue;
64            }
65        }
66        let version = p
67            .get("version")
68            .and_then(|v| v.as_str())
69            .unwrap_or("(unknown)");
70        let digest = p
71            .get("digest")
72            .and_then(|v| v.as_str())
73            .unwrap_or("(unknown)");
74        out.push_str(&format!("{name}\t{version}\t{digest}\n"));
75    }
76    if out.is_empty() {
77        Ok("(no plugins found)".to_string())
78    } else {
79        Ok(out)
80    }
81}

83/// `dwara plugin install <name> [--version V] [--registry URL] [--digest HASH] [-o DIR]`
84///
85/// Fetches the plugin's manifest entry — the highest semantic version
86/// listed for `name`, or the exact `--version` when given (see
87/// [`select_entry`]) — downloads the .wasm artifact, verifies its
88/// SHA-256 digest, and writes it to `<dir>/<name>.wasm`.
89pub fn install(
90    name: &str,
91    version: Option<&str>,
92    registry: Option<&str>,
93    digest: Option<&str>,
94    dir: &str,
95) -> Result<InstallResult, String> {
96    let base = resolve_registry(registry);
97    let manifest_url = format!("{}/manifest.json", base.trim_end_matches('/'));
98    let body = http_get_text(&manifest_url)?;
99    let plugins: Vec<serde_json::Value> =
100        serde_json::from_str(&body).map_err(|e| format!("invalid manifest JSON: {e}"))?;
101    let entry = select_entry(&plugins, name, version)?;
102    let expected_digest = digest
103        .map(|s| s.to_string())
104        .or_else(|| {
105            entry
106                .get("digest")
107                .and_then(|v| v.as_str())
108                .map(|s| s.to_string())
109        })
110        .ok_or_else(|| {
111            format!("no digest provided and registry manifest has no digest for '{name}'")
112        })?;
113    let artifact_url = entry
114        .get("url")
115        .and_then(|v| v.as_str())
116        .map(|s| s.to_string())
117        .unwrap_or_else(|| format!("{}/{}.wasm", base.trim_end_matches('/'), name));
118    let wasm_bytes = http_get_bytes(&artifact_url)?;
119    let actual_digest = sha256_hex(&wasm_bytes);
120    if actual_digest != expected_digest {
121        return Err(format!(
122            "digest mismatch: expected {expected_digest}, got {actual_digest}"
123        ));
124    }
125    std::fs::create_dir_all(dir).map_err(|e| format!("create dir {dir}: {e}"))?;
126    let path = format!("{dir}/{name}.wasm");
127    std::fs::write(&path, &wasm_bytes).map_err(|e| format!("write {path}: {e}"))?;
128    Ok(InstallResult {
129        name: name.to_string(),
130        path,
131        digest: actual_digest,
132    })
133}

135/// Pick the manifest entry to install for `name`: the entry whose
136/// `version` string equals `version` exactly when one is given,
137/// otherwise the HIGHEST semantic version among the entries listed
138/// for the name (a registry manifest can list several versions of
139/// one plugin; the first entry with a matching name must not
140/// silently win). Errors list the available versions so an exact
141/// `--version` retry is one copy-paste away.
142pub fn select_entry<'a>(
143    plugins: &'a [serde_json::Value],
144    name: &str,
145    version: Option<&str>,
146) -> Result<&'a serde_json::Value, String> {
147    let matching: Vec<&serde_json::Value> = plugins
148        .iter()
149        .filter(|p| p.get("name").and_then(|v| v.as_str()) == Some(name))
150        .collect();
151    if matching.is_empty() {
152        return Err(format!("plugin '{name}' not found in registry"));
153    }
154    let available: Vec<&str> = matching
155        .iter()
156        .filter_map(|p| p.get("version").and_then(|v| v.as_str()))
157        .collect();
158    if let Some(want) = version {
159        return matching
160            .into_iter()
161            .find(|p| p.get("version").and_then(|v| v.as_str()) == Some(want))
162            .ok_or_else(|| {
163                format!(
164                    "version '{want}' of plugin '{name}' not found in registry; available versions: {}",
165                    available.join(", ")
166                )
167            });
168    }
169    let mut best: Option<(&serde_json::Value, SemVer)> = None;
170    for p in matching {
171        let raw = p.get("version").and_then(|v| v.as_str()).unwrap_or("");
172        let parsed = parse_semver(raw).map_err(|e| {
173            format!(
174                "cannot pick the highest version of plugin '{name}': {e}; \
175                 available versions: {} (pass --version to choose one exactly)",
176                available.join(", ")
177            )
178        })?;
179        let greater = match &best {
180            None => true,
181            Some((_, prev)) => cmp_semver(&parsed, prev) == std::cmp::Ordering::Greater,
182        };
183        if greater {
184            best = Some((p, parsed));
185        }
186    }
187    best.map(|(p, _)| p)
188        .ok_or_else(|| format!("plugin '{name}' not found in registry"))
189}

191/// A parsed semantic version. Only what ordering needs: the three
192/// numeric core components and the dot-separated pre-release
193/// identifiers. Build metadata (`+...`) is accepted and dropped —
194/// per the semver spec it never affects precedence.
195#[derive(Debug, PartialEq, Eq)]
196struct SemVer {
197    major: u64,
198    minor: u64,
199    patch: u64,
200    pre: Vec<PreId>,
201}
202
203/// One dot-separated pre-release identifier. Numeric identifiers
204/// compare numerically and rank below alphanumeric ones, which
205/// compare ASCII-lexically (semver spec).
206#[derive(Debug, PartialEq, Eq)]
207enum PreId {
208    Num(u64),
209    Alpha(String),
210}
211
212/// Parse `MAJOR.MINOR.PATCH` with an optional `-prerelease` tag and
213/// `+build` metadata (dropped). A minimal hand-rolled parse — the
214/// registry only needs ordering, not a full semver implementation,
215/// so no external semver crate enters the tree.
216fn parse_semver(v: &str) -> Result<SemVer, String> {
217    let core_pre = v.split('+').next().unwrap_or(v);
218    let (core, pre) = match core_pre.split_once('-') {
219        Some((c, p)) => (c, p),
220        None => (core_pre, ""),
221    };
222    let mut parts = core.split('.');
223    let major = parse_core_num(parts.next(), v)?;
224    let minor = parse_core_num(parts.next(), v)?;
225    let patch = parse_core_num(parts.next(), v)?;
226    if parts.next().is_some() {
227        return Err(not_semver(v));
228    }
229    let mut pre_ids = Vec::new();
230    if !pre.is_empty() {
231        for id in pre.split('.') {
232            if id.is_empty() || !id.chars().all(|c| c.is_ascii_alphanumeric() || c == '-') {
233                return Err(not_semver(v));
234            }
235            pre_ids.push(if id.chars().all(|c| c.is_ascii_digit()) {
236                PreId::Num(
237                    id.parse::<u64>()
238                        .map_err(|_| format!("'{v}' has a pre-release number out of range"))?,
239                )
240            } else {
241                PreId::Alpha(id.to_string())
242            });
243        }
244    }
245    Ok(SemVer {
246        major,
247        minor,
248        patch,
249        pre: pre_ids,
250    })
251}
252
253/// One `MAJOR.MINOR.PATCH` component: a non-empty run of digits.
254fn parse_core_num(part: Option<&str>, v: &str) -> Result<u64, String> {
255    let part = part.ok_or_else(|| not_semver(v))?;
256    if part.is_empty() || !part.chars().all(|c| c.is_ascii_digit()) {
257        return Err(not_semver(v));
258    }
259    part.parse::<u64>().map_err(|_| not_semver(v))
260}
261
262fn not_semver(v: &str) -> String {
263    format!("'{v}' is not a semantic version (MAJOR.MINOR.PATCH)")
264}

266/// Semver precedence: the numeric core fields first; a release
267/// outranks any pre-release of the same core; pre-release identifier
268/// lists compare pairwise, longer winning when all preceding are
269/// equal.
270fn cmp_semver(a: &SemVer, b: &SemVer) -> std::cmp::Ordering {
271    a.major
272        .cmp(&b.major)
273        .then_with(|| a.minor.cmp(&b.minor))
274        .then_with(|| a.patch.cmp(&b.patch))
275        .then_with(|| cmp_pre(&a.pre, &b.pre))
276}
277
278/// Pre-release precedence: empty (a release) beats non-empty; equal
279/// heads fall through to the remaining identifiers.
280fn cmp_pre(a: &[PreId], b: &[PreId]) -> std::cmp::Ordering {
281    use std::cmp::Ordering;
282    match (a.is_empty(), b.is_empty()) {
283        (true, true) => Ordering::Equal,
284        (true, false) => Ordering::Greater,
285        (false, true) => Ordering::Less,
286        (false, false) => cmp_pre_id(&a[0], &b[0]).then_with(|| cmp_pre(&a[1..], &b[1..])),
287    }
288}
289
290fn cmp_pre_id(a: &PreId, b: &PreId) -> std::cmp::Ordering {
291    use std::cmp::Ordering;
292    match (a, b) {
293        (PreId::Num(x), PreId::Num(y)) => x.cmp(y),
294        (PreId::Alpha(x), PreId::Alpha(y)) => x.cmp(y),
295        (PreId::Num(_), PreId::Alpha(_)) => Ordering::Less,
296        (PreId::Alpha(_), PreId::Num(_)) => Ordering::Greater,
297    }
298}

300/// HTTP GET via `curl` (external tool). Returns the response body as
301/// text. Uses `curl -sS --fail` for silent-but-error-reporting
302/// behavior.
303fn http_get_text(url: &str) -> Result<String, String> {
304    let output = std::process::Command::new("curl")
305        .arg("-sS")
306        .arg("--fail")
307        .arg("--max-time")
308        .arg("30")
309        .arg(url)
310        .output()
311        .map_err(|e| format!("curl (is curl installed?): {e}"))?;
312    if !output.status.success() {
313        let stderr = String::from_utf8_lossy(&output.stderr);
314        return Err(format!("curl failed: {stderr}"));
315    }
316    String::from_utf8(output.stdout).map_err(|e| format!("invalid UTF-8 in response: {e}"))
317}
318
319/// HTTP GET via `curl` (external tool). Returns the response body as
320/// raw bytes.
321fn http_get_bytes(url: &str) -> Result<Vec<u8>, String> {
322    let output = std::process::Command::new("curl")
323        .arg("-sS")
324        .arg("--fail")
325        .arg("--max-time")
326        .arg("60")
327        .arg(url)
328        .output()
329        .map_err(|e| format!("curl (is curl installed?): {e}"))?;
330    if !output.status.success() {
331        let stderr = String::from_utf8_lossy(&output.stderr);
332        return Err(format!("curl failed: {stderr}"));
333    }
334    Ok(output.stdout)
335}

337/// SHA-256 hex digest of the input bytes.
338/// Shared with the publish tooling (`plugin_publish`), which computes
339/// the registry manifest digest with the same implementation the
340/// install path verifies against.
341pub(crate) fn sha256_hex(data: &[u8]) -> String {
342    let mut hasher = Sha256::new();
343    hasher.update(data);
344    let result = hasher.finalize();
345    let mut hex = String::with_capacity(64);
346    use std::fmt::Write;
347    for byte in result.iter() {
348        write!(&mut hex, "{:02x}", byte).unwrap();
349    }
350    hex
