1//! Plugin publish tooling (DW-164, #282).
2//!
3//! The author-side half of the plugin registry: `dwara plugin keygen`,
4//! `plugin sign`, and `plugin publish`. The reader-side (`plugin
5//! search` / `plugin install`) lives in [`crate::plugin_registry`].
6//!
7//! - `keygen` generates an Ed25519 keypair. The public key (32 bytes,
8//!   hex) goes into registry entries as `public_key`; the private key
9//!   never leaves the author's machine. `--out-dir` writes
10//!   `plugin.pub` (world-readable) and `plugin.key` (owner-only,
11//!   0600), refusing to overwrite existing files.
12//! - `sign` produces a detached Ed25519 signature (64 bytes, hex) over
13//!   the artifact bytes — the exact bytes whose SHA-256 is the
14//!   manifest `digest`, so digest and signature bind to the same
15//!   artifact content.
16//! - `publish` reads the built `.wasm` from disk, computes its
17//!   SHA-256, optionally signs it, and emits a registry manifest entry
18//!   (`{name, version, digest, url}` plus `signature`/`public_key`
19//!   when signed) ready to paste into the registry's `manifest.json`.
20//!   With `--pr` it opens the pull request itself.
21//!
22//! ## Manifest shape and tolerance
23//!
24//! The registry manifest is a JSON array of entries; the install
25//! reader parses it as `Vec<serde_json::Value>`, so extra fields are
26//! tolerated by construction. The typed [`ManifestEntry`] here
27//! deliberately does NOT use `deny_unknown_fields`: this is an
28//! external interchange format, not gateway config — strict-serde's
29//! job is to catch operator typos in `dwara.yaml`, while a reader
30//! that rejects unknown registry fields would break on every future
31//! manifest addition. The merge path ([`merge_manifest`]) therefore
32//! operates on raw JSON values, preserving unknown fields of entries
33//! it does not touch.
34//!
35//! ## Lean-deps pattern
36//!
37//! Like `plugin install` (external `curl`), the `--pr` flow shells out
38//! to the GitHub CLI (`gh`) instead of embedding a GitHub API client:
39//! `gh api` drives the git data endpoints (blob, tree, commit, ref)
40//! and `gh pr create` opens the PR. The flow is best-effort: it tries
41//! a branch on the target repository first and falls back to a fork
42//! (for authors without write access); on any failure the printed
43//! manifest entry is still valid for a manual PR.
44//!
45//! ## Done-when
46//!
47//! An author can go from a built `.wasm` to an open registry PR with
48//! one command, with a digest verified against the artifact on disk.

64/// A freshly generated Ed25519 keypair, hex-encoded.
65#[derive(Debug, Clone)]
66pub struct GeneratedKeypair {
67    /// The public key (verifying key): 32 bytes, 64 hex chars.
68    /// Safe to publish; this is the manifest's `public_key`.
69    pub public_key: String,
70    /// The private key (signing key seed): 32 bytes, 64 hex chars.
71    /// Secret; this signs artifacts.
72    pub private_key: String,
73}

75/// Where `keygen --out-dir` wrote the keypair.
76#[derive(Debug)]
77pub struct WrittenKeypair {
78    /// Path of the public key file (`plugin.pub`).
79    pub public_path: String,
80    /// Path of the private key file (`plugin.key`, mode 0600).
81    pub private_path: String,
82}

84/// One registry manifest entry. See the module docs for the tolerance
85/// rationale (no `deny_unknown_fields`).
86#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
87pub struct ManifestEntry {
88    /// The plugin name (registry-wide identifier).
89    pub name: String,
90    /// The plugin version (the artifact this entry points at).
91    pub version: String,
92    /// The SHA-256 hex digest of the `.wasm` artifact bytes.
93    pub digest: String,
94    /// Where the artifact is hosted.
95    pub url: String,
96    /// Ed25519 signature (hex, 64 bytes) over the artifact bytes.
97    /// Present only for signed plugins.
98    #[serde(skip_serializing_if = "Option::is_none")]
99    pub signature: Option<String>,
100    /// Ed25519 public key (hex, 32 bytes) that verifies `signature`.
101    /// Present only for signed plugins.
102    #[serde(skip_serializing_if = "Option::is_none")]
103    pub public_key: Option<String>,
104}

106/// `plugin keygen`: generate a fresh Ed25519 keypair. `OsRng` cannot
107/// fail (the OS entropy source is infallible), so neither can this.
108pub fn keygen() -> GeneratedKeypair {
109    let signing = SigningKey::generate(&mut rand_core::OsRng);
110    GeneratedKeypair {
111        public_key: to_hex(signing.verifying_key().as_bytes()),
112        private_key: to_hex(signing.as_bytes()),
113    }
114}

116/// `plugin keygen --out-dir`: write the keypair to `<dir>/plugin.pub`
117/// and `<dir>/plugin.key`. The private key is owner-only (0600 on
118/// Unix). Existing files are never overwritten — a silent overwrite
119/// could strand artifacts signed with the old key.
120pub fn write_keypair(dir: &str) -> Result<WrittenKeypair, String> {
121    let dir_path = std::path::Path::new(dir);
122    std::fs::create_dir_all(dir_path).map_err(|e| format!("cannot create dir {dir}: {e}"))?;
123    let public_path = dir_path.join("plugin.pub");
124    let private_path = dir_path.join("plugin.key");
125    for path in [&public_path, &private_path] {
126        if path.exists() {
127            return Err(format!(
128                "refusing to overwrite existing {} (move it away or pick another --out-dir)",
129                path.display()
130            ));
131        }
132    }
133    let kp = keygen();
134    std::fs::write(&public_path, &kp.public_key)
135        .map_err(|e| format!("cannot write {}: {e}", public_path.display()))?;
136    std::fs::write(&private_path, &kp.private_key)
137        .map_err(|e| format!("cannot write {}: {e}", private_path.display()))?;
138    restrict_permissions(&private_path)?;
139    Ok(WrittenKeypair {
140        public_path: public_path.display().to_string(),
141        private_path: private_path.display().to_string(),
142    })
143}

145/// Restrict a file to owner-only (0600). A no-op on non-Unix platforms
146/// (the CLI's deploy targets are Unix; the key content is still only
147/// what the operator's umask allows).
148#[cfg(unix)]
149fn restrict_permissions(path: &std::path::Path) -> Result<(), String> {
150    use std::os::unix::fs::PermissionsExt as _;
151    std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600))
152        .map_err(|e| format!("cannot set permissions on {}: {e}", path.display()))
153}
154
155#[cfg(not(unix))]
156fn restrict_permissions(_path: &std::path::Path) -> Result<(), String> {
157    Ok(())
158}

160/// Load a signing key from a path to a hex key file (as written by
161/// `keygen --out-dir`) or from a literal hex string on the command
162/// line. This keeps `--key` scriptable either way.
163pub fn load_signing_key(key: &str) -> Result<SigningKey, String> {
164    let hex = if std::path::Path::new(key).exists() {
165        std::fs::read_to_string(key)
166            .map_err(|e| format!("cannot read key file {key}: {e}"))?
167            .trim()
168            .to_string()
169    } else {
170        key.trim().to_string()
171    };
172    let bytes = parse_hex(&hex).map_err(|e| format!("invalid signing key: {e}"))?;
173    let bytes: [u8; 32] = bytes
174        .try_into()
175        .map_err(|_| "invalid signing key: must be 32 bytes (64 hex chars)".to_string())?;
176    Ok(SigningKey::from_bytes(&bytes))
177}

179/// The hex public key of a signing key (what goes into the manifest's
180/// `public_key` field).
181pub fn public_key_hex(key: &SigningKey) -> String {
182    to_hex(key.verifying_key().as_bytes())
183}

185/// `plugin sign`: Ed25519 signature over the artifact bytes, hex.
186/// Deterministic for a given key and artifact.
187pub fn sign_artifact(key: &SigningKey, artifact: &[u8]) -> String {
188    to_hex(&key.sign(artifact).to_bytes())
189}

191/// Verify a hex signature against the artifact bytes with a hex
192/// public key — the same check the gateway-side verification performs
193/// on load. Ok(()) when the signature matches.
194pub fn verify_artifact(
195    public_key_hex: &str,
196    signature_hex: &str,
197    artifact: &[u8],
198) -> Result<(), String> {
199    let key_bytes = parse_hex(public_key_hex).map_err(|e| format!("invalid public key: {e}"))?;
200    let key_bytes: [u8; 32] = key_bytes
201        .try_into()
202        .map_err(|_| "invalid public key: must be 32 bytes (64 hex chars)".to_string())?;
203    let verifying =
204        VerifyingKey::from_bytes(&key_bytes).map_err(|e| format!("invalid public key: {e}"))?;
205    let sig_bytes = parse_hex(signature_hex).map_err(|e| format!("invalid signature: {e}"))?;
206    let sig_bytes: [u8; 64] = sig_bytes
207        .try_into()
208        .map_err(|_| "invalid signature: must be 64 bytes (128 hex chars)".to_string())?;
209    verifying
210        .verify(artifact, &Signature::from_bytes(&sig_bytes))
211        .map_err(|e| format!("signature verification failed: {e}"))
212}

214/// Build a manifest entry from the artifact bytes read from disk.
215///
216/// The digest is computed from `artifact` itself (the on-disk bytes —
217/// the same bytes a registry download verifies), the URL defaults to
218/// `<registry_base>/<name>-<version>.wasm`, and when `signing_key` is
219/// given the entry carries `signature` + `public_key`, self-verified
220/// against the artifact before the entry is returned so a broken
221/// signature can never reach a manifest.
222pub fn build_entry(
223    artifact: &[u8],
224    name: &str,
225    version: &str,
226    url: Option<&str>,
227    registry_base: &str,
228    signing_key: Option<&SigningKey>,
229) -> Result<ManifestEntry, String> {
230    if artifact.is_empty() {
231        return Err("artifact is empty (did the build produce a .wasm?)".to_string());
232    }
233    validate_name(name)?;
234    validate_version(version)?;
235    let digest = crate::plugin_registry::sha256_hex(artifact);
236    let url = match url {
237        Some(u) => u.to_string(),
238        None => format!(
239            "{}/{name}-{version}.wasm",
240            registry_base.trim_end_matches('/')
241        ),
242    };
243    let (signature, public_key) = match signing_key {
244        Some(key) => {
245            let sig = sign_artifact(key, artifact);
246            let pk = public_key_hex(key);
247            verify_artifact(&pk, &sig, artifact)?;
248            (Some(sig), Some(pk))
249        }
250        None => (None, None),
251    };
252    Ok(ManifestEntry {
253        name: name.to_string(),
254        version: version.to_string(),
255        digest,
256        url,
257        signature,
258        public_key,
259    })
260}

262/// Serialize an entry as the ready-to-paste JSON object (pretty, one
263/// field per line, keys in serde_json's default alphabetical order —
264/// matching the registry manifest's formatting).
265pub fn entry_json(entry: &ManifestEntry) -> Result<String, String> {
266    let mut out = serde_json::to_string_pretty(entry)
267        .map_err(|e| format!("cannot serialize manifest entry: {e}"))?;
268    out.push('\n');
269    Ok(out)
270}

272/// Merge an entry into a registry manifest (a JSON array).
273///
274/// The existing entry with the same `name` AND `version` is replaced
275/// (re-publishing an identical version is an update, not a duplicate);
276/// any other entry is preserved value-for-value with unknown fields
277/// intact, and entry ORDER is stable (untouched entries keep their
278/// positions in the array). The merge operates on raw JSON values,
279/// not typed entries, so it never drops data it does not understand.
280///
281/// The whole manifest is re-serialized with one consistent format
282/// (serde_json pretty-printed, object keys in its default sorted
283/// order), so untouched entries may reindent or reorder their keys in
284/// the committed file even though their values never change — values
285/// are byte-identical, formatting is not. The consistent formatting
286/// keeps PR diffs limited to the entries that actually changed after
287/// the first such reformat. Returns the full merged manifest
288/// (pretty-printed with a trailing newline).
289pub fn merge_manifest(existing: &str, entry: &ManifestEntry) -> Result<String, String> {
290    let mut arr: Vec<serde_json::Value> = if existing.trim().is_empty() {
291        Vec::new()
292    } else {
293        serde_json::from_str(existing)
294            .map_err(|e| format!("registry manifest is not a JSON array: {e}"))?
295    };
296    let new_value =
297        serde_json::to_value(entry).map_err(|e| format!("cannot serialize entry: {e}"))?;
298    match arr.iter_mut().find(|v| {
299        v.get("name").and_then(serde_json::Value::as_str) == Some(entry.name.as_str())
300            && v.get("version").and_then(serde_json::Value::as_str) == Some(entry.version.as_str())
301    }) {
302        Some(slot) => *slot = new_value,
303        None => arr.push(new_value),
304    }
305    let mut out = serde_json::to_string_pretty(&arr)
306        .map_err(|e| format!("cannot serialize manifest: {e}"))?;
307    out.push('\n');
308    Ok(out)
309}

311/// The branch name for a publish PR: `plugin/<name>-<version>` with
312/// characters git refs reject replaced by `-`.
313pub fn branch_name(name: &str, version: &str) -> String {
314    format!("plugin/{}-{}", sanitize_ref(name), sanitize_ref(version))
315}

317/// Map a string onto the git-ref-safe alphabet [A-Za-z0-9._+-] (`+` is
318/// valid in ref names and common in semver build metadata).
319fn sanitize_ref(s: &str) -> String {
320    s.chars()
321        .map(|c| {
322            if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.' | '+') {
323                c
324            } else {
325                '-'
326            }
327        })
328        .collect()
329}

331/// Registry plugin names follow the scaffold's crate-name rules so a
332/// `plugin new` project can publish under its own name.
333fn validate_name(name: &str) -> Result<(), String> {
334    if name.is_empty() {
335        return Err("plugin name cannot be empty".to_string());
336    }
337    if name.len() > 64 {
338        return Err("plugin name cannot be longer than 64 characters".to_string());
339    }
340    let first = name.chars().next().unwrap();
341    if !first.is_ascii_alphabetic() && first != '_' {
342        return Err(format!(
343            "plugin name must start with a letter or underscore, got '{first}'"
344        ));
345    }
346    if !name
347        .chars()
348        .all(|c| c.is_ascii_alphanumeric() || c == '_' || c == '-')
349    {
350        return Err(
351            "plugin name can only contain letters, digits, underscores, and hyphens".to_string(),
352        );
353    }
354    Ok(())
355}

357/// Versions accept the semver character set (digits, letters, `.`,
358/// `+`, `-`). A full semver parse is deliberately not enforced: the
359/// registry treats versions as opaque strings, and the URL derives
360/// from them verbatim.
361fn validate_version(version: &str) -> Result<(), String> {
362    if version.is_empty() {
363        return Err("version cannot be empty".to_string());
364    }
365    if !version
366        .chars()
367        .all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '+' | '-'))
368    {
369        return Err("version can only contain letters, digits, '.', '+', and '-'".to_string());
370    }
371    Ok(())
372}

374/// `OWNER/NAME` GitHub repository reference (e.g. the default
375/// `shristilabs/dwara-plugins`).
376fn validate_repo(repo: &str) -> Result<(), String> {
377    let (owner, name) = repo.split_once('/').ok_or_else(|| {
378        format!("repository must be OWNER/NAME (e.g. {DEFAULT_REGISTRY_REPO}), got '{repo}'")
379    })?;
380    for part in [owner, name] {
381        if part.is_empty()
382            || !part
383                .chars()
384                .all(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.'))
385        {
386            return Err(format!("invalid repository reference '{repo}'"));
387        }
388    }
389    Ok(())
390}

392/// A git branch name safe to pass through `gh api` paths and args.
393fn validate_branch(branch: &str) -> Result<(), String> {
394    if branch.is_empty()
395        || branch.starts_with('-')
396        || !branch
397            .chars()
398            .all(|c| c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | '.' | '/'))
399    {
400        return Err(format!("invalid branch name '{branch}'"));
401    }
402    Ok(())
403}

405/// Open the registry PR for an entry, best-effort via the `gh` CLI.
406///
407/// Flow: fetch the registry's current `manifest.json`, merge the
408/// entry, commit the merged manifest to a `plugin/<name>-<version>`
409/// branch (git data API: blob -> tree -> commit -> ref), and open a
410/// PR. The branch is created on the target repository first; if that
411/// fails (usually no write access), the same commit lands on a fork
412/// (`gh repo fork`) and the PR head becomes `<login>:<branch>`.
413/// Returns the PR URL.
414pub fn open_registry_pr(repo: &str, base: &str, entry: &ManifestEntry) -> Result<String, String> {
415    validate_repo(repo)?;
416    validate_branch(base)?;
417    if !gh_available() {
418        return Err(
419            "the GitHub CLI (gh) is not installed or not on PATH; install it from \
420             https://cli.github.com/ or open the PR manually with the printed entry"
421                .to_string(),
422        );
423    }
424    let branch = branch_name(&entry.name, &entry.version);
425    let current = fetch_manifest(repo)?;
426    let merged = merge_manifest(&current, entry)?;
427
428    match commit_manifest_branch(repo, base, &branch, &merged, entry) {
429        Ok(()) => create_pr(repo, &branch, base, entry),
430        Err(direct_err) => {
431            let fork = ensure_fork(repo).map_err(|e| {
432                format!("cannot push to {repo} ({direct_err}) and cannot fork: {e}")
433            })?;
434            commit_manifest_branch(&fork, base, &branch, &merged, entry).map_err(|e| {
435                format!("cannot push to {repo} ({direct_err}); pushing to fork {fork} failed: {e}")
436            })?;
437            let fork_owner = fork.split_once('/').map(|(o, _)| o).unwrap_or_default();
438            let head = format!("{fork_owner}:{branch}");
439            create_pr(repo, &head, base, entry)
440        }
441    }
442}

444/// Fetch the registry repository's current manifest via `gh api`
445/// (raw accept header). The repository itself is probed first: a 404
446/// on `repos/{repo}` means a bad (or inaccessible) `--repo` and fails
447/// fast with the repo name — GitHub's stderr for a missing repo and a
448/// missing manifest path is identical, and treating the former as
449/// "fresh registry" would push the failure into a confusing fork
450/// fallback. Only then does a 404 on the manifest path start a fresh
451/// `[]` — the first publish into a new registry.
452fn fetch_manifest(repo: &str) -> Result<String, String> {
453    if let Err(e) = gh_api(&[&format!("repos/{repo}")], None) {
454        if e.contains("404") {
455            return Err(format!(
456                "repository '{repo}' not found or not accessible (is --repo correct?)"
457            ));
458        }
459        return Err(e);
460    }
461    match gh_api(
462        &[
463            &format!("repos/{repo}/contents/{MANIFEST_PATH}"),
464            "-H",
465            "Accept: application/vnd.github.raw",
466        ],
467        None,
468    ) {
469        Ok(body) => Ok(body),
470        Err(e) if e.contains("404") => Ok("[]".to_string()),
471        Err(e) => Err(e),
472    }
473}

475/// Commit the merged manifest to a branch on `repo` using the git
476/// data API (blob -> tree on top of the base tree -> commit with the
477/// base as parent -> ref creation). All payloads are JSON built with
478/// serde_json, so manifest content with any characters is escaped
479/// safely; `encoding: utf-8` sends the JSON text without base64.
480///
481/// On a re-run after partial success (branch pushed, PR step failed),
482/// ref creation fails with "Reference already exists"; when the
483/// branch is ours (its name matches the generated one), the existing
484/// ref is force-moved to the fresh commit instead — the branch is a
485/// throwaway PR head this tool owns, and this keeps re-runs from
486/// falling into the fork fallback. Any other ref-creation error
487/// (permissions, protected refs) propagates to the caller, where the
488/// fork fallback handles it.
489fn commit_manifest_branch(
490    repo: &str,
491    base: &str,
492    branch: &str,
493    merged_manifest: &str,
494    entry: &ManifestEntry,
495) -> Result<(), String> {
496    let base_sha = gh_api(
497        &[
498            &format!("repos/{repo}/git/ref/heads/{base}"),
499            "--jq",
500            ".object.sha",
501        ],
502        None,
503    )?;
504    let base_tree = gh_api(
505        &[
506            &format!("repos/{repo}/git/commits/{base_sha}"),
507            "--jq",
508            ".tree.sha",
509        ],
510        None,
511    )?;
512    let blob = serde_json::json!({"content": merged_manifest, "encoding": "utf-8"});
513    let blob_sha = gh_api(
514        &[
515            &format!("repos/{repo}/git/blobs"),
516            "--input",
517            "-",
518            "--jq",
519            ".sha",
520        ],
521        Some(&blob.to_string()),
522    )?;
523    let tree = serde_json::json!({
524        "base_tree": base_tree,
525        "tree": [{
526            "path": MANIFEST_PATH,
527            "mode": "100644",
528            "type": "blob",
529            "sha": blob_sha,
530        }],
531    });
532    let tree_sha = gh_api(
533        &[
534            &format!("repos/{repo}/git/trees"),
535            "--input",
536            "-",
537            "--jq",
538            ".sha",
539        ],
540        Some(&tree.to_string()),
541    )?;
542    let commit = serde_json::json!({
543        "message": commit_message(entry),
544        "tree": tree_sha,
545        "parents": [base_sha],
546    });
547    let commit_sha = gh_api(
548        &[
549            &format!("repos/{repo}/git/commits"),
550            "--input",
551            "-",
552            "--jq",
553            ".sha",
554        ],
555        Some(&commit.to_string()),
556    )?;
557    let reference = serde_json::json!({"ref": format!("refs/heads/{branch}"), "sha": commit_sha});
558    if let Err(e) = gh_api(
559        &[&format!("repos/{repo}/git/refs"), "--input", "-"],
560        Some(&reference.to_string()),
561    ) {
562        if e.contains("Reference already exists")
563            && branch == branch_name(&entry.name, &entry.version)
564        {
565            let patch = serde_json::json!({"sha": commit_sha, "force": true});
566            return gh_api(
567                &[
568                    &format!("repos/{repo}/git/refs/heads/{branch}"),
569                    "--input",
570                    "-",
571                    "-X",
572                    "PATCH",
573                ],
574                Some(&patch.to_string()),
575            )
576            .map(|_| ());
577        }
578        return Err(e);
579    }
580    Ok(())
581}

583/// Ensure a fork of `repo` exists for the authenticated user and
584/// return its full name (`<login>/<name>`). `gh repo fork` is
585/// idempotent (an existing fork succeeds), so this doubles as the
586/// permissions fallback's setup step.
587fn ensure_fork(repo: &str) -> Result<String, String> {
588    gh(&["repo", "fork", repo, "--clone=false"])?;
589    let login = gh(&["api", "user", "--jq", ".login"])?;
590    let short = repo.split_once('/').map(|(_, n)| n).unwrap_or_default();
591    Ok(format!("{login}/{short}"))
592}

594/// Open the pull request and return its URL.
595fn create_pr(repo: &str, head: &str, base: &str, entry: &ManifestEntry) -> Result<String, String> {
596    gh(&[
597        "pr",
598        "create",
599        "--repo",
600        repo,
601        "--head",
602        head,
603        "--base",
604        base,
605        "--title",
606        &format!("plugin: publish {} {}", entry.name, entry.version),
607        "--body",
608        &pr_body(entry),
609    ])
610}

618/// (name, version, digest, URL, signature state). No secrets.

621        (Some(sig), Some(pk)) => format!("yes (Ed25519; public key {pk})\nsignature: {sig}"),

628         | digest | `{digest}` |\n| url | {url} |\n| signed | {signed} |\n\n\

631        version = entry.version,
632        digest = entry.digest,
633        url = entry.url,

638/// Whether the `gh` CLI is installed and runnable.
639fn gh_available() -> bool {
640    std::process::Command::new("gh")
641        .arg("--version")
642        .output()
643        .is_ok_and(|out| out.status.success())
644}

646/// Run a `gh` subcommand and return its trimmed stdout.
647fn gh(args: &[&str]) -> Result<String, String> {
648    let output = std::process::Command::new("gh")
649        .args(args)
650        .output()
651        .map_err(|e| format!("gh (is the GitHub CLI installed?): {e}"))?;
652    if !output.status.success() {
653        let stderr = String::from_utf8_lossy(&output.stderr).trim().to_string();
654        return Err(format!(
655            "gh {} failed: {stderr}",
656            args.first().unwrap_or(&"")
657        ));
658    }
659    Ok(String::from_utf8_lossy(&output.stdout).trim().to_string())
660}

662/// Run `gh api` with optional JSON request bodies piped over stdin
663/// (`--input -`). Stdin is written then dropped (EOF) before the
664/// output is collected; a broken pipe (gh exiting early on an API
665/// error) surfaces as gh's own stderr, not a write error.
666fn gh_api(args: &[&str], body: Option<&str>) -> Result<String, String> {
667    let mut cmd = std::process::Command::new("gh");
668    cmd.arg("api").args(args);
669    let output = match body {
670        None => cmd
671            .output()
672            .map_err(|e| format!("gh (is the GitHub CLI installed?): {e}"))?,
673        Some(body) => {
674            use std::io::Write as _;
675            cmd.stdin(std::process::Stdio::piped());
676            let mut child = cmd
677                .spawn()
678                .map_err(|e| format!("gh (is the GitHub CLI installed?): {e}"))?;
679            {
680                let mut stdin = child.stdin.take().unwrap();
681                // A write failure means gh exited before reading (API
682                // error); fall through so its stderr is reported.
683                let _ = stdin.write_all(body.as_bytes());
684            }
685            child
686                .wait_with_output()
687                .map_err(|e| format!("gh api: {e}"))?
688        }
689    };
690    if !output.status.success() {
691        let stderr = String::from_utf8_lossy(&output.stderr).trim().to_string();
692        return Err(format!("gh api failed: {stderr}"));
693    }
694    Ok(String::from_utf8_lossy(&output.stdout).trim().to_string())
695}

697/// Lowercase hex encoding of the input bytes.
698fn to_hex(bytes: &[u8]) -> String {
699    use std::fmt::Write;
700    let mut hex = String::with_capacity(bytes.len() * 2);
701    for byte in bytes {
702        write!(&mut hex, "{byte:02x}").expect("hex write is infallible");
703    }
704    hex
705}
706
707/// Decode a (trimmed) hex string to bytes.
708fn parse_hex(s: &str) -> Result<Vec<u8>, String> {
709    let s = s.trim();
710    if !s.len().is_multiple_of(2) {
711        return Err("hex string must have an even number of characters".to_string());
712    }
713    let mut out = Vec::with_capacity(s.len() / 2);
714    for pair in s.as_bytes().chunks(2) {
715        let hi = hex_val(pair[0])?;
716        let lo = hex_val(pair[1])?;
717        out.push(hi << 4 | lo);
718    }
719    Ok(out)
720}
721
722/// The value of a single hex digit.
723fn hex_val(c: u8) -> Result<u8, String> {
724    match c {
725        b'0'..=b'9' => Ok(c - b'0'),
726        b'a'..=b'f' => Ok(c - b'a' + 10),
727        b'A'..=b'F' => Ok(c - b'A' + 10),
728        _ => Err(format!("invalid hex character '{}'", c as char)),
729    }
730}
