1//! Example-plugin templates for `plugin new --template` (DW-166, #284).
2//!
3//! `dwara plugin new <name> --template <example>` scaffolds from one
4//! of the four runnable examples in `plugins/examples/` instead of
5//! the hello-world default ([`crate::plugin_scaffold`]). The example
6//! crates are vendored here as template strings only, mirroring
7//! `plugin_scaffold.rs`'s pattern: a released `dwara-cli` binary must
8//! scaffold without a dwara source checkout on disk.
9//!
10//! The templates are the raw-ABI examples (no `proxy-wasm` crate
11//! dependency): `src/abi.rs` binds the proxy-wasm host imports
12//! directly and doubles as a fake host on non-wasm targets, so the
13//! scaffolded project runs `cargo test` on the host offline and
14//! builds for `wasm32-wasip1` with zero dependencies. Vendored files
15//! are verbatim copies of the examples; after changing an example,
16//! re-vendor the file here in the same change.
17//!
18//! ## Substitution
19//!
20//! The example's hyphenated crate name and its underscored lib form
21//! are replaced with the target name's forms everywhere (Cargo.toml,
22//! `use` lines in tests, the `x-plugin-name` stamp, log prefixes, doc
23//! comments). Within one template every occurrence of either form
24//! refers to the crate itself, so a global replace is safe; prose in
25//! the generated README deliberately never names the source example.

137/// The bump allocator the host calls when it must write a return
138/// value (a header value, a buffer chunk) into plugin memory. Routed
139/// through the Rust allocator so it can never collide with plugin
140/// state; the bytes live for the request (each request gets a fresh
141/// plugin instance, so nothing accumulates).
142#[no_mangle]
143pub extern "C" fn proxy_on_memory_allocate(size: i32) -> *mut u8 {
144    use std::alloc::{alloc, Layout};
145    if size <= 0 {
146        return std::ptr::null_mut();
147    }
148    // Align 8 is what the proxy-wasm spec asks for; any positive size
149    // is a valid Layout at that alignment on wasm32.
150    match Layout::from_size_align(size as usize, 8) {
151        Ok(layout) => unsafe { alloc(layout) },
152        Err(_) => std::ptr::null_mut(),
153    }
154}

156/// Serialize headers into the proxy-wasm wire format `proxy_send_http_response`
157/// expects: a sequence of (u32 key length, key, u32 value length, value)
158/// tuples, all lengths big-endian.
159fn serialize_headers(headers: &[(&str, &str)]) -> Vec<u8> {
160    let mut buf = Vec::new();
161    for (key, value) in headers {
162        buf.extend_from_slice(&(key.len() as u32).to_be_bytes());
163        buf.extend_from_slice(key.as_bytes());
164        buf.extend_from_slice(&(value.len() as u32).to_be_bytes());
165        buf.extend_from_slice(value.as_bytes());
166    }
167    buf
168}

197fn get_header(map_type: i32, key: &str) -> Option<String> {
198    #[cfg(target_family = "wasm")]
199    {
200        let mut ptr: i32 = 0;
201        let mut size: i32 = 0;
202        let status = unsafe {
203            proxy_get_header_map_value(

217    #[cfg(not(target_family = "wasm"))]
218    {

220    }

251/// Read a host buffer. `max_size` should be the size the phase
252/// callback received (or the plugin configuration size in
253/// `proxy_on_configure`) so the read never exceeds what exists.
254pub fn get_buffer(buffer_type: i32, start: i32, max_size: i32) -> Option<Vec<u8>> {
255    #[cfg(target_family = "wasm")]
256    {
257        let mut ptr: i32 = 0;
258        let mut size: i32 = 0;
259        let status =
260            unsafe { proxy_get_buffer_bytes(buffer_type, start, max_size, &mut ptr, &mut size) };
261        if status != 0 {
262            // The host refused the read (no such buffer, bad range):
263            // report absence, mirroring the fake host. Callers fail
264            // closed on None (e.g. `proxy_on_configure` sees an empty
265            // config and refuses to activate).
266            return None;
267        }
268        if size <= 0 {
269            return Some(Vec::new());
270        }
271        Some(unsafe { std::slice::from_raw_parts(ptr as *const u8, size as usize) }.to_vec())
272    }
273    #[cfg(not(target_family = "wasm"))]
274    {
275        fake::get_buffer(buffer_type, start, max_size)
276    }
277}

302/// Send a local response and end the stream: the gateway returns
303/// `status`/`headers`/`body` verbatim and never dials the upstream.
304/// The phase callback must return `ACTION_END_STREAM` after this.
305///
306/// Hostcall note: this imports `proxy_send_http_response`, dwara's
307/// original name and argument order. The proxy-wasm spec hostcall is
308/// `proxy_send_local_response` with a different parameter list; the
309/// dwara host links both spellings, other hosts may link only the
310/// spec one.
311pub fn send_http_response(status: u16, headers: &[(&str, &str)], body: &[u8]) {
312    #[cfg(target_family = "wasm")]
313    {
314        let wire = serialize_headers(headers);
315        unsafe {
316            proxy_send_http_response(
317                status as i32,
318                if wire.is_empty() {
319                    std::ptr::null()
320                } else {
321                    wire.as_ptr()
322                },
323                wire.len() as i32,
324                if body.is_empty() {
325                    std::ptr::null()
326                } else {
327                    body.as_ptr()
328                },
329                body.len() as i32,
330                std::ptr::null(),
331                0,
332            );
333        }
334    }
335    #[cfg(not(target_family = "wasm"))]
336    {
337        fake::send_http_response(status, headers, body);
338    }
339}

508/// The shared flat-JSON config reader (`src/json.rs`), byte-identical
509/// in the three examples that parse a plugin config.
510const JSON_RS: &str = r#"//! A deliberately tiny flat-JSON reader for plugin configuration.
511//!
512//! dwara hands a plugin's `config:` string to `proxy_on_configure`
513//! as raw bytes; the convention is a small JSON object the plugin
514//! parses itself. These example plugins depend on nothing, so instead
515//! of pulling a JSON crate into the wasm module they use this
516//! purpose-built reader, which supports exactly what plugin configs
517//! need: a flat object whose values are strings or arrays of strings.
518//!
519//! Supported: `{ "key": "value", "key2": ["a", "b"] }` with standard
520//! whitespace, empty arrays, empty objects, and the string escapes
521//! `\"` `\\` `\/` `\b` `\f` `\n` `\r` `\t`. Anything else (nested
522//! objects, numbers, booleans, null, `\uXXXX`) is rejected with an
523//! error, so a malformed config fails closed at configure time
524//! instead of misbehaving per request.
525
526// A shared helper file: not every consumer uses every accessors.
527#![allow(dead_code)]

529/// A parsed JSON value: a string or an array of strings.
530#[derive(Debug, Clone, PartialEq, Eq)]
531pub enum JsonValue {
532    Str(String),
533    Arr(Vec<String>),
534}
535
536impl JsonValue {
537    /// The value as a string, or `None` when it is an array.
538    pub fn as_str(&self) -> Option<&str> {
539        match self {
540            JsonValue::Str(s) => Some(s),
541            JsonValue::Arr(_) => None,
542        }
543    }
544
545    /// The value as an array of strings, or `None` when it is a
546    /// string.
547    pub fn as_arr(&self) -> Option<&[String]> {
548        match self {
549            JsonValue::Str(_) => None,
550            JsonValue::Arr(items) => Some(items),
551        }
552    }
553}

555/// Parse a flat JSON object into ordered key/value pairs. Duplicate
556/// keys keep the first occurrence (last-write-wins is a footgun for
557/// security-relevant config).
558pub fn parse_flat_object(bytes: &[u8]) -> Result<Vec<(String, JsonValue)>, String> {
559    let mut parser = Parser { bytes, pos: 0 };
560    parser.skip_ws();
561    parser.expect(b'{')?;
562    let mut pairs = Vec::new();
563    parser.skip_ws();
564    if parser.peek() == Some(b'}') {
565        parser.pos += 1;
566    } else {
567        loop {
568            parser.skip_ws();
569            let key = parser.parse_string()?;
570            parser.skip_ws();
571            parser.expect(b':')?;
572            parser.skip_ws();
573            let value = parser.parse_value()?;
574            if !pairs.iter().any(|(k, _): &(String, JsonValue)| *k == key) {
575                pairs.push((key, value));
576            }
577            parser.skip_ws();
578            match parser.peek() {
579                Some(b',') => {
580                    parser.pos += 1;
581                }
582                Some(b'}') => {
583                    parser.pos += 1;
584                    break;
585                }
586                _ => return Err(format!("expected ',' or '}}' at byte {}", parser.pos)),
587            }
588        }
589    }
590    parser.skip_ws();
591    if parser.pos != bytes.len() {
592        return Err(format!("trailing data at byte {}", parser.pos));
593    }
594    Ok(pairs)
595}
596
597struct Parser<'a> {
598    bytes: &'a [u8],
599    pos: usize,
600}
601
602impl<'a> Parser<'a> {
603    fn peek(&self) -> Option<u8> {
604        self.bytes.get(self.pos).copied()
605    }
606
607    fn expect(&mut self, byte: u8) -> Result<(), String> {
608        match self.peek() {
609            Some(b) if b == byte => {
610                self.pos += 1;
611                Ok(())
612            }
613            _ => Err(format!("expected '{}' at byte {}", byte as char, self.pos)),
614        }
615    }
616
617    fn skip_ws(&mut self) {
618        while matches!(
619            self.peek(),
620            Some(b' ') | Some(b'\t') | Some(b'\n') | Some(b'\r')
621        ) {
622            self.pos += 1;
623        }
624    }
625
626    fn parse_string(&mut self) -> Result<String, String> {
627        self.expect(b'"')?;
628        let mut out: Vec<u8> = Vec::new();
629        loop {
630            match self.peek() {
631                None => return Err("unterminated string".to_string()),

686                    self.skip_ws();
687                    match self.peek() {
688                        Some(b',') => {

690                        }
691                        Some(b']') => {
692                            self.pos += 1;
693                            return Ok(JsonValue::Arr(items));
694                        }
695                        _ => {
696                            return Err(format!("expected ',' or ']' at byte {}", self.pos));
697                        }
698                    }
699                }
700            }
701            _ => Err(format!(
702                "expected a string or array of strings at byte {} \
703                 (numbers, booleans, null, and nested objects are not supported)",
704                self.pos
705            )),

869/// Constant-time byte-slice equality: the comparison walks
870/// `max(a.len(), b.len())` bytes and folds every XOR difference, so
871/// the running time does not depend on where the first difference
872/// occurs. Length differences still leak (that is unavoidable and
873/// harmless for token comparison; the token length is not secret).
874pub fn constant_time_eq(a: &[u8], b: &[u8]) -> bool {
875    let len = a.len().max(b.len());
876    // The fold must stay usize-wide: truncating the length XOR to u8
877    // would make lengths differing by a multiple of 256 compare equal
878    // (an all-zero buffer and a 256-zero-byte buffer, for example).
879    let mut diff: usize = a.len() ^ b.len();
880    for i in 0..len {
881        let byte_a = if i < a.len() { a[i] } else { 0 };
882        let byte_b = if i < b.len() { b[i] } else { 0 };
883        diff |= usize::from(byte_a ^ byte_b);
884    }
885    diff == 0
886}

1817/// Replace every non-overlapping occurrence of `literal` with `*`
1818/// repeated to the same length (length-preserving).
1819pub fn replace_literal(body: &mut Vec<u8>, literal: &[u8]) {
1820    if literal.is_empty() || body.len() < literal.len() {
1821        return;
1822    }
1823    let mut i = 0;
1824    while i + literal.len() <= body.len() {
1825        if &body[i..i + literal.len()] == literal {
1826            for byte in &mut body[i..i + literal.len()] {
1827                *byte = b'*';
1828            }
1829            i += literal.len();
1830        } else {
1831            i += 1;
1832        }
1833    }

1837/// groups allowed), keeping the last four digits. Runs are treated
1838/// whole: a run that is too long, too short, or checksum-invalid
1839/// passes through untouched (no partial suffix matching).
1840pub fn mask_card_numbers(body: &mut Vec<u8>) {
1841    let len = body.len();
1842    let mut i = 0;
1843    while i < len {
1844        if !body[i].is_ascii_digit() {
1845            i += 1;
1846            continue;
1847        }

1849        // between groups when another digit follows.

1851        let mut digits: Vec<u8> = Vec::new();
1852        let mut j = i;
1853        while j < len && body[j].is_ascii_digit() {
1854            digit_positions.push(j);
1855            digits.push(body[j]);
1856            j += 1;
1857            if j < len
1858                && (body[j] == b'-' || body[j] == b' ')
1859                && j + 1 < len
1860                && body[j + 1].is_ascii_digit()
1861            {
1862                j += 1;
1863            }
1864        }

1878/// The Luhn checksum (ISO/IEC 7812-1): every second digit from the
1879/// right is doubled (9 subtracted on overflow) and the sum must be a
1880/// multiple of ten. Total over arbitrary bytes: a non-digit input is
1881/// not a card number and returns false (the scanner only feeds this
1882/// ASCII digits, but the function must not rely on that).
1883pub fn luhn_valid(digits: &[u8]) -> bool {
1884    let mut sum: u32 = 0;
1885    for (offset, digit) in digits.iter().rev().enumerate() {
1886        if !digit.is_ascii_digit() {
1887            return false;
1888        }
1889        let mut value = u32::from(digit - b'0');
1890        if offset % 2 == 1 {
1891            value *= 2;
1892            if value > 9 {
1893                value -= 9;
1894            }
1895        }
1896        sum += value;
1897    }
1898    sum % 10 == 0
1899}

2649/// Render every file of the scaffolded project as `(relative path,
2650/// contents)` pairs, with the target crate name substituted
2651/// throughout (see the module docs).
2652pub fn render_files(template: &Template, name: &str) -> Vec<(String, String)> {
2653    let under = name.replace('-', "_");
2654    // The template's own hyphenated and underscored names; every
2655    // occurrence refers to the crate itself, so replace both forms
2656    // globally. The underscored forms are replaced first so a
2657    // hypothetical name whose hyphen form is a substring of another
2658    // template's underscore form cannot interleave (none is today;
2659    // the order makes that structural, not accidental).
2660    let sub = |text: &str| -> String {
2661        let t = text.replace(&template.name.replace('-', "_"), &under);
2662        t.replace(template.name, name)
2663    };
2664
2665    let mut files = vec![
2666        ("Cargo.toml".to_string(), sub(template.cargo_toml)),
2667        (".gitignore".to_string(), "/target\n".to_string()),
2668        ("README.md".to_string(), sub(template.readme)),
2669        ("dwara.yaml".to_string(), sub(template.dwara_yaml)),
2670        ("src/lib.rs".to_string(), sub(template.lib_rs)),
2671        ("src/abi.rs".to_string(), sub(ABI_RS)),
2672    ];
2673    if template.json {
2674        files.push(("src/json.rs".to_string(), sub(JSON_RS)));
2675    }
2676    for (file, contents) in template.tests {
2677        files.push((format!("tests/{file}"), sub(contents)));
2678    }
2679    files
