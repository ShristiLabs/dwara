1//! Operator CLI logic (DW-022, decision 6): the pure halves of the
2//! `dwara-cli` subcommands, kept library-shaped so tests (and future
3//! callers) exercise exactly what the binary runs.
4//!
5//! Exit-code contract (documented, load-bearing):
6//!
7//! - `validate`: 0 = valid config; 1 = ANY schema/parse/validation/
8//!   compile issue (all issues are printed, never fail-fast).
9//! - `lint`: 0 = no advisory warnings; 2 = warnings found; 1 = the file
10//!   could not be parsed/validated at all (fix that first — linting an
11//!   invalid config would report noise). The distinct 2 keeps "your
12//!   config is wrong" and "your config compiles but smells" separable
13//!   in scripts.
14//!
15//! Lint rules are ADVISORY findings beyond validation (see
16//! [`lint_config`]): they flag config that compiles and routes traffic
17//! but likely does not do what the author meant.
18//!
19//! The [`loadgen`] module holds the DW-024 macro load-generator rig
20//! (public for reuse and testing; the `dwara-loadgen` binary is a thin
21//! wrapper over it).
22

61/// Result of validating a config document end to end.
62pub enum ValidateOutcome {
63    /// Parsed, validated, compiled; carries the route count.
64    Valid { routes: usize },
65    /// Every problem found (parse error, validation issues, compile
66    /// errors), human-formatted one per line.
67    Invalid(Vec<String>),
68}
69
70/// Parse + validate + compile-dry-run a config document. This is the
71/// same pipeline the gateway runs at startup and reload.
72pub fn validate_config_text(text: &str) -> ValidateOutcome {
73    let gateway = match parse_gateway(text) {
74        Ok(g) => g,
75        Err(err) => return ValidateOutcome::Invalid(vec![format!("parse failed: {err}")]),
76    };
77    let issues = validate(&gateway);
78    if !issues.is_empty() {
79        return ValidateOutcome::Invalid(
80            issues
81                .iter()
82                .map(|i| format!("config error: {i}"))
83                .collect(),
84        );
85    }
86    match compile(&gateway) {
87        Ok(_) => ValidateOutcome::Valid {
88            routes: gateway.routes.len(),
89        },
90        Err(err) => ValidateOutcome::Invalid(vec![format!("{err}")]),
91    }
92}

94/// Normalize a config document: parse, then re-serialize with
95/// `gateway_to_yaml` (stable field order, defaulted-empty collections
96/// omitted). Round-trip guarantee: the output parses back to the same
97/// typed value.
98pub fn format_config_text(text: &str) -> Result<String, String> {
99    let gateway = parse_gateway(text).map_err(|e| format!("parse failed: {e}"))?;
100    gateway_to_yaml(&gateway).map_err(|e| format!("serialize failed: {e}"))
101}

103/// CFG-03 (#239): migrate a config to the current schema version.
104/// Parses the config, sets the version to `CURRENT_CONFIG_VERSION`,
105/// and re-serializes. Returns the migrated config YAML and a list of
106/// migration notes (what changed, if anything).
107pub fn migrate_config(text: &str) -> Result<(String, Vec<String>), String> {
108    let mut gateway = parse_gateway(text).map_err(|e| format!("parse failed: {e}"))?;
109    let mut notes = Vec::new();
110
111    if gateway.version < dwara_core::config::CURRENT_CONFIG_VERSION {
112        notes.push(format!(
113            "upgraded config version from {} to {}",
114            gateway.version,
115            dwara_core::config::CURRENT_CONFIG_VERSION
116        ));
117        gateway.version = dwara_core::config::CURRENT_CONFIG_VERSION;
118    } else if gateway.version == dwara_core::config::CURRENT_CONFIG_VERSION {
119        notes.push("config is already at the current version".to_string());
120    } else {
121        notes.push(format!(
122            "config version {} is newer than the current schema version {}; downgrading to {}",
123            gateway.version,
124            dwara_core::config::CURRENT_CONFIG_VERSION,
125            dwara_core::config::CURRENT_CONFIG_VERSION
126        ));
127        gateway.version = dwara_core::config::CURRENT_CONFIG_VERSION;
128    }
129
130    let yaml = gateway_to_yaml(&gateway).map_err(|e| format!("serialize failed: {e}"))?;
131    Ok((yaml, notes))
132}

134/// `validate --watch` mode (USA-10, #251): watch the config file and
135/// re-validate on every change, printing a live status line. Runs
136/// until the process is interrupted (Ctrl-C). Returns exit code 0.
137pub fn validate_watch(file: &str, profile: Option<&str>) -> i32 {
138    use notify::Watcher;
139    use std::path::Path;
140    use std::sync::mpsc;
141    use std::time::Duration;
142
143    let path = Path::new(file);
144    let dir = path.parent().unwrap_or(Path::new(".")).to_path_buf();
145    let name = path
146        .file_name()
147        .map(std::ffi::OsStr::to_owned)
148        .unwrap_or_default();
149
150    // Initial validation.
151    run_validate(file, profile);
152
153    // Set up the file watcher (same pattern as dwara-bin's
154    // spawn_file_watcher).
155    let (tx, rx) = mpsc::channel();
156    let mut watcher =
157        match notify::recommended_watcher(move |event: Result<notify::Event, notify::Error>| {
158            let Ok(event) = event else { return };
159            match event.kind {
160                notify::EventKind::Modify(notify::event::ModifyKind::Metadata(_))
161                | notify::EventKind::Access(_)
162                | notify::EventKind::Other => return,
163                _ => {}
164            }
165            let touches = event
166                .paths
167                .iter()
168                .any(|p| p.file_name().is_some_and(|n| n == name));
169            if touches {
170                let _ = tx.send(());
171            }
172        }) {
173            Ok(w) => w,
174            Err(e) => {
175                eprintln!("error: file watch unavailable: {e}");
176                return 1;
177            }
178        };
179
180    if let Err(e) = watcher.watch(&dir, notify::RecursiveMode::NonRecursive) {
181        eprintln!("error: cannot watch {dir:?}: {e}");
182        return 1;
183    }
184
185    // Debounce loop: wait for events, drain bursts, re-validate.
186    let debounce = Duration::from_millis(250);
187    loop {
188        if rx.recv().is_err() {
189            break;
190        }
191        std::thread::sleep(debounce);
192        while rx.try_recv().is_ok() {}
193        run_validate(file, profile);
194        // Keep the watcher alive.
195        let _ = &watcher;
196    }
197    0
198}

200fn run_validate(file: &str, profile: Option<&str>) {
201    use std::io::Write;
202    let timestamp = chrono_like_now();
203    match read_file(file) {
204        Err(e) => {
205            print!("\r[{timestamp}] {file}: read error: {e}                    ");
206            let _ = std::io::stdout().flush();
207        }
208        Ok(text) => {
209            let base_dir = std::path::Path::new(file)
210                .parent()
211                .unwrap_or(std::path::Path::new("."))
212                .to_path_buf();
213            let text = match dwara_core::config::includes::preprocess(&text, &base_dir, profile) {
214                Ok(t) => t,
215                Err(e) => {
216                    print!("\r[{timestamp}] {file}: invalid (preprocess: {e})                    ");
217                    let _ = std::io::stdout().flush();
218                    return;
219                }
220            };
221            match validate_config_text(&text) {
222                ValidateOutcome::Valid { routes } => {
223                    print!("\r[{timestamp}] {file}: ok ({routes} routes)                    ");
224                    let _ = std::io::stdout().flush();
225                }
226                ValidateOutcome::Invalid(issues) => {
227                    let n = issues.len();
228                    print!("\r[{timestamp}] {file}: invalid ({n} issues)                    ");
229                    let _ = std::io::stdout().flush();
230                    eprintln!();
231                    for i in &issues {
232                        eprintln!("  {i}");
233                    }
234                }
235            }
236        }
237    }
238}

244fn chrono_like_now() -> String {
245    let now = std::time::SystemTime::now()
246        .duration_since(std::time::UNIX_EPOCH)
247        .unwrap_or_default();
248    let secs = now.as_secs();
249    let h = (secs / 3600) % 24;
250    let m = (secs / 60) % 60;
251    let s = secs % 60;
252    format!("{h:02}:{m:02}:{s:02}")
253}

255/// Compile both documents and report route/upstream/consumer deltas as
256/// plain text: `+ kind name` (added), `- kind name` (removed), and
257/// `~ kind name` (present in both sides under the same name but with
258/// different content — changed endpoints, timeouts, credentials, ...
259/// detected by comparing per-entity content hashes of the normalized
260/// serialization, so source key order never shows up as a change).
261/// Returns Err when either side is invalid (a diff against a broken
262/// config is meaningless).
263pub fn diff_configs(a_text: &str, b_text: &str) -> Result<String, String> {
264    let parse = |t| parse_gateway(t).map_err(|e| format!("parse failed: {e}"));
265    let a = parse(a_text)?;
266    let b = parse(b_text)?;
267    for (side, gw) in [("a", &a), ("b", &b)] {
268        let issues = validate(gw);
269        if !issues.is_empty() {
270            let mut m = vec![format!("config {side} is invalid:")];
271            m.extend(issues.iter().map(|i| format!("  {i}")));
272            return Err(m.join("\n"));
273        }
274    }
275    // Name -> per-entity content hash, per kind. Validation has run, so
276    // names are unique within each kind and keying is safe.
277    let mut out = String::new();
278    let mut deltas = 0usize;
279    for (label, ha, hb) in [
280        (
281            "route",
282            hash_by_name(
283                a.routes
284                    .iter()
285                    .map(|r| (r.name.as_str(), entity_content_hash(r))),
286            )?,
287            hash_by_name(
288                b.routes
289                    .iter()
290                    .map(|r| (r.name.as_str(), entity_content_hash(r))),
291            )?,
292        ),
293        (
294            "upstream",
295            hash_by_name(
296                a.upstreams
297                    .iter()
298                    .map(|u| (u.name.as_str(), entity_content_hash(u))),
299            )?,
300            hash_by_name(
301                b.upstreams
302                    .iter()
303                    .map(|u| (u.name.as_str(), entity_content_hash(u))),
304            )?,
305        ),
306        (
307            "consumer",
308            hash_by_name(
309                a.consumers
310                    .iter()
311                    .map(|c| (c.name.as_str(), entity_content_hash(c))),
312            )?,
313            hash_by_name(
314                b.consumers
315                    .iter()
316                    .map(|c| (c.name.as_str(), entity_content_hash(c))),
317            )?,
318        ),
319    ] {
320        let names_a: BTreeSet<&str> = ha.keys().copied().collect();
321        let names_b: BTreeSet<&str> = hb.keys().copied().collect();
322        for added in names_b.difference(&names_a) {
323            out.push_str(&format!("+ {label} {added}\n"));
324            deltas += 1;
325        }
326        for removed in names_a.difference(&names_b) {
327            out.push_str(&format!("- {label} {removed}\n"));
328            deltas += 1;
329        }
330        // Same name on both sides: report it only when the content
331        // differs (the defect this fixes — name-set comparison alone
332        // silently passed same-name changes).
333        for changed in names_a.intersection(&names_b) {
334            if ha[changed] != hb[changed] {
335                out.push_str(&format!("~ {label} {changed}\n"));
336                deltas += 1;
337            }
338        }
339    }
340    if deltas == 0 {
341        out.push_str("no route/upstream/consumer differences\n");
342    }
343    Ok(out)
344}

346/// Collect one entity kind's `(name, content hash)` pairs into a map,
347/// propagating any serialization failure.
348fn hash_by_name<'a>(
349    pairs: impl Iterator<Item = (&'a str, Result<u64, String>)>,
350) -> Result<BTreeMap<&'a str, u64>, String> {
351    pairs.map(|(name, hash)| hash.map(|h| (name, h))).collect()
352}

354/// One advisory lint finding: `kind/name: message`.
355/// USA-08 (#230): `auto_fix` carries a suggested command or edit that
356/// would resolve the finding, when one is available. `None` means the
357/// finding is advisory only (no automatic fix is safe).
358pub struct LintWarning {
359    pub kind: &'static str,
360    pub name: String,
361    pub message: String,
362    /// USA-08 (#230): a suggested auto-fix command or description of
363    /// the edit that would resolve this finding. Printed by `dwara lint
364    /// --fix` as a hint. `None` for findings with no safe automatic fix.
365    pub auto_fix: Option<String>,
366}
367
368impl std::fmt::Display for LintWarning {
369    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
370        write!(f, "{}/{}: {}", self.kind, self.name, self.message)?;
371        if let Some(fix) = &self.auto_fix {
372            write!(f, "\n  fix: {fix}")?;
373        }
374        Ok(())
375    }
376}

378/// Advisory lint rules BEYOND validation (a lint-clean config still
379/// validates; these findings are about intent, not correctness):
380///
381/// - `prefix-duplicate`: two prefix routes with the identical pattern;
382///   the earlier-declared one wins the equal-length tie, so the later
383///   route can never match (validation does not see this — longest
384///   prefix ties are resolved silently at lookup time).
385/// - `regex-shadowed-by-exact`: an exact route whose path fully matches
386///   a regex route's pattern; that path is always served by the exact
387///   route (lookup precedence), so the regex can never see it.
388/// - `consumer-unused`: no authorization rule at any level (allowed/
389///   denied lists) and no JWT provider binds the consumer. Static
390///   over-approximation: a consumer can also be used purely by
391///   presenting credentials at runtime, so this is advisory, not an
392///   error.
393/// - `policy-unused`: the policy is attached to no consumer, route,
394///   service, listener, or gateway (global).
395/// - `upstream-unreferenced`: no service targets the upstream.
396///
397/// Listener port duplication across listeners is deliberately NOT a
398/// lint rule — validation already rejects conflicting binds.
399pub fn lint_config(gateway: &Gateway) -> Vec<LintWarning> {
400    let mut warnings = Vec::new();
401
402    // CFG-03 (#239): config version lint. Warn when the config's
403    // version is older than the current schema version (the config may
404    // use deprecated fields or miss new defaults). Also warn when the
405    // version is NEWER than the current schema (the config was written
406    // for a newer dwara version and may use unknown features).
407    if gateway.version < dwara_core::config::CURRENT_CONFIG_VERSION {
408        warnings.push(LintWarning {
409            kind: "config",
410            name: "version".to_string(),
411            message: format!(
412                "config version {} is older than the current schema version {}; run 'dwara migrate' to upgrade",
413                gateway.version,
414                dwara_core::config::CURRENT_CONFIG_VERSION
415            ),
416            auto_fix: Some("dwara migrate <config-file>".to_string()),
417        });
418    } else if gateway.version > dwara_core::config::CURRENT_CONFIG_VERSION {
419        warnings.push(LintWarning {
420            kind: "config",
421            name: "version".to_string(),
422            message: format!(
423                "config version {} is newer than the current schema version {}; the config may use features not supported by this dwara version",
424                gateway.version,
425                dwara_core::config::CURRENT_CONFIG_VERSION
426            ),
427            auto_fix: None,
428        });
429    }
430
431    // prefix-duplicate: equal-length ties resolve to the FIRST declared
432    // prefix route, so later duplicates are dead config.
433    let mut seen_prefixes: BTreeSet<&str> = BTreeSet::new();
434    for route in &gateway.routes {
435        if route.r#match.path.kind == PathMatchKind::Prefix
436            && !seen_prefixes.insert(&route.r#match.path.value)
437        {
438            warnings.push(LintWarning {
439                kind: "route",
440                name: route.name.clone(),
441                message: format!(
442                    "duplicate prefix pattern '{}' (an earlier prefix route wins equal-length ties; this route never matches)",
443                    route.r#match.path.value
444                ),
445                auto_fix: Some(format!(
446                    "remove route '{}' or change its prefix pattern to a unique value",
447                    route.name
448                )),
449            });
450        }
451    }
452
453    // regex-shadowed-by-exact: exact lookup happens first.
454    for route in &gateway.routes {
455        let path = &route.r#match.path;
456        if path.kind != PathMatchKind::Regex {
457            continue;
458        }
459        let Ok(re) = regex::Regex::new(&path.value) else {
460            // Invalid regex is a validation/compile failure, not lint.
461            continue;
462        };
463        let shadowed_by = gateway
464            .routes
465            .iter()
466            .find(|other| {
467                other.r#match.path.kind == PathMatchKind::Exact
468                    && re.is_match_at(&other.r#match.path.value, 0)
469                    && re
470                        .find_at(&other.r#match.path.value, 0)
471                        .is_some_and(|m| m.end() == other.r#match.path.value.len())
472            })
473            .map(|other| other.name.clone());
474        if let Some(exact) = shadowed_by {
475            warnings.push(LintWarning {
476                kind: "route",
477                name: route.name.clone(),
478                message: format!(
479                    "regex '{}' is shadowed for path(s) it matches by exact route '{exact}' \
480                     (exact lookup wins); those paths never reach this route",
481                    path.value
482                ),
483                auto_fix: Some(format!(
484                    "remove exact route '{exact}' or tighten the regex to exclude its paths"
485                )),
486            });
487        }
488    }
489
490    // consumer-unused: referenced by no authorization rule (at ANY
491    // attachment level — route, service, listener, gateway, or another
492    // consumer's block) and bound to no JWT provider.
493    let mut consumers_used: BTreeSet<&str> = BTreeSet::new();
494    let mut authz_blocks: Vec<&dwara_core::config::Authz> = Vec::new();
495    for route in &gateway.routes {
496        if let Some(a) = &route.authorization {
497            authz_blocks.push(a);
498        }
499    }
500    for s in &gateway.services {
501        if let Some(a) = &s.authorization {
502            authz_blocks.push(a);
503        }
504    }
505    for l in &gateway.listeners {
506        if let Some(a) = &l.authorization {
507            authz_blocks.push(a);
508        }
509    }
510    if let Some(a) = &gateway.authorization {
511        authz_blocks.push(a);
512    }
513    for c in &gateway.consumers {
514        if let Some(a) = &c.authorization {
515            authz_blocks.push(a);
516        }
517    }
518    for authz in authz_blocks {
519        for c in authz
520            .allowed_consumers
521            .iter()
522            .chain(&authz.denied_consumers)
523        {
524            consumers_used.insert(c);
525        }
526    }
527    for p in &gateway.jwt_providers {
528        if let Some(c) = &p.consumer {
529            consumers_used.insert(c);
530        }
531    }
532    for c in &gateway.consumers {
533        if !consumers_used.contains(c.name.as_str()) {
534            warnings.push(LintWarning {
535                kind: "consumer",
536                name: c.name.clone(),
537                message:
538                    "unused: referenced by no authorization rule and bound to no jwt provider \
539                          (advisory: runtime credential use is not statically visible)"
540                        .to_string(),
541                auto_fix: Some(format!(
542                    "remove consumer '{}' or add it to an authorization rule / jwt provider",
543                    c.name
544                )),
545            });
546        }
547    }
548
549    // policy-unused (#123: every attachment level counts — consumer,
550    // route, service, listener, and gateway/global).
551    let mut policies_used: BTreeSet<&str> = BTreeSet::new();
552    for r in &gateway.routes {
553        policies_used.extend(r.policies.iter().map(String::as_str));
554    }
555    for s in &gateway.services {
556        policies_used.extend(s.policies.iter().map(String::as_str));
557    }
558    for c in &gateway.consumers {
559        policies_used.extend(c.policies.iter().map(String::as_str));
560    }
561    for l in &gateway.listeners {
562        policies_used.extend(l.policies.iter().map(String::as_str));
563    }
564    policies_used.extend(gateway.global_policies.iter().map(String::as_str));
565    for p in &gateway.policies {
566        if !policies_used.contains(p.name.as_str()) {
567            warnings.push(LintWarning {
568                kind: "policy",
569                name: p.name.clone(),
570                message: "unused: attached to no consumer, route, service, listener, or \
571                          gateway"
572                    .to_string(),
573                auto_fix: Some(format!(
574                    "remove policy '{}' or attach it to a consumer, route, service, listener, or gateway",
575                    p.name
576                )),
577            });
578        }
579    }
580
581    // upstream-unreferenced (DW-040: a split target counts as a
582    // reference too).
583    for u in &gateway.upstreams {
584        let referenced = gateway.services.iter().any(|s| {
585            s.upstream.as_deref() == Some(u.name.as_str())
586                || s.split
587                    .as_ref()
588                    .is_some_and(|sp| sp.targets.iter().any(|t| t.upstream == u.name))
589        });
590        if !referenced {
591            warnings.push(LintWarning {
592                kind: "upstream",
593                name: u.name.clone(),
594                message: "unreferenced: no service targets this upstream".to_string(),
595                auto_fix: Some(format!(
596                    "remove upstream '{}' or add a service that targets it",
597                    u.name
598                )),
599            });
600        }
601    }
602
603    warnings
604}
