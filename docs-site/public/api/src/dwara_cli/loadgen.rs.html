1//! Load generator rig (DW-024, PERF-06); public for reuse and testing.
2//!
3//! A dependency-free (wrk/k6-less) load generator used by
4//! scripts/bench-macro.sh, scripts/bench-regression.sh, and
5//! .github/workflows/bench*.yml to measure the gateway's end-to-end
6//! throughput and latency across HTTP/1, HTTP/2, and (feature-gated)
7//! HTTP/3, plus pool-reuse and streaming macro workloads.
8//!
9//! ## Protocols
10//!
11//! - `h1` (default): one worker task per connection; each worker owns a
12//!   persistent hyper HTTP/1.1 connection (`client::conn::http1`, no
13//!   pool) and issues back-to-back requests.
14//! - `h2`: each worker owns a persistent HTTP/2 connection
15//!   (`client::conn::http2`, h2c prior-knowledge over cleartext) and
16//!   issues back-to-back requests multiplexed on it.
17//! - `h3`: feature-gated behind the `h3` cargo feature (QUIC + h3). Each
18//!   worker owns one QUIC connection and opens one bidirectional stream
19//!   per request. h3 has NO in-process echo upstream (`--echo` is
20//!   rejected with `--protocol h3`): point it at a real h3 listener
21//!   (the gateway with a `protocol: h3` listener). `--insecure` skips
22//!   certificate verification for loopback benchmark targets only.
23//!
24//! ## Workloads
25//!
26//! - `throughput` (default): owned persistent connection, back-to-back
27//!   requests against a fixed-body echo. Measures raw max throughput.
28//! - `pool-reuse`: a shared pooled client (hyper-util legacy for h1/h2;
29//!   one shared QUIC connection for h3) issues requests across all
30//!   workers, exercising connection/stream reuse on the client side and
31//!   the gateway's upstream pool on the server side.
32//! - `streaming`: owned persistent connection against a streaming echo
33//!   that emits `--stream-chunks` chunks of `--stream-chunk-bytes` bytes
34//!   each (with `--stream-chunk-delay-ms` between chunks); the client
35//!   drains the full streaming body. Measures streaming throughput.
36//!
37//! ## Output
38//!
39//! STDOUT: a human-readable block plus a machine-parseable `RESULT:`
40//! line (`RESULT rps=<f64> errors=<u64> p50_ns=.. p90_ns=.. p99_ns=..
41//! p999_ns=.. err_p99_ns=..`). With `--json`, an additional `JSON:`
42//! line carries the same metrics plus the protocol/workload labels for
43//! the regression gate (`scripts/bench-regression.py`). Success
44//! percentiles cover only successful requests on warm connections (one
45//! unrecorded warmup request per worker); failed and timed-out requests
46//! land in err_p99. Exit code 1 if any request failed (the CI macro job
47//! asserts errors=0 via this).
48//!
49//! 100k-connection runs: this rig imposes no connection cap, but the
50//! OS does - on Linux raise file descriptors first
51//! (`ulimit -n 1048576`); the macOS dev default is far lower, so local
52//! smoke runs should stay at 10k connections or fewer (see
53//! scripts/bench-macro.sh).

179/// Latency samples in nanoseconds; percentile computed by sorting at
180/// report time. Correctness is unit-tested in `tests/loadgen_unit.rs`.
181///
182/// Memory is bounded for long runs: once the sample vector reaches
183/// [`SAMPLE_CAP`], it is halved (every 2nd sample kept) and the recording
184/// stride doubles, so the vector stays O(SAMPLE_CAP) regardless of run
185/// length. Subsampling is uniform in arrival order, so nearest-rank
186/// percentiles over the retained set estimate the true percentiles; at
187/// multi-million sample counts the discretization error is far below
188/// run-to-run noise.
189#[derive(Debug)]
190pub struct Histogram {
191    /// Retained (subsampled) latency samples in ns.
192    pub samples: Vec<u64>,
193    /// Recording stride: only every stride-th arriving sample is kept
194    /// (1 until the first halving, then doubling per halving).
195    pub stride: u64,
196    /// Total samples ever recorded, retained or dropped.
197    pub seen: u64,
198}
199
200impl Default for Histogram {
201    fn default() -> Self {
202        Histogram {
203            samples: Vec::new(),
204            stride: 1,
205            seen: 0,
206        }
207    }
208}
209
210/// Retained-sample ceiling (about 32 MB of u64 samples).
211pub const SAMPLE_CAP: usize = 4_000_000;
212
213impl Histogram {
214    /// Record one latency sample (ns); halves the retained vector and
215    /// doubles the stride once [`SAMPLE_CAP`] is reached.
216    pub fn record(&mut self, ns: u64) {
217        self.seen += 1;
218        if !self.seen.is_multiple_of(self.stride) {
219            return;
220        }
221        self.samples.push(ns);
222        if self.samples.len() >= SAMPLE_CAP {
223            let mut kept: Vec<u64> = Vec::with_capacity(self.samples.len() / 2);
224            kept.extend(self.samples.iter().step_by(2).copied());
225            self.samples = kept;
226            self.stride *= 2;
227        }
228    }
229
230    /// Nearest-rank percentile: the smallest sample at or above the given
231    /// fraction (p50 of [1..100] = 50th smallest). `p` is 0.0-1.0.
232    pub fn percentile(&mut self, p: f64) -> u64 {
233        if self.samples.is_empty() {
234            return 0;
235        }
236        self.samples.sort_unstable();
237        let p = p.clamp(0.0, 1.0);
238        let rank = ((self.samples.len() as f64) * p).ceil() as usize;
239        // ceil of len*1.0 == len; ceil of len*0.0 == 0 -> clamp to 1..=len.
240        self.samples[rank.clamp(1, self.samples.len()) - 1]
241    }
242}

253/// Pacing slice: the dispenser wakes and tops up every 50 ms, and a
254/// starved worker sleeps to the NEXT slice boundary. One constant (and
255/// the single epoch in [`Pacer`]) is what keeps the two sides of pacing
256/// on the same grid.
257pub const PACE_SLICE: Duration = Duration::from_millis(50);
258
259/// Shared pacing state: the permit balance plus the single time base
260/// both sides of pacing use (#127).
261///
262/// The dispenser task's interval is anchored to `epoch`, and every
263/// starved worker sleeps to the next `epoch + k*PACE_SLICE` boundary.
264/// Before this was shared, the starve-sleep re-anchored to its own
265/// `now + 50ms` on every wake, drifting against the dispenser's grid -
266/// a worker could land just before a dispensation and pay a second
267/// slice of wait for the same token.
268pub struct Pacer {
269    /// Permit balance. Workers CAS-decrement; the pacing task tops up
270    /// with [`pace_top_up`]. In unbounded mode (`--rate 0`) it is
271    /// pre-filled with a huge balance so workers never wait.
272    pub permits: std::sync::atomic::AtomicU64,
273    /// Pacing epoch; `None` in unbounded mode (workers never starve).
274    pub epoch: Option<Instant>,
275}
276
277impl Pacer {
278    /// Unbounded-mode pacer: a huge pre-filled balance, no epoch.
279    pub fn unbounded() -> Self {
280        Pacer {
281            permits: std::sync::atomic::AtomicU64::new(u64::MAX / 2),
282            epoch: None,
283        }
284    }
285}

287/// One dispenser tick's top-up decision (#127): how many permits to add,
288/// given the cumulative total the rate schedule owes (`owed`), the
289/// cumulative total already dispensed (`paid`), and the current
290/// unconsumed `balance`. The top-up target is
291/// `min(owed, consumed + per_slice)` with `consumed = paid - balance`:
292///
293/// - the owed-total keeps the exact requested schedule (rates below one
294///   token per slice remain expressible), and
295/// - the balance may never exceed one slice above what workers have
296///   actually CONSUMED, so a worker that stops consuming and later
297///   resumes cannot discharge the accumulated backlog as one burst - a
298///   burst would contaminate paced latency percentiles.
299///
300/// Concurrency: `balance` is read once per tick; a worker consuming
301/// concurrently only lowers the real balance, so the computed target is
302/// an upper bound with at most the intended one slice of slack -
303/// over-dispensing past the cap is impossible.
304///
305/// Pure arithmetic; unit-tested in `tests/loadgen_unit.rs`.
306pub fn pace_top_up(owed: u64, paid: u64, balance: u64, per_slice: u64) -> u64 {
307    let consumed = paid.saturating_sub(balance);
308    owed.min(consumed.saturating_add(per_slice))
309        .saturating_sub(paid)
310}
311
312/// Wait from `now` until the next pacing-slice boundary STRICTLY after
313/// `now`, where boundaries are `epoch + k * PACE_SLICE` (k = 1, 2, ...).
314/// Pure arithmetic over `Instant`s; unit-tested in
315/// `tests/loadgen_unit.rs`. `now` before `epoch` (never in practice)
316/// reads as phase 0 and waits one full slice.
317pub fn until_next_tick(now: Instant, epoch: Instant) -> Duration {
318    let slice = PACE_SLICE.as_millis() as u64;
319    let phase = now.saturating_duration_since(epoch).as_millis() as u64 % slice;
320    Duration::from_millis(if phase == 0 { slice } else { slice - phase })
321}

324/// success and error latency histograms (error latencies kept separate so

328    pub totals: Arc<std::sync::Mutex<Totals>>,
329    /// Success-path latency samples.
330    pub histogram: Arc<std::sync::Mutex<Histogram>>,
331    /// Failed/timeout latency samples.
332    pub err_histogram: Arc<std::sync::Mutex<Histogram>>,
333}

335/// Run the whole rig per `args`: optionally the in-process echo upstream,
336/// the pacing dispenser (positive `--rate`), one worker per connection
337/// (or per pooled-client slot), then the report. Returns the process
338/// exit code (1 if any request failed, 2 on invalid arguments).
339pub async fn run(args: Args) -> i32 {
340    if args.connections == 0 {
341        eprintln!("--connections must be >= 1");
342        return 2;
343    }
344    if args.protocol == Protocol::H3 && args.echo.is_some() {
345        eprintln!("--echo is not supported with --protocol h3 (no in-process h3 echo; point at a real h3 listener)");
346        return 2;
347    }
348
349    if let Some(port) = args.echo {
350        if args.echo_only {
351            if args.workload == Workload::Streaming {
352                echo_server_streaming(
353                    port,
354                    args.stream_chunks,
355                    args.stream_chunk_bytes,
356                    args.stream_chunk_delay_ms,
357                )
358                .await;
359            } else {
360                echo_server(port, args.echo_body).await;
361            }
362            return 0;
363        }
364        if args.workload == Workload::Streaming {
365            tokio::spawn(echo_server_streaming(
366                port,
367                args.stream_chunks,
368                args.stream_chunk_bytes,
369                args.stream_chunk_delay_ms,
370            ));
371        } else {
372            tokio::spawn(echo_server(port, args.echo_body));
373        }
374        // Give the echo listener a moment to bind before the gateway (or
375        // the load) needs it.
376        tokio::time::sleep(Duration::from_millis(200)).await;
377    }
378
379    let start = Instant::now();
380    let deadline = start + Duration::from_secs(args.duration);
381    let totals = Arc::new(std::sync::Mutex::new(Totals::default()));
382    let histogram = Arc::new(std::sync::Mutex::new(Histogram::default()));
383    // Error-path latencies (failed/timeout requests) are recorded in a
384    // separate histogram so they never contaminate the success percentiles;
385    // they are reported as err_p99 on the RESULT line.
386    let err_histogram = Arc::new(std::sync::Mutex::new(Histogram::default()));
387    // Unbounded mode pre-fills a huge permit balance so workers never wait;
388    // paced mode starts at 0, is dispensed by the pacing task below, and
389    // its epoch anchors BOTH the dispenser tick and the workers'
390    // starve-sleeps (see Pacer).
391    let pacer = Arc::new(if args.rate > 0 {
392        Pacer {
393            permits: std::sync::atomic::AtomicU64::new(0),
394            epoch: Some(Instant::now()),
395        }
396    } else {
397        Pacer::unbounded()
398    });
399
400    if let Some(epoch) = pacer.epoch {
401        // Pacing: dispense `rate` tokens per second in PACE_SLICE slices.
402        // Each tick tops the cumulative dispensed total up to
403        // `min(rate * elapsed + one slice, consumed + one slice)` via
404        // [pace_top_up]: the owed-total keeps the exact requested schedule
405        // (even below 20 rps, where per-slice dispensing cannot), while the
406        // consumed-relative cap bounds catch-up - unused tokens never
407        // accumulate behind an idle worker, so resumed workers cannot
408        // discharge a backlog as one burst (#127).
409        let pacer = pacer.clone();
410        let per_slice = ((args.rate as f64 / 20.0).ceil() as u64).max(1);
411        let rate = args.rate;
412        tokio::spawn(async move {
413            // interval_at anchors the grid to the shared epoch so the
414            // first tick fires immediately and later ticks land exactly on
415            // the boundaries starved workers sleep to.
416            let mut tick =
417                tokio::time::interval_at(tokio::time::Instant::from_std(epoch), PACE_SLICE);
418            // Cumulative tokens ever dispensed.
419            let mut paid: u64 = 0;
420            loop {
421                tick.tick().await;
422                let owed = (rate as f64 * epoch.elapsed().as_secs_f64()) as u64 + per_slice;
423                let balance = pacer.permits.load(std::sync::atomic::Ordering::Relaxed);
424                let add = pace_top_up(owed, paid, balance, per_slice);
425                if add > 0 {
426                    pacer
427                        .permits
428                        .fetch_add(add, std::sync::atomic::Ordering::Relaxed);
429                    paid += add;
430                }
431            }
432        });
433    }
434
435    let url: hyper::Uri = match args.url.parse() {
436        Ok(u) => u,
437        Err(e) => {
438            eprintln!("invalid --url: {e}");
439            return 2;
440        }
441    };
442    let authority = url
443        .authority()
444        .map(|a| a.as_str().to_string())
445        .unwrap_or_else(|| "127.0.0.1".into());
446    // --insecure disables certificate verification and is only safe for
447    // loopback benchmark targets. Reject it for non-loopback hosts to
448    // prevent accidental use against remote/production endpoints.
449    if args.insecure {
450        let host = authority
451            .rsplit_once(':')
452            .map(|(h, _)| h)
453            .unwrap_or(&authority);
454        let is_loopback = matches!(host, "127.0.0.1" | "::1" | "localhost" | "[::1]");
455        if !is_loopback {
456            eprintln!(
457                "--insecure is only permitted for loopback targets (127.0.0.1, ::1, localhost); \
458                 got '{host}'"
459            );
460            return 2;
461        }
462    }
463    let path = url
464        .path_and_query()
465        .map(|p| p.as_str().to_owned())
466        .unwrap_or_else(|| "/".into());
467    let scheme = url.scheme_str().unwrap_or("http").to_string();
468
469    let timeout = Duration::from_millis(args.timeout_ms);
470    // Shared per-run state handed to every worker as one Arc (keeps the
471    // worker signature small and clippy-happy).
472    let shared = Arc::new(SharedState {
473        totals,
474        histogram,
475        err_histogram,
476    });
477
478    let code = spawn_workers(
479        &args,
480        scheme,
481        authority,
482        path,
483        deadline,
484        timeout,
485        pacer,
486        Arc::clone(&shared),
487    )
488    .await;
489
490    let totals = shared.totals.lock().unwrap();
491    let mut histogram = shared.histogram.lock().unwrap();
492    let mut err_histogram = shared.err_histogram.lock().unwrap();
493    report(
494        &args,
495        start.elapsed(),
496        &totals,
497        &mut histogram,
498        &mut err_histogram,
499    )
500    .max(code)
501}

503/// Dispatch workers per protocol + workload, await them all, and return
504/// a nonzero exit code if any worker observed a failure (1) or the
505/// dispatch itself was misconfigured (2).
506#[allow(clippy::too_many_arguments)]
507async fn spawn_workers(
508    args: &Args,
509    scheme: String,
510    authority: String,
511    path: String,
512    deadline: Instant,
513    timeout: Duration,
514    pacer: Arc<Pacer>,
515    shared: Arc<SharedState>,
516) -> i32 {
517    let protocol = args.protocol;
518    let workload = args.workload;
519    match (protocol, workload) {
520        (Protocol::H1, Workload::Throughput) | (Protocol::H1, Workload::Streaming) => {
521            let mut workers = Vec::with_capacity(args.connections);
522            for _ in 0..args.connections {
523                workers.push(tokio::spawn(worker(
524                    scheme.clone(),
525                    authority.clone(),
526                    path.clone(),
527                    deadline,
528                    timeout,
529                    pacer.clone(),
530                    shared.clone(),
531                )));
532            }
533            for w in workers {
534                let _ = w.await;
535            }
536            0
537        }
538        (Protocol::H2, Workload::Throughput) | (Protocol::H2, Workload::Streaming) => {
539            let mut workers = Vec::with_capacity(args.connections);
540            for _ in 0..args.connections {
541                workers.push(tokio::spawn(worker_h2(
542                    scheme.clone(),
543                    authority.clone(),
544                    path.clone(),
545                    deadline,
546                    timeout,
547                    pacer.clone(),
548                    shared.clone(),
549                )));
550            }
551            for w in workers {
552                let _ = w.await;
553            }
554            0
555        }
556        (Protocol::H3, Workload::Throughput) | (Protocol::H3, Workload::Streaming) => {
557            let target = H3Target {
558                authority: authority.clone(),
559                path: path.clone(),
560                insecure: args.insecure,
561            };
562            let mut workers = Vec::with_capacity(args.connections);
563            for _ in 0..args.connections {
564                workers.push(tokio::spawn(worker_h3(
565                    target.clone(),
566                    deadline,
567                    timeout,
568                    pacer.clone(),
569                    shared.clone(),
570                )));
571            }
572            for w in workers {
573                let _ = w.await;
574            }
575            0
576        }
577        (Protocol::H1, Workload::PoolReuse) => {
578            let client = match build_pooled_h1_client() {
579                Ok(c) => c,
580                Err(e) => {
581                    eprintln!("pooled h1 client build failed: {e}");
582                    return 2;
583                }
584            };
585            let mut workers = Vec::with_capacity(args.connections);
586            for _ in 0..args.connections {
587                workers.push(tokio::spawn(worker_pooled_h1(
588                    client.clone(),
589                    authority.clone(),
590                    path.clone(),
591                    deadline,
592                    timeout,
593                    pacer.clone(),
594                    shared.clone(),
595                )));
596            }
597            for w in workers {
598                let _ = w.await;
599            }
600            0
601        }
602        (Protocol::H2, Workload::PoolReuse) => {
603            // One shared h2 connection; all workers get their own clone
604            // of the `SendRequest` (Clone — clones share the connection
605            // but issue concurrent streams without mutex serialization).
606            match shared_h2_connection(&authority).await {
607                Ok(send_request) => {
608                    let mut workers = Vec::with_capacity(args.connections);
609                    for _ in 0..args.connections {
610                        workers.push(tokio::spawn(worker_pooled_h2(
611                            send_request.clone(),
612                            authority.clone(),
613                            path.clone(),
614                            deadline,
615                            timeout,
616                            pacer.clone(),
617                            shared.clone(),
618                        )));
619                    }
620                    for w in workers {
621                        let _ = w.await;
622                    }
623                    0
624                }
625                Err(e) => {
626                    eprintln!("h2 pool connection failed: {e}");
627                    2
628                }
629            }
630        }
631        (Protocol::H3, Workload::PoolReuse) => {
632            let target = H3Target {
633                authority: authority.clone(),
634                path: path.clone(),
635                insecure: args.insecure,
636            };
637            match shared_h3_connection(&target).await {
638                Ok((_endpoint, send_request, _driver)) => {
639                    // Keep `_endpoint` and `_driver` alive for the
640                    // duration of the run — dropping the endpoint closes
641                    // the UDP socket and kills the QUIC connection.
642                    // Each worker gets its own clone of `SendRequest`
643                    // (Clone — clones share the connection but issue
644                    // concurrent streams without mutex serialization).
645                    let mut workers = Vec::with_capacity(args.connections);
646                    for _ in 0..args.connections {
647                        workers.push(tokio::spawn(worker_pooled_h3(
648                            send_request.clone(),
649                            target.clone(),
650                            deadline,
651                            timeout,
652                            pacer.clone(),
653                            shared.clone(),
654                        )));
655                    }
656                    for w in workers {
657                        let _ = w.await;
658                    }
659                    0
660                }
661                Err(e) => {
662                    eprintln!("h3 pool connection failed: {e}");
663                    2
664                }
665            }
666        }
667    }
668}

670/// Acquire one pacing permit (unbounded mode: a single uncontended
671/// atomic; paced mode: CAS-decrement, starve-sleeping to the next
672/// dispenser boundary on the pacer's shared epoch grid). Returns when a
673/// permit is held or `deadline` has passed.
674async fn acquire_permit(pacer: &Arc<Pacer>, deadline: Instant) {
675    loop {
676        let b = pacer.permits.load(std::sync::atomic::Ordering::Relaxed);
677        if b > 0
678            && pacer
679                .permits
680                .compare_exchange(
681                    b,
682                    b - 1,
683                    std::sync::atomic::Ordering::Relaxed,
684                    std::sync::atomic::Ordering::Relaxed,
685                )
686                .is_ok()
687        {
688            return;
689        }
690        if Instant::now() >= deadline {
691            return;
692        }
693        // Starved of permits: sleep to the NEXT dispenser boundary on the
694        // pacer's shared epoch grid rather than busy-yielding - a yield_now
695        // spin burns a full core per waiting worker and contaminates the
696        // paced latencies of everyone else. The `None` arm is unreachable
697        // in unbounded mode (the balance is pre-filled huge); the
698        // full-slice fallback just stays honest.
699        let wait = pacer
700            .epoch
701            .map(|epoch| until_next_tick(Instant::now(), epoch))
702            .unwrap_or(PACE_SLICE);
703        tokio::time::sleep(wait).await;
704    }
705}

707/// Record one request outcome into the shared totals + histograms. The
708/// outer `Err` is a timeout (the `Elapsed` from `tokio::time::timeout`,
709/// normalized to `()` by the callers via `map_err`); the inner `Err` is
710/// a transport/status failure. Failed/timeout messages are printed
711/// rate-limited (first five).
712fn record_outcome(
713    shared: &Arc<SharedState>,
714    elapsed: Duration,
715    outcome: Result<Result<(), String>, ()>,
716    timeout: Duration,
717) {
718    let mut totals = shared.totals.lock().unwrap();
719    totals.requests += 1;
720    match outcome {
721        Ok(Ok(())) => shared
722            .histogram
723            .lock()
724            .unwrap()
725            .record(elapsed.as_nanos() as u64),
726        Ok(Err(e)) => {
727            totals.errors += 1;
728            shared
729                .err_histogram
730                .lock()
731                .unwrap()
732                .record(elapsed.as_nanos() as u64);
733            if totals.errors <= 5 {
734                eprintln!("request error: {e}");
735            }
736        }
737        Err(()) => {
738            totals.errors += 1;
739            shared
740                .err_histogram
741                .lock()
742                .unwrap()
743                .record(elapsed.as_nanos() as u64);
744            if totals.errors <= 5 {
745                eprintln!("request error: timeout after {}ms", timeout.as_millis());
746            }
747        }
748    }
749}

751/// One load-generation connection: owns a persistent hyper connection,
752/// acquires pacing permits from `pacer`, and records each request's
753/// outcome into `shared` until `deadline`. The first (warmup) request per
754/// connection is unrecorded, so percentiles measure pure RTT on warm
755/// connections.
756pub async fn worker(
757    scheme: String,
758    authority: String,
759    path: String,
760    deadline: Instant,
761    timeout: Duration,
762    pacer: Arc<Pacer>,
763    shared: Arc<SharedState>,
764) {
765    let mut conn: Option<(
766        hyper::client::conn::http1::SendRequest<Empty<Bytes>>,
767        tokio::task::JoinHandle<()>,
768    )> = None;
769    // Warmup: one unrecorded request per worker before the measurement
770    // loop, so TCP connect + HTTP handshake latency never contaminates a
771    // worker's first sample. RESULT percentiles therefore mean pure
772    // request RTT on warm connections. A warmup failure is not fatal -
773    // the measurement loop re-handshakes on demand.
774    let _ = tokio::time::timeout(
775        timeout,
776        do_request_h1(&mut conn, &scheme, &authority, &path),
777    )
778    .await;
779    while Instant::now() < deadline {
780        acquire_permit(&pacer, deadline).await;
781        if Instant::now() >= deadline {
782            break;
783        }
784        let start = Instant::now();
785        let outcome = tokio::time::timeout(
786            timeout,
787            do_request_h1(&mut conn, &scheme, &authority, &path),
788        )
789        .await;
790        let elapsed = start.elapsed();
791        // A failed/timed-out h1 connection is dead; force re-handshake.
792        if !matches!(outcome, Ok(Ok(()))) {
793            if let Some((_, driver)) = conn.take() {
794                driver.abort();
795            }
796        }
797        record_outcome(&shared, elapsed, outcome.map_err(|_| ()), timeout);
798    }
799    if let Some((_, driver)) = conn.take() {
800        driver.abort();
801    }
802}

1183/// A self-paced streaming response body: emits `chunks` data frames of
1184/// `chunk_bytes` bytes each, sleeping `delay` between frames (after the
1185/// first). Implements `hyper::body::Body` directly so the streaming echo
1186/// needs no extra feature on `http-body-util` (the `channel` feature is
1187/// off in the workspace). `Unpin` (all fields are `Unpin`), so `Pin<&mut
1188/// Self>` dereferences freely.
1189struct StreamingBody {
1190    /// Frames still to emit.
1191    remaining: usize,
1192    /// Pre-allocated reusable chunk (cloned per frame — `Bytes::clone`
1193    /// is a cheap Arc refcount bump, no heap allocation).
1194    chunk: Bytes,
1195    /// Inter-frame delay.
1196    delay: Duration,
1197    /// A pending sleep before the next frame, if any.
1198    sleep: Option<std::pin::Pin<Box<tokio::time::Sleep>>>,
1199}
1200
1201impl StreamingBody {
1202    fn new(chunks: usize, chunk_bytes: usize, delay: Duration) -> Self {
1203        StreamingBody {
1204            remaining: chunks,
1205            chunk: Bytes::from(vec![b'x'; chunk_bytes]),
1206            delay,
1207            sleep: None,
1208        }
1209    }
1210}
1211
1212impl hyper::body::Body for StreamingBody {
1213    type Data = Bytes;
1214    type Error = std::convert::Infallible;
1215
1216    fn poll_frame(
1217        mut self: std::pin::Pin<&mut Self>,
1218        cx: &mut std::task::Context<'_>,
1219    ) -> std::task::Poll<Option<Result<hyper::body::Frame<Bytes>, std::convert::Infallible>>> {
1220        use std::future::Future;
1221        // If a sleep is pending, wait for it to fire before emitting the
1222        // next frame.
1223        if let Some(sleep) = self.sleep.as_mut() {
1224            if sleep.as_mut().poll(cx).is_pending() {
1225                return std::task::Poll::Pending;
1226            }
1227            self.sleep = None;
1228        }
1229        if self.remaining == 0 {
1230            return std::task::Poll::Ready(None);
1231        }
1232        let chunk = self.chunk.clone();
1233        self.remaining -= 1;
1234        // Schedule the delay before the NEXT frame (no delay after the
1235        // last frame, and none when the delay is zero).
1236        if self.remaining > 0 && !self.delay.is_zero() {
1237            self.sleep = Some(Box::pin(tokio::time::sleep(self.delay)));
1238        }
1239        std::task::Poll::Ready(Some(Ok(hyper::body::Frame::data(chunk))))
1240    }
1241}

1318    /// Build a rustls client config for h3 (ALPN "h3"). With

1333        };

1355    /// Split `host:port` into (host, port, server_name).
1356    fn split_authority(authority: &str) -> Result<(String, u16, String), String> {
1357        let (host, port) = authority
1358            .rsplit_once(':')
1359            .ok_or_else(|| format!("authority '{authority}' missing :port"))?;
1360        let port: u16 = port
1361            .parse()
1362            .map_err(|e| format!("authority port '{port}': {e}"))?;
1363        Ok((host.to_string(), port, host.to_string()))
1364    }

1366    /// Resolve `host:port` to a SocketAddr via async DNS (handles both
1367    /// IP literals and hostnames).
1368    async fn resolve_authority(authority: &str) -> Result<std::net::SocketAddr, String> {
1369        let (host, port, _server_name) = split_authority(authority)?;
1370        // Fast path: it is already an IP literal.
1371        if let Ok(addr) = format!("{host}:{port}").parse::<std::net::SocketAddr>() {
1372            return Ok(addr);
1373        }
1374        // Hostname: async DNS lookup, first answer. Pass an owned
1375        // `(String, u16)` so the lookup future does not borrow `host`
1376        // (kept for the error message).
1377        let mut iter = tokio::net::lookup_host((host.clone(), port))
1378            .await
1379            .map_err(|e| format!("resolve {host}:{port}: {e}"))?;
1380        iter.next()
1381            .ok_or_else(|| format!("resolve {host}:{port}: no addresses"))
1382    }

1384    /// Dial one QUIC connection and wrap it in h3, returning the
1385    /// cloneable request sender and the driver join handle.
1386    async fn dial(
1387        endpoint: &Endpoint,
1388        target: &H3Target,
1389    ) -> Result<
1390        (
1391            h3::client::SendRequest<h3_quinn::OpenStreams, Bytes>,
1392            tokio::task::JoinHandle<()>,
1393        ),
1394        String,
1395    > {
1396        let (_host, _port, server_name) = split_authority(&target.authority)?;
1397        let addr = resolve_authority(&target.authority).await?;
1398        let conn = endpoint
1399            .connect(addr, &server_name)
1400            .map_err(|e| format!("quic connect: {e}"))?
1401            .await
1402            .map_err(|e| format!("quic handshake: {e}"))?;
1403        let (mut h3_conn, send_request) = h3::client::new(h3_quinn::Connection::new(conn))
1404            .await
1405            .map_err(|e| format!("h3 client: {e}"))?;
1406        let driver = tokio::spawn(async move {
1407            let _ = h3_conn.wait_idle().await;
1408        });
1409        Ok((send_request, driver))
1410    }

1412    /// One h3 request over a `SendRequest` (one stream per call).
1413    async fn do_request_h3(
1414        send_request: &mut h3::client::SendRequest<h3_quinn::OpenStreams, Bytes>,
1415        authority: &str,
1416        path: &str,
1417    ) -> Result<(), String> {
1418        let req = hyper::Request::builder()
1419            .method(hyper::Method::GET)
1420            .uri(path)
1421            .header(hyper::header::HOST, authority)
1422            .body(())
1423            .map_err(|e| format!("build: {e}"))?;
1424        let mut stream = send_request
1425            .send_request(req)
1426            .await
1427            .map_err(|e| format!("h3 send_request: {e}"))?;
1428        stream
1429            .finish()
1430            .await
1431            .map_err(|e| format!("h3 finish: {e}"))?;
1432        let resp = stream
1433            .recv_response()
1434            .await
1435            .map_err(|e| format!("h3 recv_response: {e}"))?;
1436        let status = resp.status();
1437        while let Some(mut chunk) = stream
1438            .recv_data()
1439            .await
1440            .map_err(|e| format!("h3 recv_data: {e}"))?
1441        {
1442            let bytes = chunk.copy_to_bytes(chunk.remaining());
1443            let _ = bytes;
1444        }
1445        let _ = stream.recv_trailers().await;
1446        if !status.is_success() {
1447            return Err(format!("status {status}"));
1448        }
1449        Ok(())
1450    }

1452    /// One h3 request over an owned connection, dialing first if the
1453    /// connection is absent. Mirrors the h1/h2 owned-request helpers.
1454    async fn do_request_h3_owned(
1455        endpoint: &Endpoint,
1456        conn: &mut Option<(
1457            h3::client::SendRequest<h3_quinn::OpenStreams, Bytes>,
1458            tokio::task::JoinHandle<()>,
1459        )>,
1460        target: &H3Target,
1461    ) -> Result<(), String> {
1462        if conn.is_none() {
1463            let (sr, driver) = dial(endpoint, target).await?;
1464            *conn = Some((sr, driver));
1465        }
1466        let sr = &mut conn.as_mut().expect("h3 conn present").0;
1467        do_request_h3(sr, &target.authority, &target.path).await
1468    }

1470    /// h3 owned-connection worker: one QUIC connection, back-to-back
1471    /// streams.
1472    pub async fn worker_h3(
1473        target: H3Target,
1474        deadline: Instant,
1475        timeout: Duration,
1476        pacer: Arc<Pacer>,
1477        shared: Arc<SharedState>,
1478    ) {
1479        let endpoint = match client_endpoint(target.insecure) {
1480            Ok(e) => e,
1481            Err(e) => {
1482                eprintln!("h3 endpoint build failed: {e}");
1483                return;
1484            }
1485        };
1486        let mut conn: Option<(
1487            h3::client::SendRequest<h3_quinn::OpenStreams, Bytes>,
1488            tokio::task::JoinHandle<()>,
1489        )> = None;
1490        // Warmup (unrecorded).
1491        let _ =
1492            tokio::time::timeout(timeout, do_request_h3_owned(&endpoint, &mut conn, &target)).await;
1493        while Instant::now() < deadline {
1494            acquire_permit(&pacer, deadline).await;
1495            if Instant::now() >= deadline {
1496                break;
1497            }
1498            let start = Instant::now();
1499            let outcome =
1500                tokio::time::timeout(timeout, do_request_h3_owned(&endpoint, &mut conn, &target))
1501                    .await;
1502            let elapsed = start.elapsed();
1503            if !matches!(outcome, Ok(Ok(()))) {
1504                if let Some((_, driver)) = conn.take() {
1505                    driver.abort();
1506                }
1507            }
1508            record_outcome(&shared, elapsed, outcome.map_err(|_| ()), timeout);
1509        }
1510        if let Some((_, driver)) = conn.take() {
1511            driver.abort();
1512        }
1513    }

1535    /// h3 pool-reuse worker: each worker holds its own clone of the h3
1536    /// `SendRequest` (Clone — clones share the same QUIC connection but
1537    /// can issue concurrent streams without mutex serialization). This
1538    /// exercises real h3 stream multiplexing: many workers, many
1539    /// concurrent streams, one QUIC connection.
1540    pub async fn worker_pooled_h3(
1541        mut send_request: h3::client::SendRequest<h3_quinn::OpenStreams, Bytes>,
1542        target: H3Target,
1543        deadline: Instant,
1544        timeout: Duration,
1545        pacer: Arc<Pacer>,
1546        shared: Arc<SharedState>,
1547    ) {
1548        // Warmup (unrecorded).
1549        let _ = tokio::time::timeout(timeout, async {
1550            do_request_h3(&mut send_request, &target.authority, &target.path).await
1551        })
1552        .await;
1553        while Instant::now() < deadline {
1554            acquire_permit(&pacer, deadline).await;
1555            if Instant::now() >= deadline {
1556                break;
1557            }
1558            let start = Instant::now();
1559            let outcome = tokio::time::timeout(timeout, async {
1560                do_request_h3(&mut send_request, &target.authority, &target.path).await
1561            })
1562            .await;
1563            let elapsed = start.elapsed();
1564            record_outcome(&shared, elapsed, outcome.map_err(|_| ()), timeout);
1565        }
1566    }

1571/// Print the human block + RESULT line (+ optional JSON line); returns
1572/// the exit code (1 iff any request failed).
1573fn report(
1574    args: &Args,
1575    ran_for: Duration,
1576    totals: &Totals,
1577    histogram: &mut Histogram,
1578    err_histogram: &mut Histogram,
1579) -> i32 {
1580    let ran_for = ran_for.max(Duration::from_secs(1)).as_secs_f64();
1581    let rps = totals.requests as f64 / ran_for;
1582    let p50 = histogram.percentile(0.50);
1583    let p90 = histogram.percentile(0.90);
1584    let p99 = histogram.percentile(0.99);
1585    let p999 = histogram.percentile(0.999);
1586    let err_p99 = err_histogram.percentile(0.99);
1587    println!(
1588        "protocol={} workload={} connections={} duration={}s rate={}",
1589        args.protocol.as_str(),
1590        args.workload.as_str(),
1591        args.connections,
1592        args.duration,
1593        args.rate
1594    );
1595    println!(
1596        "requests={} errors={} rps={:.0}",
1597        totals.requests, totals.errors, rps
1598    );
1599    println!(
1600        "p50={}us p90={}us p99={}us p999={}us",
1601        p50 / 1_000,
1602        p90 / 1_000,
1603        p99 / 1_000,
1604        p999 / 1_000
1605    );
1606    if totals.errors > 0 {
1607        println!("err_p99={}us (failed/timeout requests)", err_p99 / 1_000);
1608    }
1609    println!(
1610        "RESULT: rps={rps:.2} errors={} p50_ns={p50} p90_ns={p90} p99_ns={p99} p999_ns={p999} err_p99_ns={err_p99}",
1611        totals.errors
1612    );
1613    if args.json {
1614        // The JSON line carries the protocol/workload labels so the
1615        // regression gate can key each workload independently. serde_json
1616        // is already a workspace dependency of dwara-cli (the `schema`
1617        // subcommand); no new external crate enters the tree.
1618        let json = serde_json::json!({
1619            "protocol": args.protocol.as_str(),
1620            "workload": args.workload.as_str(),
1621            "connections": args.connections,
1622            "duration_s": args.duration,
1623            "rate": args.rate,
1624            "requests": totals.requests,
1625            "errors": totals.errors,
1626            "rps": rps,
1627            "p50_ns": p50,
1628            "p90_ns": p90,
1629            "p99_ns": p99,
1630            "p999_ns": p999,
1631            "err_p99_ns": err_p99,
1632        });
1633        println!("JSON: {json}");
1634    }
1635    if totals.errors > 0 {
1636        1
1637    } else {
1638        0
1639    }
