1//! NGINX config import (DW-065): `dwara import nginx` reads an NGINX
2//! config file and generates a Dwara config YAML with routes derived
3//! from the NGINX `location` blocks.
4//!
5//! This is a switching-cost lever for teams migrating off NGINX to
6//! Dwara. The import is a one-shot scaffolding step: it produces a
7//! config the operator edits to add Dwara-specific features (authn,
8//! rate limiting, etc.) that NGINX does not have native equivalents
9//! for.
10//!
11//! ## Supported NGINX directives
12//!
13//! - `server` blocks with `listen` (port) and `server_name` (host)
14//! - `location` blocks with `proxy_pass` (upstream URL)
15//! - `location` match modifiers: exact (`=`), prefix (none), regex
16//!   (`~` case-sensitive, `~*` case-insensitive)
17//! - `upstream` blocks with `server` directives (endpoints)
18//!
19//! ## Unsupported constructs
20//!
21//! The import reports unsupported constructs so the operator knows
22//! what to review manually:
23//! - `if` directives (NGINX's `if` is notoriously unpredictable)
24//! - `rewrite` directives (Dwara has its own rewrite system)
25//! - `auth_basic` (Dwara has its own authn system)
26//! - `limit_req` (Dwara has its own rate limiting)
27//! - Custom modules (Lua, Perl, etc.)
28//! - `try_files` (Dwara is a proxy, not a file server)
29//!
30//! No new dependencies: a minimal NGINX config parser is implemented
31//! inline (NGINX config syntax is simple enough for a line-based
32//! parser to handle the common cases).
33

42/// Import an NGINX config and produce a Dwara config YAML.
43///
44/// Returns the generated config and a list of warnings about
45/// unsupported constructs (appended as comments at the end of the
46/// YAML).
47pub fn import_nginx(config_text: &str) -> Result<ImportResult, String> {
48    let nginx_conf = parse_nginx_config(config_text)?;
49    let (gateway, warnings) = build_gateway_from_nginx(&nginx_conf);
50    let route_count = gateway.routes.len();
51    let mut yaml =
52        dwara_core::config::gateway_to_yaml(&gateway).map_err(|e| format!("serialize: {e}"))?;
53
54    // Append warnings as a comment block at the end of the YAML.
55    if !warnings.is_empty() {
56        yaml.push_str("\n# --- Import warnings ---\n");
57        yaml.push_str("# The following NGINX constructs are not supported and were skipped.\n");
58        yaml.push_str("# Review and handle them manually in Dwara config.\n");
59        for w in &warnings {
60            yaml.push_str(&format!("# - {w}\n"));
61        }
62    }
63
64    Ok(ImportResult { yaml, route_count })
65}

67/// The parsed NGINX config: a list of `http`-level upstreams and
68/// server blocks.
69#[derive(Debug, Default)]
70struct NginxConfig {
71    upstreams: Vec<NginxUpstream>,
72    servers: Vec<NginxServer>,
73}
74
75/// An NGINX `upstream` block.
76#[derive(Debug, Default)]
77struct NginxUpstream {
78    name: String,
79    servers: Vec<String>, // e.g. "127.0.0.1:8080"
80}

82/// An NGINX `server` block.
83#[derive(Debug, Default)]
84struct NginxServer {
85    listen: Vec<u16>,
86    server_name: Vec<String>,
87    locations: Vec<NginxLocation>,
88    warnings: Vec<String>,
89}

91/// An NGINX `location` block.
92#[derive(Debug, Default)]
93struct NginxLocation {
94    /// The match modifier: "=", "", "~", "~*", "^~"
95    modifier: String,
96    /// The match path/pattern.
97    path: String,
98    /// The `proxy_pass` URL, if any.
99    proxy_pass: Option<String>,
100    /// Unsupported directives found in this location.
101    warnings: Vec<String>,
102}

104/// Parse an NGINX config text into a minimal structure.
105fn parse_nginx_config(text: &str) -> Result<NginxConfig, String> {
106    let mut config = NginxConfig::default();
107    let mut stack: Vec<BlockContext> = Vec::new();
108
109    for (line_no, raw_line) in text.lines().enumerate() {
110        let line = raw_line.trim();
111        if line.is_empty() || line.starts_with('#') {
112            continue;
113        }
114
115        // Remove trailing comments (naive: doesn't handle # in strings).
116        let line = if let Some(idx) = find_comment(line) {
117            line[..idx].trim_end()
118        } else {
119            line
120        };
121
122        if line.is_empty() {
123            continue;
124        }
125
126        // Check for block open (ends with `{`).
127        if let Some(directive) = line.strip_suffix('{') {
128            let directive = directive.trim();
129            let parts: Vec<&str> = directive.splitn(2, char::is_whitespace).collect();
130            let block_type = parts[0];
131            let block_args = parts.get(1).unwrap_or(&"").trim();
132
133            match block_type {
134                "http" => stack.push(BlockContext::Http),
135                "upstream" => {
136                    let name = block_args.to_string();
137                    // Pre-create the upstream so server directives can find it.
138                    config.upstreams.push(NginxUpstream {
139                        name: name.clone(),
140                        servers: vec![],
141                    });
142                    stack.push(BlockContext::Upstream(name));
143                }
144                "server" => {
145                    // Accept server blocks both inside http {} and at the
146                    // top level (many NGINX configs omit the http wrapper
147                    // in examples, and the test configs do too).
148                    stack.push(BlockContext::Server(NginxServer::default()));
149                }
150                "location" => {
151                    if let Some(BlockContext::Server(_)) = stack.last() {
152                        let (modifier, path) = parse_location_args(block_args);
153                        let loc = NginxLocation {
154                            modifier,
155                            path,
156                            proxy_pass: None,
157                            warnings: vec![],
158                        };
159                        stack.push(BlockContext::Location(loc));
160                    }
161                }
162                _ => {
163                    stack.push(BlockContext::Unknown);
164                }
165            }
166            continue;
167        }
168
169        // Check for block close (`}`).
170        if line == "}" {
171            if let Some(ctx) = stack.pop() {
172                match ctx {
173                    BlockContext::Server(server) => {
174                        // Accept server blocks at any nesting level (http {}
175                        // or top-level).
176                        config.servers.push(server);
177                    }
178                    BlockContext::Location(loc) => {
179                        if let Some(BlockContext::Server(server)) = stack.last_mut() {
180                            server.locations.push(loc);
181                        }
182                    }
183                    _ => {}
184                }
185            }
186            continue;
187        }
188
189        // Simple directive (inside a block).
190        let parts: Vec<&str> = line.splitn(2, char::is_whitespace).collect();
191        let directive = parts[0];
192        let args = parts.get(1).unwrap_or(&"").trim().trim_end_matches(';');
193
194        match stack.last_mut() {
195            Some(BlockContext::Upstream(name)) => {
196                if directive == "server" {
197                    if let Some(u) = config.upstreams.iter_mut().find(|u| u.name == *name) {
198                        u.servers.push(args.to_string());
199                    }
200                }
201            }
202            Some(BlockContext::Server(server)) => match directive {
203                "listen" => {
204                    if let Some(port) = parse_listen_port(args) {
205                        server.listen.push(port);
206                    }
207                }
208                "server_name" => {
209                    server
210                        .server_name
211                        .extend(args.split_whitespace().map(String::from));
212                }
213                "auth_basic" | "auth_basic_user_file" => {
214                    server.warnings.push(format!(
215                        "line {}: {} -- use Dwara authn (API key / Basic / JWT / mTLS)",
216                        line_no + 1,
217                        directive
218                    ));
219                }
220                "limit_req" | "limit_req_zone" => {
221                    server.warnings.push(format!(
222                        "line {}: {} -- use Dwara rate limiting",
223                        line_no + 1,
224                        directive
225                    ));
226                }
227                _ if is_unsupported_directive(directive) => {
228                    server.warnings.push(format!(
229                        "line {}: {} -- unsupported, review manually",
230                        line_no + 1,
231                        directive
232                    ));
233                }
234                _ => {}
235            },
236            Some(BlockContext::Location(loc)) => match directive {
237                "proxy_pass" => {
238                    loc.proxy_pass = Some(args.to_string());
239                }
240                "rewrite" => {
241                    loc.warnings.push(format!(
242                        "line {}: rewrite -- use Dwara path rewrite/regex rewrite",
243                        line_no + 1
244                    ));
245                }
246                "auth_basic" | "auth_basic_user_file" => {
247                    loc.warnings.push(format!(
248                        "line {}: {} -- use Dwara authn",
249                        line_no + 1,
250                        directive
251                    ));
252                }
253                "limit_req" => {
254                    loc.warnings.push(format!(
255                        "line {}: limit_req -- use Dwara rate limiting",
256                        line_no + 1
257                    ));
258                }
259                "try_files" => {
260                    loc.warnings.push(format!(
261                        "line {}: try_files -- Dwara is a proxy, not a file server",
262                        line_no + 1
263                    ));
264                }
265                "if" => {
266                    loc.warnings.push(format!(
267                        "line {}: if -- NGINX if is not supported, use Dwara CEL conditions",
268                        line_no + 1
269                    ));
270                }
271                _ if is_unsupported_directive(directive) => {
272                    loc.warnings.push(format!(
273                        "line {}: {} -- unsupported, review manually",
274                        line_no + 1,
275                        directive
276                    ));
277                }
278                _ => {}
279            },
280            _ => {}
281        }
282    }
283
284    Ok(config)
285}

287/// The block context for the parser's stack.
288#[derive(Debug)]
289enum BlockContext {
290    Http,
291    Upstream(String),
292    Server(NginxServer),
293    Location(NginxLocation),
294    Unknown,
295}

297/// Parse the `listen` directive args to extract the port.
298fn parse_listen_port(args: &str) -> Option<u16> {
299    let first = args.split_whitespace().next()?;
300    let port_str = if let Some((_, port)) = first.rsplit_once(':') {
301        port
302    } else {
303        first
304    };
305    port_str.parse::<u16>().ok()
306}
307
308/// Parse the `location` directive args to extract the modifier and path.
309fn parse_location_args(args: &str) -> (String, String) {
310    let args = args.trim();
311    if let Some(rest) = args.strip_prefix('=') {
312        ("=".to_string(), rest.trim().to_string())
313    } else if let Some(rest) = args.strip_prefix("~*") {
314        ("~*".to_string(), rest.trim().to_string())
315    } else if let Some(rest) = args.strip_prefix('~') {
316        ("~".to_string(), rest.trim().to_string())
317    } else if let Some(rest) = args.strip_prefix("^~") {
318        ("^~".to_string(), rest.trim().to_string())
319    } else {
320        ("".to_string(), args.to_string())
321    }
322}

333            | "proxy_set_header"

341            | "ssl_certificate"

365/// Find a comment (`#`) in a line, respecting quoted strings.
366fn find_comment(line: &str) -> Option<usize> {
367    let mut in_single = false;
368    let mut in_double = false;
369    for (i, c) in line.char_indices() {
370        match c {
371            '\'' if !in_double => in_single = !in_single,
372            '"' if !in_single => in_double = !in_double,
373            '#' if !in_single && !in_double => return Some(i),
374            _ => {}
375        }
376    }
377    None
378}

380/// Build a Dwara Gateway from the parsed NGINX config.
381fn build_gateway_from_nginx(conf: &NginxConfig) -> (Gateway, Vec<String>) {
382    let mut warnings = Vec::new();
383    let mut services = BTreeMap::new();
384    let mut upstreams = BTreeMap::new();
385    let mut routes = Vec::new();
386
387    // Convert NGINX upstreams to Dwara upstreams + services.
388    for up in &conf.upstreams {
389        let endpoints: Vec<Endpoint> = up
390            .servers
391            .iter()
392            .filter_map(|s| parse_endpoint(s))
393            .collect();
394
395        if endpoints.is_empty() {
396            warnings.push(format!(
397                "upstream '{}' has no valid server endpoints",
398                up.name
399            ));
400            continue;
401        }
402
403        let upstream_name = format!("{}-upstream", up.name);
404        let service_name = format!("{}-service", up.name);
405
406        upstreams.insert(
407            upstream_name.clone(),
408            Upstream {
409                name: upstream_name.clone(),
410                load_balancer: dwara_core::config::LoadBalancer::RoundRobin,
411                hash_on: None,
412                protocol: dwara_core::config::UpstreamProtocol::Http1,
413                trusted_ca_file: None,
414                use_system_roots: false,
415                endpoints,
416                connection_cap: None,
417                slow_start_ms: None,
418                health: None,
419                active_health: None,
420                retries: None,
421                breaker: None,
422                max_pending: None,
423                timeouts: None,
424                oauth2_client_credentials: None,
425                dns_discovery: None,
426                peak_ewma: None,
427                locality: None,
428                pq: false,
429                cert_pinning: None,
430                mtls: None,
431                pool: None,
432            },
433        );
434
435        services.insert(
436            service_name.clone(),
437            Service {
438                name: service_name.clone(),
439                upstream: Some(upstream_name),
440                split: None,
441                sticky: None,
442                base_path: None,
443                version: None,
444                policies: Vec::new(),
445                authorization: None,
446            },
447        );
448    }
449
450    // Convert NGINX server locations to Dwara routes.
451    for server in &conf.servers {
452        for loc in &server.locations {
453            if loc.proxy_pass.is_none() {
454                if !loc.path.is_empty() {
455                    warnings.push(format!(
456                        "location '{}' has no proxy_pass -- skipped (Dwara is a proxy)",
457                        loc.path
458                    ));
459                }
460                continue;
461            }
462
463            let proxy_pass = loc.proxy_pass.as_ref().unwrap();
464            let (target, is_upstream_ref) = resolve_proxy_pass(proxy_pass, &conf.upstreams);
465
466            let service_name = if is_upstream_ref {
467                target
468            } else {
469                let endpoint = match parse_endpoint(&target) {
470                    Some(e) => e,
471                    None => {
472                        warnings.push(format!(
473                            "location '{}' proxy_pass '{}' -- could not parse endpoint",
474                            loc.path, proxy_pass
475                        ));
476                        continue;
477                    }
478                };
479                let idx = routes.len();
480                let upstream_name = format!("route-{idx}-upstream");
481                let svc_name = format!("route-{idx}-service");
482                upstreams.insert(
483                    upstream_name.clone(),
484                    Upstream {
485                        name: upstream_name.clone(),
486                        load_balancer: dwara_core::config::LoadBalancer::RoundRobin,
487                        hash_on: None,
488                        protocol: dwara_core::config::UpstreamProtocol::Http1,
489                        trusted_ca_file: None,
490                        use_system_roots: false,
491                        endpoints: vec![endpoint],
492                        connection_cap: None,
493                        slow_start_ms: None,
494                        health: None,
495                        active_health: None,
496                        retries: None,
497                        breaker: None,
498                        max_pending: None,
499                        timeouts: None,
500                        oauth2_client_credentials: None,
501                        dns_discovery: None,
502                        peak_ewma: None,
503                        locality: None,
504                        pq: false,
505                        cert_pinning: None,
506                        mtls: None,
507                        pool: None,
508                    },
509                );
510                services.insert(
511                    svc_name.clone(),
512                    Service {
513                        name: svc_name.clone(),
514                        upstream: Some(upstream_name),
515                        split: None,
516                        sticky: None,
517                        base_path: None,
518                        version: None,
519                        policies: Vec::new(),
520                        authorization: None,
521                    },
522                );
523                svc_name
524            };
525
526            let (match_type, path_value) = convert_location_match(&loc.modifier, &loc.path);
527
528            let idx = routes.len();
529            let route_name = format!("route-{idx}");
530            routes.push(Route {
531                name: route_name,
532                service: service_name,
533                r#match: RouteMatch {
534                    path: PathMatch {
535                        kind: match_type,
536                        value: path_value,
537                    },
538                    host: None,
539                    methods: Vec::new(),
540                    headers: BTreeMap::new(),
541                    query: Vec::new(),
542                    cookies: Vec::new(),
543                    accept: None,
544                },
545                action: RouteAction::Proxy { rewrite: None },
546                policies: Vec::new(),
547                priority: None,
548                auth_required: false,
549                cors: None,
550                compression: None,
551                limits: None,
552                authorization: None,
553                security_headers_opt_out: false,
554                deprecation: None,
555                maintenance: None,
556                transforms: None,
557                security_headers: None,
558                masking: None,
559                cache: None,
560                methods: Vec::new(),
561                slo: None,
562                websocket: None,
563                waf: None,
564                graphql: None,
565                grpc_web: None,
566                translation: None,
567                request_validation: None,
568                openapi: None,
569                mirror: None,
570                fault_injection: None,
571                plugins: Vec::new(),
572                filter_chain: None,
573                oidc_login: None,
574            });
575
576            warnings.extend(loc.warnings.iter().cloned());
577        }
578
579        warnings.extend(server.warnings.iter().cloned());
580    }
581
582    let allow_empty_routes = routes.is_empty();
583    let gateway = Gateway {
584        version: 1,
585        listeners: Vec::new(),
586        routes,
587        services: services.into_values().collect(),
588        upstreams: upstreams.into_values().collect(),
589        consumers: Vec::new(),
590        policies: Vec::new(),
591        global_policies: Vec::new(),
592        authorization: None,
593        default_security_headers: None,
594        waf: None,
595        trusted_proxies: Vec::new(),
596        max_concurrent_requests: None,
597        load_shed_dry_run: false,
598        jwt_providers: Vec::new(),
599        admin: None,
600        allow_empty_routes,
601        hmac_auth: None,
602        webhooks: Vec::new(),
603        analytics: None,
604        analytics_stream: None,
605        geoip: None,
606        admission_queue: None,
607        mtls_consumer_mapping: None,
608        mtls_forward_headers: None,
609        license: None,
610        oidc_providers: Vec::new(),
611        redis_rate_limiter: None,
612        redis_quotas: None,
613        redis_cache: None,
614        config_convergence: None,
615        plugins: Vec::new(),
616        filter_chain: None,
617        plugin_registry: None,
618        ai: None,
619        fleet: None,
620        lifecycle: None,
621        mesh: None,
622        ssrf_filter: None,
623    };
624
625    (gateway, warnings)
626}

628/// Resolve a `proxy_pass` URL to a service name.
629///
630/// Returns (target, is_upstream_ref). If the proxy_pass references an
631/// NGINX upstream (e.g. `proxy_pass http://my_upstream;`),
632/// is_upstream_ref is true and target is the generated service name.
633/// Otherwise, target is the host:port from the URL.
634fn resolve_proxy_pass(url: &str, upstreams: &[NginxUpstream]) -> (String, bool) {
635    let rest = url
636        .strip_prefix("http://")
637        .or_else(|| url.strip_prefix("https://"))
638        .unwrap_or(url);
639
640    let host = rest.split('/').next().unwrap_or(rest);
641
642    for up in upstreams {
643        if host == up.name {
644            return (format!("{}-service", up.name), true);
645        }
646    }
647
648    (host.to_string(), false)
649}

651/// Parse an endpoint string like "127.0.0.1:8080" into an Endpoint.
652fn parse_endpoint(s: &str) -> Option<Endpoint> {
653    let s = s.trim();
654    let (host, port) = s.rsplit_once(':')?;
655    let port: u16 = port.parse().ok()?;
656    Some(Endpoint {
657        address: host.to_string(),
658        port,
659        weight: 1,
660        region: None,
661        zone: None,
662    })
663}

665/// Convert an NGINX location match to a Dwara path match.
666fn convert_location_match(modifier: &str, path: &str) -> (PathMatchKind, String) {
667    match modifier {
668        "=" => (PathMatchKind::Exact, path.to_string()),
669        "~" | "~*" => {
670            // Dwara regex paths must start with '/'. NGINX regex
671            // patterns often start with '^' -- strip it and prepend '/'.
672            let trimmed = path.strip_prefix('^').unwrap_or(path);
673            let value = if trimmed.starts_with('/') {
674                trimmed.to_string()
675            } else {
676                format!("/{trimmed}")
677            };
678            (PathMatchKind::Regex, value)
679        }
680        _ => (PathMatchKind::Prefix, path.to_string()),
681    }
