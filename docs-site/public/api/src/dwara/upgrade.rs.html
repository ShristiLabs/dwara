1//! Zero-downtime binary upgrade (DW-049).
2//!
3//! Approach: SO_REUSEPORT hand-off with a readiness signal.
4//!
5//! 1. Every listening socket is bound with `SO_REUSEPORT` (in addition to
6//!    `SO_REUSEADDR`), so a second process can bind the SAME port while the
7//!    first is still listening. On Linux the kernel load-balances accepts
8//!    across both sockets; on macOS both sockets may accept (the balance is
9//!    less even but the hand-off still works — no port-loss window).
10//! 2. On `SIGUSR2` the OLD process spawns a NEW copy of the binary (the
11//!    same path by default, or `DWARA_UPGRADE_BINARY`) with the env var
12//!    [`UPGRADE_READY_SOCKET_ENV`] pointing at a freshly-created Unix
13//!    domain socket.
14//! 3. The NEW process binds its listeners (SO_REUSEPORT lets it bind
15//!    alongside the old), spawns its accept tasks, then connects to the
16//!    ready socket and sends `READY\n`. This is the hand-off signal.
17//! 4. The OLD process receives `READY\n` (bounded wait), THEN initiates
18//!    the same drain sequence as SIGTERM: stop accepting, flush backlogs,
19//!    drain HTTP connections within the shutdown budget, exit 0. Because
20//!    the new process is already accepting, no connection is refused and
21//!    no in-flight connection is reset.
22//! 5. If the new process fails to bind/start, it never signals READY; the
23//!    old process times out, logs an ERROR, and KEEPS running — a failed
24//!    upgrade never takes the gateway down.
25//!
26//! This is the nginx/Envoy model without FD passing: SO_REUSEPORT removes
27//! the need to transfer file descriptors, so the coordination is a single
28//! byte frame over a Unix domain socket (portable, no SCM_RIGHTS).
29//!
30//! Operator trigger: `dwara upgrade` (CLI) sends SIGUSR2 to the PID in the
31//! PID file (`DWARA_PID_FILE`) or to an explicit `--pid`.
32//!
33//! Environment:
34//! - `DWARA_PID_FILE`: write the process PID here on startup (default:
35//!   none). The `dwara upgrade` CLI reads it to find the process to
36//!   signal. The new process overwrites it after signaling READY.
37//! - `DWARA_UPGRADE_BINARY`: path to the new binary for an upgrade
38//!   (default: `std::env::current_exe()`).
39//! - [`UPGRADE_READY_SOCKET_ENV`]: set by the old process on the spawned
40//!   child; the child connects to this Unix socket to signal readiness.
41//!   Unset on a normal (non-upgrade) start.
42
43use std::io;
44use std::net::SocketAddr;
45use std::path::PathBuf;
46use std::time::Duration;

48use socket2::{Domain, Protocol, Socket, Type};
49use tokio::net::TcpListener;

51/// Env var the OLD process sets on the spawned child: the path of the
52/// Unix domain socket the child connects to after it is accepting.
53pub(crate) const UPGRADE_READY_SOCKET_ENV: &str = "DWARA_UPGRADE_READY_SOCKET";
54
55/// The readiness frame the new process sends once its accept loops are
56/// running. A single line so a `read_line` drains it cleanly.
57const READY_FRAME: &str = "READY\n";
58
59/// How long the old process waits for the new process to signal READY
60/// before giving up (and keeping the old process running). Generous:
61/// the new process must load config, bind, and spawn accept tasks.
62const DEFAULT_UPGRADE_READY_TIMEOUT: Duration = Duration::from_secs(30);
63
64/// Bind a TCP listener with `SO_REUSEADDR` AND `SO_REUSEPORT` so a second
65/// process can bind the same address during an upgrade hand-off. Returns
66/// a tokio `TcpListener` (non-blocking, close-on-exec handled by tokio).
67///
68/// `SO_REUSEPORT` is available on Linux and macOS (and modern FreeBSD);
69/// the call is a no-op-fallback on platforms without it (none of the
70/// supported targets lack it). The socket is bound in the same address
71/// family as `addr`.
72pub(crate) fn bind_with_reuse_port(addr: SocketAddr) -> io::Result<TcpListener> {
73    let domain = Domain::for_address(addr);
74    let socket = Socket::new(domain, Type::STREAM, Some(Protocol::TCP))?;
75    // SO_REUSEADDR (already set by tokio's bind; set explicitly here so
76    // the socket2-owned path matches the semantics the gateway relied on).
77    socket.set_reuse_address(true)?;
78    // SO_REUSEPORT: the upgrade enabler. Multiple processes may bind the
79    // same port; the kernel distributes incoming connections. Without it
80    // the new process's bind would fail with EADDRINUSE while the old
81    // process is still listening.
82    socket.set_reuse_port(true)?;
83    socket.bind(&addr.into())?;
84    // Match tokio's default backlog (1024). The exact value is not
85    // load-bearing for correctness; the backlog flush drains whatever is
86    // queued at shutdown.
87    socket.listen(1024)?;
88    socket.set_nonblocking(true)?;
89    // socket2 -> std -> tokio. The std TcpListener takes ownership of the
90    // FD; tokio's from_std registers it with its reactor.
91    let std_listener: std::net::TcpListener = socket.into();
92    TcpListener::from_std(std_listener)
93}

95/// A unique Unix domain socket path for one upgrade hand-off. Lives
96/// under `/tmp` (NOT `std::env::temp_dir()`: on macOS the temp dir is a
97/// long `/var/folders/...` path that can exceed the Unix domain socket
98/// SUN_LEN limit of 104 bytes), namespaced by the OLD process's PID so
99/// concurrent upgrades on the same host do not collide.
100pub(crate) fn ready_socket_path(old_pid: u32) -> PathBuf {
101    PathBuf::from(format!("/tmp/dwara-upgrade-{old_pid}.sock"))
102}
103
104/// Resolve the new binary path for an upgrade. `DWARA_UPGRADE_BINARY`
105/// overrides; otherwise the current executable. Falls back to a literal
106/// `dwara` on PATH only if current_exe cannot be resolved (extremely
107/// rare — the binary was exec'd to get here).
108fn new_binary_path() -> PathBuf {
109    if let Ok(p) = std::env::var("DWARA_UPGRADE_BINARY") {
110        if !p.is_empty() {
111            return PathBuf::from(p);
112        }
113    }
114    std::env::current_exe().unwrap_or_else(|_| PathBuf::from("dwara"))
115}

117/// The upgrade ready timeout. `DWARA_UPGRADE_READY_TIMEOUT_SECS` overrides
118/// the default; an unparseable value falls back to the default.
119fn ready_timeout() -> Duration {
120    std::env::var("DWARA_UPGRADE_READY_TIMEOUT_SECS")
121        .ok()
122        .and_then(|v| v.parse::<u64>().ok())
123        .map(Duration::from_secs)
124        .unwrap_or(DEFAULT_UPGRADE_READY_TIMEOUT)
125}

127/// Spawn the new binary as an upgrade child. The child inherits the
128/// environment (DWARA_CONFIG, DWARA_BIND, ... — it must serve the SAME
129/// listeners) plus [`UPGRADE_READY_SOCKET_ENV`] pointing at `ready_socket`.
130/// The child is detached (no stdio piped) so it outlives the old process's
131/// exit independently.
132pub(crate) fn spawn_upgrade_child(ready_socket: &str) -> io::Result<std::process::Child> {
133    let mut cmd = std::process::Command::new(new_binary_path());
134    cmd.env(UPGRADE_READY_SOCKET_ENV, ready_socket);
135    // Detach stdio so the child is not tied to the old process's pipes
136    // (the old process exits after the hand-off; the child must survive).
137    cmd.stdin(std::process::Stdio::null())
138        .stdout(std::process::Stdio::inherit())
139        .stderr(std::process::Stdio::inherit());
140    // On Unix, start the child in its own process group so signals sent
141    // to the old process's group (e.g. Ctrl-C in a terminal) do not
142    // cascade to the new gateway. `process_group(0)` makes the child's
143    // PID its own process-group ID (std since 1.64; no libc dep needed).
144    #[cfg(unix)]
145    {
146        use std::os::unix::process::CommandExt as _;
147        cmd.process_group(0);
148    }
149    cmd.spawn()
150}

152/// Wait for the new process to signal READY over the Unix domain socket.
153/// The old process has already bound `listener` at `ready_socket_path`
154/// BEFORE spawning the child (so the child's connect cannot race a
155/// missing listener). Returns Ok(()) on READY, Err on timeout/IO error
156/// (the caller keeps running on Err — a failed upgrade is safe).
157pub(crate) async fn await_ready(
158    listener: &tokio::net::UnixListener,
159    timeout: Duration,
160) -> Result<(), String> {
161    let accept = listener.accept();
162    let (mut stream, _peer) = tokio::select! {
163        res = accept => match res {
164            Ok(conn) => conn,
165            Err(err) => return Err(format!("ready-socket accept failed: {err}")),
166        },
167        _ = tokio::time::sleep(timeout) => {
168            return Err(format!(
169                "upgrade timed out waiting for the new process to signal READY \
170                 after {timeout:?}; keeping the old process running"
171            ));
172        }
173    };
174    use tokio::io::AsyncReadExt as _;
175    let mut buf = [0u8; 16];
176    let n = tokio::time::timeout(timeout, stream.read(&mut buf))
177        .await
178        .map_err(|_| "ready frame read timed out".to_string())?
179        .map_err(|err| format!("ready frame read failed: {err}"))?;
180    let frame = std::str::from_utf8(&buf[..n]).unwrap_or("");
181    if frame.trim_end() == READY_FRAME.trim_end() {
182        Ok(())
183    } else {
184        Err(format!("unexpected ready frame: {frame:?}"))
185    }
186}

188/// The OLD process's upgrade entry point. Called from the SIGUSR2 handler
189/// in main. Binds the ready socket, spawns the child, awaits READY, and
190/// returns Ok(()) when the old process should proceed to drain+exit. On
191/// any error the old process keeps running (returns the error for logging).
192///
193/// `old_pid` is this process's PID (used to name the ready socket).
194pub(crate) async fn initiate_upgrade(
195    old_pid: u32,
196) -> Result<(tokio::net::UnixListener, PathBuf), String> {
197    let socket_path = ready_socket_path(old_pid);
198    // Clean up a stale socket from a previous failed attempt, then bind.
199    let _ = std::fs::remove_file(&socket_path);
200    let listener = tokio::net::UnixListener::bind(&socket_path)
201        .map_err(|err| format!("cannot bind ready socket {}: {err}", socket_path.display()))?;
202    let path_str = socket_path.to_string_lossy().to_string();
203    tracing::info!(
204        code = "upgrade_initiated",
205        ready_socket = %path_str,
206        "SIGUSR2 received: spawning new binary for zero-downtime upgrade"
207    );
208    let child = spawn_upgrade_child(&path_str)
209        .map_err(|err| format!("cannot spawn upgrade child: {err}"))?;
210    tracing::info!(
211        code = "upgrade_child_spawned",
212        pid = child.id(),
213        "upgrade child spawned; waiting for READY signal"
214    );
215    let timeout = ready_timeout();
216    match await_ready(&listener, timeout).await {
217        Ok(()) => {
218            tracing::info!(
219                code = "upgrade_ready",
220                "new process signaled READY; the old process will drain and exit"
221            );
222            // The child is now accepting; the old process can drain.
223            // Keep the ready-listener alive (returned) so the socket file
224            // is cleaned up by the caller after drain.
225            Ok((listener, socket_path))
226        }
227        Err(err) => {
228            tracing::error!(
229                code = "upgrade_failed",
230                "upgrade failed: {err}; the old process keeps running"
231            );
232            // Best-effort cleanup of the ready socket and the child.
233            let _ = std::fs::remove_file(&socket_path);
234            Err(err)
235        }
236    }
237}

239/// The NEW process's readiness signal. Called from main AFTER all
240/// listeners are bound and accept tasks are spawned. Connects to the
241/// ready socket named by [`UPGRADE_READY_SOCKET_ENV`] and sends READY.
242/// Errors are logged but do NOT stop the new process — it is already
243/// serving traffic; the old process will time out and keep running (or
244/// the operator notices the old process did not exit and investigates).
245pub(crate) async fn signal_ready() {
246    let Some(path) = std::env::var(UPGRADE_READY_SOCKET_ENV).ok() else {
247        return; // normal (non-upgrade) start
248    };
249    if path.is_empty() {
250        return;
251    }
252    let path = PathBuf::from(&path);
253    // The old process bound the socket before spawning us, but a tiny
254    // connect race is possible on a loaded host; retry briefly.
255    let deadline = tokio::time::Instant::now() + Duration::from_secs(5);
256    let mut stream = loop {
257        match tokio::net::UnixStream::connect(&path).await {
258            Ok(s) => break s,
259            Err(err) if tokio::time::Instant::now() < deadline => {
260                tokio::time::sleep(Duration::from_millis(50)).await;
261                let _ = err; // retry
262            }
263            Err(err) => {
264                tracing::error!(
265                    code = "upgrade_ready_signal_failed",
266                    socket = %path.display(),
267                    "cannot connect to the upgrade ready socket: {err}; \
268                     the old process will time out and keep running"
269                );
270                return;
271            }
272        }
273    };
274    use tokio::io::AsyncWriteExt as _;
275    if let Err(err) = stream.write_all(READY_FRAME.as_bytes()).await {
276        tracing::error!(
277            code = "upgrade_ready_signal_failed",
278            "cannot send READY frame: {err}; the old process will time out"
279        );
280        return;
281    }
282    tracing::info!(
283        code = "upgrade_ready_signaled",
284        "upgrade child signaled READY to the old process"
285    );
286    // Clear the inherited env so a FUTURE upgrade from this process does
287    // not re-use the stale socket path (spawn_upgrade_child always sets a
288    // fresh path on its child, but a clean env avoids confusion).
289    std::env::remove_var(UPGRADE_READY_SOCKET_ENV);
290}

292/// Write `pid` to the PID file at `path` (atomic temp + rename, mirroring
293/// the CLI's config writer). Overwrites any existing file. The new
294/// process calls this after signaling READY so the PID file always names
295/// the live acceptor.
296pub(crate) fn write_pid_file(path: &std::path::Path, pid: u32) -> io::Result<()> {
297    let dir = path
298        .parent()
299        .filter(|p| !p.as_os_str().is_empty())
300        .unwrap_or_else(|| std::path::Path::new("."));
301    // Unique temp name in the same directory (same filesystem -> atomic
302    // rename). No tempfile crate dependency: std only.
303    let tmp = dir.join(format!(
304        ".{}.pid.tmp",
305        path.file_name().and_then(|n| n.to_str()).unwrap_or("dwara")
306    ));
307    {
308        use std::io::Write as _;
309        let mut f = std::fs::OpenOptions::new()
310            .write(true)
311            .create(true)
312            .truncate(true)
313            .open(&tmp)?;
314        writeln!(f, "{pid}")?;
315        f.flush()?;
316        f.sync_all()?;
317    }
318    std::fs::rename(&tmp, path)?;
319    Ok(())
320}

330    /// A short, unique Unix socket path under /tmp (macOS caps Unix
331    /// domain socket paths at SUN_LEN=104; the temp dir's
332    /// `/var/folders/...` prefix plus a nanos timestamp can exceed it).
333    fn short_sock_path(tag: &str) -> PathBuf {
334        static SEQ: std::sync::atomic::AtomicU64 = std::sync::atomic::AtomicU64::new(0);
335        let n = SEQ.fetch_add(1, std::sync::atomic::Ordering::Relaxed);
336        PathBuf::from(format!(
337            "/tmp/dwara-upgrade-test-{}-{n}-{tag}.sock",
338            std::process::id()
339        ))
340    }
341
342    #[tokio::test]
343    async fn reuse_port_bind_succeeds() {
344        // Bind an ephemeral port with SO_REUSEPORT; a second bind to the
345        // SAME port must also succeed (the upgrade enabler). tokio test
346        // because TcpListener::from_std registers with the reactor.
347        let listener =
348            bind_with_reuse_port("127.0.0.1:0".parse().unwrap()).expect("reuse_port bind failed");
349        let addr = listener.local_addr().expect("local addr");
350        let _second = bind_with_reuse_port(addr)
351            .expect("second bind with SO_REUSEPORT failed on the same port");
352        // Both drop here; no port-loss window because both were bound.
353    }
354
355    #[tokio::test]
356    async fn ready_handoff_round_trip() {
357        // The new process signals READY; the old process receives it.
358        let path = short_sock_path("rt");
359        let _ = std::fs::remove_file(&path);
360        let listener = tokio::net::UnixListener::bind(&path).unwrap();
361        let path_clone = path.clone();
362        let sender = tokio::spawn(async move {
363            // Simulate the new process: connect and send READY.
364            let mut s = tokio::net::UnixStream::connect(&path_clone).await.unwrap();
365            use tokio::io::AsyncWriteExt as _;
366            s.write_all(READY_FRAME.as_bytes()).await.unwrap();
367        });
368        let res = await_ready(&listener, Duration::from_secs(2)).await;
369        sender.await.unwrap();
370        let _ = std::fs::remove_file(&path);
371        assert!(res.is_ok(), "await_ready failed: {:?}", res);
372    }
373
374    #[tokio::test]
375    async fn await_ready_times_out() {
376        let path = short_sock_path("to");
377        let _ = std::fs::remove_file(&path);
378        let listener = tokio::net::UnixListener::bind(&path).unwrap();

380        let _ = std::fs::remove_file(&path);
381        assert!(res.is_err(), "expected timeout, got {:?}", res);
382    }
