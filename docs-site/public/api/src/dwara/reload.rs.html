1//! Hot reload: config snapshot and TLS certificate reloads.
2//!
3//! Hot reload semantics:
4//! - Config file watch + SIGHUP: re-read, validate, `compile_and_publish`
5//!   (DW-006 semantics unchanged). A file-watch reload of UNCHANGED
6//!   content is a no-op (the generation only moves when content changes);
7//!   SIGHUP always re-publishes (forced reload). Watch events limited to
8//!   create/modify-data/remove/rename — metadata churn (atime) from the
9//!   reload's own reads must not re-trigger it. On success, terminate
10//!   listeners' TLS configurations are rebuilt from the new snapshot and
11//!   the active health probe loops (DW-013) are respawned against the new
12//!   generation (endpoints persisting across the swap keep their health
13//!   trackers).
14//! - Certificate hot reload: cert/key files of terminate listeners are
15//!   watched; a change rebuilds the `ServerConfig` behind an `ArcSwap`
16//!   WITHOUT dropping connections. New handshakes use the new material;
17//!   in-flight sessions keep the configuration they negotiated.
18//! - SCALE-03 (#182): listener hot-reload. On a successful reload, the
19//!   listener set is diffed: added listeners are bound, removed
20//!   listeners are drained, changed listeners are restarted. The
21//!   `ListenerManager` lives in the reload driver task and is applied
22//!   after every successful publish. The bind identity (address, port,
23//!   protocol, proxy_protocol, TLS mode) determines whether a listener
24//!   needs a restart; everything else (policies, authorization, alt_svc,
25//!   TLS cert material) is read from the live snapshot at request time
26//!   and needs no restart.
27

42/// One reload attempt: re-read from disk, validate, publish atomically.
43/// Never fatal: on any failure the published snapshot is untouched and the
44/// process keeps running. On success, the dataplane's (snapshot, registry)
45/// pair is rebuilt (routes and upstream pools hot-swap together; in-flight
46/// requests keep the old pair) and terminate listeners' TLS configs are
47/// rebuilt from the new snapshot (certificate material follows the config
48/// generation it belongs to).
49#[allow(clippy::too_many_arguments)]
50pub(crate) async fn reload(
51    state: &ConfigState,
52    dp: &DataPlane,
53    source: &FileConfigSource,
54    trigger: &str,
55    tls_states: &BTreeMap<String, Arc<TlsTermination>>,
56    probes: &mut dwara_core::active::ActiveProbes,
57    discovery: &mut dwara_core::dataplane::discovery::DiscoveryTasks,
58    dns_resolver: &Arc<dwara_core::dataplane::discovery::DnsResolver>,
59    obs: &Arc<Observability>,
60    listener_manager: &mut ListenerManager,
61    tls_states_mut: &mut BTreeMap<String, Arc<TlsTermination>>,
62) {
63    let old = state.snapshot();
64    match source.load().await {
65        Ok(gateway) => {
66            // File-watch reloads of identical content are no-ops: the
67            // generation only moves when content changes (editors and
68            // the admin API's atomic rename both re-deliver events for
69            // already-current content). SIGHUP stays a forced reload —
70            // operators use it to re-publish and reset per-generation
71            // state. Compile failures fall through to the publish path,
72            // which reports them without touching the running snapshot.
73            if trigger == "file-watch" {
74                if let Ok(compiled) = dwara_core::snapshot::compile(&gateway) {
75                    if compiled.content_hash() == old.content_hash() {
76                        tracing::info!(
77                            code = "config_reload_unchanged",
78                            trigger,
79                            generation = old.generation(),
80                            "config reload skipped: content unchanged"
81                        );
82                        return;
83                    }
84                }
85            }
86            match state.compile_and_publish(&gateway) {
87                Ok(info) => {
88                    tracing::info!(
89                    code = "config_reloaded",
90                    trigger,
91                    from_generation = old.generation(),
92                    generation = info.generation,
93                    routes = info.route_count,
94                    "config reloaded: generation {} -> {} content_hash={:#x} routes={} [retiring generation {}; freed when its last in-flight request completes]",
95                    old.generation(),
96                    info.generation,
97                    info.content_hash,
98                    info.route_count,
99                    old.generation(),
100                );
101                    // DW-165 review: registry plugin-source resolution
102                    // does blocking IO (DNS via getaddrinfo, synchronous
103                    // TLS reads). Resolve it on the blocking pool — off
104                    // this async reload task — and hand the dataplane
105                    // the results, so refresh never runs network IO on a
106                    // worker thread. Bounded by the resolver's
107                    // per-publish budget (60s across all source plugins);
108                    // a JoinError (cancellation/panic) leaves an empty
109                    // map, which fails every source plugin closed for
110                    // this publish — the next reload retries.
111                    let sources = resolve_sources_off_worker(state).await;
112                    dp.refresh_with_sources(sources);
113                    refresh_tls_states(&state.snapshot(), tls_states, trigger);
114                    // DW-054: publish the new local generation to the
115                    // convergence backend so other instances converge
116                    // to it. A no-op when convergence is not configured
117                    // or not licensed (the ent-gated method returns
118                    // immediately when no coordinator is attached).
119                    #[cfg(feature = "ent")]
120                    dp.publish_convergence_local().await;
121                    // Active health checks (DW-013): probe loops are per
122                    // generation — cancel the old tasks and spawn against the
123                    // new registry. Endpoints whose address:port persists keep
124                    // their health trackers (carried by the balancer), so an
125                    // ejection streak survives the swap.
126                    probes.respawn(&dp.registry(), &state.snapshot());
127                    // DW-042: DNS discovery tasks are per generation —
128                    // cancel the old tasks and spawn against the new
129                    // registry, mirroring ActiveProbes. Upstreams without
130                    // `dns_discovery` spawn nothing.
131                    discovery.respawn(
132                        &dp.registry(),
133                        &state.snapshot(),
134                        Arc::clone(dns_resolver),
135                        Arc::clone(obs),
136                    );
137                    // SCALE-03 (#182): listener hot-reload. Diff the
138                    // new listener set against the currently bound set
139                    // and apply changes (add/remove/restart). Pass the
140                    // mutable tls_states so the manager can register
141                    // new terminate listeners for cert watching.
142                    let configured = crate::configured_listeners(&state.snapshot());
143                    listener_manager.apply(&configured, tls_states_mut).await;
144                }
145                Err(err) => {
146                    // CompileError::Validation's Display lists every issue.
147                    tracing::error!(
148                        code = "config_reload_rejected",
149                        trigger,
150                        "config reload rejected: {err}"
151                    );
152                    tracing::error!(
153                        code = "config_reload_kept_previous",
154                        generation = old.generation(),
155                        "keeping running generation {} (content_hash={:#x})",
156                        old.generation(),
157                        old.content_hash()
158                    );
159                }
160            }
161        }
162        Err(err) => {
163            tracing::error!(
164                code = "config_reload_read_failed",
165                trigger,
166                "config reload failed to read source: {err}"
167            );
168            tracing::error!(
169                code = "config_reload_kept_previous",
170                generation = old.generation(),
171                "keeping running generation {} (content_hash={:#x})",
172                old.generation(),
173                old.content_hash()
174            );
175        }
176    }
177}

179/// Resolve registry plugin sources (DW-165) on the blocking pool.
180///
181/// The resolver's downloader is synchronous by design (see
182/// `wasm::source`: resolution runs inside the synchronous
183/// `DataPlane::reload_plugins`, where a nested `block_on` would
184/// panic) — DNS via getaddrinfo and the sync TLS reads are blocking
185/// syscalls. Running it inside `spawn_blocking` keeps those syscalls
186/// off the async worker; the reload task awaits instead of holding
187/// the thread, and the resolver's per-publish budget (60s across all
188/// source plugins) bounds the wait. Configs without `source:` plugins
189/// skip the thread hop entirely.
190async fn resolve_sources_off_worker(
191    state: &ConfigState,
192) -> dwara_core::wasm::source::SourceResolutions {
193    let snapshot = state.snapshot();
194    if !snapshot
195        .gateway()
196        .plugins
197        .iter()
198        .any(|p| p.source.is_some())
199    {
200        return Default::default();
201    }
202    let published = snapshot.gateway().clone();
203    match tokio::task::spawn_blocking(move || dwara_core::wasm::source::resolve_sources(&published))
204        .await
205    {
206        Ok(sources) => sources,
207        Err(join_err) => {
208            tracing::warn!(
209                code = "plugin_source_resolution_join_failed",
210                "registry source resolution task failed: {join_err}; source \
211                 plugins fail closed for this publish"
212            );
213            Default::default()
214        }
215    }
216}

218/// Rebuild every known terminate listener's TLS config from `snapshot`.
219/// Listeners are matched by name (the bind set is fixed at startup); a
220/// listener that changed mode or vanished from the config keeps its
221/// current certificate material until restart (documented limitation).
222/// Failures log and keep the previous config.
223pub(crate) fn refresh_tls_states(
224    snapshot: &Snapshot,
225    tls_states: &BTreeMap<String, Arc<TlsTermination>>,
226    trigger: &str,
227) {
228    for (name, term) in tls_states {
229        let Some(l) = snapshot
230            .gateway()
231            .listeners
232            .iter()
233            .find(|l| &l.name == name)
234        else {
235            continue;
236        };
237        let Some(tls_cfg) = &l.tls else {
238            continue;
239        };
240        match term.reload(tls_cfg) {
241            Ok(()) => {
242                tracing::info!(code = "tls_reloaded", trigger, listener = %name, "tls config reloaded")
243            }
244            Err(err) => tracing::warn!(
245                code = "tls_reload_rejected",
246                trigger,
247                listener = %name,
248                "tls reload rejected: {err}; keeping previous certificates"
249            ),
250        }
251    }
252}

254/// Notify watcher over a set of directories, forwarding events that touch
255/// any of `names` into the async channel. Mirrors the config-watcher
256/// pattern (directory watch so atomic rename saves are observed).
257pub(crate) fn spawn_file_watcher(
258    dirs: Vec<PathBuf>,
259    names: Vec<std::ffi::OsString>,
260) -> Result<mpsc::UnboundedReceiver<()>, notify::Error> {
261    let (tx, rx) = std::sync::mpsc::channel();
262    let mut watcher =
263        notify::recommended_watcher(move |event: Result<notify::Event, notify::Error>| {
264            let Ok(event) = event else { return };
265            // Metadata-only churn (atime/ctime) and access events must
266            // not trigger reloads: on Linux inotify, a reload re-reads
267            // the file, the read bumps atime, IN_ATTRIB re-fires the
268            // watcher — a self-sustaining reload loop at the debounce
269            // cadence. Only creation, data modification, removal, and
270            // rename events count.
271            match event.kind {
272                notify::EventKind::Modify(notify::event::ModifyKind::Metadata(_))
273                | notify::EventKind::Access(_)
274                | notify::EventKind::Other => return,
275                _ => {}
276            }
277            let touches_target = event
278                .paths
279                .iter()
280                .any(|p| p.file_name().is_some_and(|n| names.contains(&n.to_owned())));
281            if touches_target {
282                let _ = tx.send(());
283            }
284        })?;
285    for dir in &dirs {
286        watcher.watch(dir, notify::RecursiveMode::NonRecursive)?;
287    }
288    let (out_tx, out_rx) = mpsc::unbounded_channel();
289    std::thread::spawn(move || {
290        let _watcher = watcher;
291        while rx.recv().is_ok() {
292            let _ = out_tx.send(());
293        }
294    });
295    Ok(out_rx)
296}
