1//! Listener hot-reload (SCALE-03, #182).
2//!
3//! On config reload, the listener set is diffed against the currently
4//! bound set:
5//! - Added listeners are bound and spawned.
6//! - Removed listeners are drained (their shutdown watch is signalled;
7//!   the accept loop exits and in-flight connections drain via the
8//!   graceful shutdown watcher).
9//! - Changed listeners (address, port, protocol, proxy_protocol, or
10//!   TLS mode changed) are drained and re-bound.
11//! - Unchanged listeners keep running (their TLS cert material is
12//!   refreshed separately by the cert watcher / `refresh_tls_states`).
13//!
14//! Each listener gets its own per-listener shutdown watch so the manager
15//! can stop one listener without affecting the others. The process-wide
16//! shutdown watch still cascades to every listener via the manager.
17

32/// The drain timeout for a single listener being stopped or restarted.
33/// Shorter than the process shutdown timeout: a listener being removed
34/// mid-run should not hold the reload path for the full shutdown budget.
35const LISTENER_DRAIN_TIMEOUT: std::time::Duration = std::time::Duration::from_secs(5);
36
37/// One managed listener: its task handle, per-listener shutdown sender,
38/// and the bind identity (so we can detect changes that require a
39/// restart).
40struct ManagedListener {
41    addr: String,
42    protocol: ListenerProtocol,
43    proxy_protocol: bool,
44    tls_mode: Option<TlsMode>,
45    /// Per-listener shutdown: signalling this stops the accept loop and
46    /// drains in-flight connections via the graceful watcher.
47    shutdown_tx: watch::Sender<()>,
48    task: JoinHandle<()>,
49}
50
51/// The bind identity of a listener: the fields that, if changed,
52/// require a restart (re-bind + re-spawn). Everything else (policies,
53/// authorization, alt_svc, TLS cert material) is read from the live
54/// snapshot at request time and needs no restart.
55fn bind_identity(l: &Listener) -> (String, ListenerProtocol, bool, Option<TlsMode>) {
56    let addr = format!("{}:{}", l.address, l.port);
57    let tls_mode = l.tls.as_ref().map(|t| t.mode);
58    (addr, l.protocol, l.proxy_protocol, tls_mode)
59}
60
61/// True when the listener's bind identity changed (requires restart).
62fn identity_changed(old: &ManagedListener, new: &Listener) -> bool {
63    let (new_addr, new_proto, new_pp, new_tls) = bind_identity(new);
64    old.addr != new_addr
65        || old.protocol != new_proto
66        || old.proxy_protocol != new_pp
67        || old.tls_mode != new_tls
68}

70/// Manages the lifecycle of bound listeners. Created at startup with
71/// the initial listener set; `apply` is called on every config reload
72/// to diff the new listener set against the current one.
73pub(crate) struct ListenerManager {
74    listeners: BTreeMap<String, ManagedListener>,
75    state: Arc<ConfigState>,
76    dp: Arc<DataPlane>,
77    graceful: Arc<GracefulShutdown>,
78    hardening: Arc<HttpHardening>,
79    splice_drain: Arc<SpliceDrain>,
80    /// The process-wide shutdown watch. Each per-listener shutdown is
81    /// cascaded from this: when the process shuts down, every listener
82    /// stops. Per-listener shutdowns (for removed/changed listeners) are
83    /// independent.
84    process_shutdown: watch::Receiver<()>,
85    /// The drain timeout for the process-wide shutdown (used for
86    /// per-listener drains too, scaled down).
87    shutdown_timeout: std::time::Duration,
88}
89
90impl ListenerManager {
91    /// Create a new manager. The initial listener set is applied via
92    /// `apply` (pass the startup config's listeners).
93    pub(crate) fn new(
94        state: Arc<ConfigState>,
95        dp: Arc<DataPlane>,
96        graceful: Arc<GracefulShutdown>,
97        hardening: Arc<HttpHardening>,
98        splice_drain: Arc<SpliceDrain>,
99        process_shutdown: watch::Receiver<()>,
100        shutdown_timeout: std::time::Duration,
101    ) -> Self {
102        Self {
103            listeners: BTreeMap::new(),
104            state,
105            dp,
106            graceful,
107            hardening,
108            splice_drain,
109            process_shutdown,
110            shutdown_timeout,
111        }
112    }
113
114    /// Apply a new listener set: bind new listeners, drain removed
115    /// listeners, restart changed listeners. Returns the updated
116    /// `tls_states` map (name -> TlsTermination) for terminate listeners
117    /// so the caller can wire cert-file watching.
118    pub(crate) async fn apply(
119        &mut self,
120        configured: &[Listener],
121        tls_states: &mut BTreeMap<String, Arc<TlsTermination>>,
122    ) {
123        let new_names: std::collections::BTreeSet<&str> =
124            configured.iter().map(|l| l.name.as_str()).collect();
125
126        // 1. Drain and remove listeners that are no longer present or
127        //    whose bind identity changed.
128        let mut to_remove: Vec<String> = Vec::new();
129        for (name, managed) in &self.listeners {
130            if !new_names.contains(name.as_str()) {
131                to_remove.push(name.clone());
132                continue;
133            }
134            let new_listener = configured.iter().find(|l| l.name == *name).unwrap();
135            if identity_changed(managed, new_listener) {
136                to_remove.push(name.clone());
137            }
138        }
139        for name in &to_remove {
140            self.stop_listener(name).await;
141            tls_states.remove(name);
142        }
143
144        // 2. Bind and spawn new listeners (added or changed).
145        for l in configured {
146            // Skip H3 and UDP listeners — they are managed separately in
147            // main.rs (UDP/QUIC, not TCP). The listener manager only
148            // handles TCP listeners (http, https, tcp).
149            if l.protocol == ListenerProtocol::H3 || l.protocol == ListenerProtocol::Udp {
150                continue;
151            }
152            // Already managed and unchanged — skip.
153            if self.listeners.contains_key(&l.name) {
154                continue;
155            }
156            match self.spawn_listener(l).await {
157                Ok(Some(term)) => {
158                    // Terminate listener: register TLS state for cert
159                    // watching.
160                    tls_states.insert(l.name.clone(), term);
161                }
162                Ok(None) => {}
163                Err(err) => {
164                    tracing::error!(
165                        code = "listener_bind_failed",
166                        listener = %l.name,
167                        "failed to bind listener: {err}"
168                    );
169                }
170            }
171        }
172    }
173
174    /// Stop and drain one listener. Signals its per-listener shutdown
175    /// watch, then waits for the accept task to finish (with a bounded
176    /// drain timeout).
177    async fn stop_listener(&mut self, name: &str) {
178        let Some(managed) = self.listeners.remove(name) else {
179            return;
180        };
181        tracing::info!(
182            code = "listener_draining",
183            listener = %name,
184            addr = %managed.addr,
185            "draining listener (removed or changed)"
186        );
187        let _ = managed.shutdown_tx.send(());
188        let task = managed.task;
189        // Wait for the accept task to finish, with a bounded timeout.
190        let drain = tokio::time::timeout(LISTENER_DRAIN_TIMEOUT, task);
191        match drain.await {
192            Ok(Ok(())) => {}
193            Ok(Err(err)) => {
194                tracing::warn!(
195                    code = "listener_task_error",
196                    listener = %name,
197                    "listener task ended with error: {err}"
198                );
199            }
200            Err(_) => {
201                tracing::warn!(
202                    code = "listener_drain_timeout",
203                    listener = %name,
204                    "listener drain timed out after {:?}; aborting",
205                    LISTENER_DRAIN_TIMEOUT
206                );
207                // The task was consumed by the timeout; we can't abort
208                // it directly, but it will be dropped when the timeout
209                // future is dropped (which happens here).
210            }
211        }
212        tracing::info!(
213            code = "listener_drained",
214            listener = %name,
215            "listener drained"
216        );
217    }

219    /// Bind and spawn one listener. On success, the listener is
220    /// tracked in `self.listeners`. Returns the `TlsTermination` if
221    /// the listener is a terminate-mode HTTPS listener (for cert
222    /// watcher registration).
223    async fn spawn_listener(
224        &mut self,
225        l: &Listener,
226    ) -> Result<Option<Arc<TlsTermination>>, Box<dyn std::error::Error + Send + Sync>> {
227        let (tcp, bound) = bind_listener(l).await?;
228        let addr = bound.addr.clone();
229        let protocol = l.protocol;
230        let proxy_protocol = l.proxy_protocol;
231        let tls_mode = l.tls.as_ref().map(|t| t.mode);
232
233        // Extract TLS termination state before moving bound.
234        let term = match &bound.mode {
235            ListenerMode::Terminate(term) => Some(Arc::clone(term)),
236            _ => None,
237        };
238
239        let name = bound.name.clone();
240        let (shutdown_tx, shutdown_rx) = watch::channel(());
241        // Cascade: when the process-wide shutdown fires, signal this
242        // listener's shutdown too. We spawn a tiny forwarder task.
243        let process_shutdown = self.process_shutdown.clone();
244        let listener_shutdown_tx = shutdown_tx.clone();
245        tokio::spawn(async move {
246            let mut process_shutdown = process_shutdown;
247            process_shutdown.changed().await.ok();
248            let _ = listener_shutdown_tx.send(());
249        });
250
251        let state = Arc::clone(&self.state);
252        let dp = Arc::clone(&self.dp);
253        let graceful = Arc::clone(&self.graceful);
254        let hardening = Arc::clone(&self.hardening);
255        let splice_drain = Arc::clone(&self.splice_drain);
256        let timeout = self.shutdown_timeout;
257
258        tracing::info!(
259            code = "listening",
260            addr = %addr,
261            listener = %name,
262            mode = match &bound.mode {
263                ListenerMode::Cleartext => "cleartext http/1.1+h2c",
264                ListenerMode::Terminate(_) => "tls terminate",
265                ListenerMode::Passthrough => "tls passthrough",
266                ListenerMode::L4 { .. } => "l4 tcp proxy",
267            },
268            "listener bound (hot-reload)"
269        );
270
271        let task = tokio::spawn(run_listener_supervised(
272            bound,
273            Arc::new(tcp),
274            state,
275            dp,
276            graceful,
277            shutdown_rx,
278            timeout,
279            hardening,
280            splice_drain,
281        ));
282
283        self.listeners.insert(
284            name.clone(),
285            ManagedListener {
286                addr,
287                protocol,
288                proxy_protocol,
289                tls_mode,
290                shutdown_tx,
291                task,
292            },
293        );
294        Ok(term)
295    }

297    /// Signal all listeners to stop (used during process shutdown).
298    /// The process-wide shutdown watch already cascades, but this
299    /// ensures all per-listener watches are signalled too.
300    pub(crate) fn signal_all(&self) {
301        for managed in self.listeners.values() {
302            let _ = managed.shutdown_tx.send(());
303        }
304    }
305
306    /// Wait for all listener tasks to finish (used during process
307    /// shutdown drain).
308    pub(crate) async fn join_all(&mut self) {
309        let listeners = std::mem::take(&mut self.listeners);
310        for (name, managed) in listeners {
311            let _ = managed.shutdown_tx.send(());
312            if let Err(err) = managed.task.await {
313                tracing::warn!(
314                    code = "listener_task_error",
315                    listener = %name,
316                    "listener task ended with error: {err}"
317                );
318            }
319        }
320    }
