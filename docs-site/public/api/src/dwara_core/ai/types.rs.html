1//! Canonical chat-completion types (DW-075).
2//!
3//! One request/response vocabulary that every adapter translates to and
4//! from. The canonical shape is a SUPERSET of the OpenAI
5//! chat-completions surface (the facade dialect clients speak — see
6//! [`crate::ai::openai_compat`]) carrying exactly the fields the three
7//! shipped provider dialects can express; anything a client sends that
8//! the canonical surface does not model is preserved verbatim in
9//! [`ChatRequest::other`] and re-emitted only by the OpenAI adapter
10//! (dialect-specific parameters cannot be translated honestly, so the
11//! Anthropic/Gemini adapters DROP them — documented behavior, not
12//! silent corruption).
13//!
14//! Token accounting is PROVIDER-REPORTED ONLY (locked M4 decision): the
15//! gateway never estimates token counts itself. [`Usage`] fields are
16//! optional because providers report subsets (Anthropic streams split
17//! input/output tokens across different events).
18

22/// The role of one chat message.
23#[derive(Debug, Clone, Copy, PartialEq, Eq)]
24pub enum ChatRole {
25    System,
26    User,
27    Assistant,
28    Tool,
29}

31impl ChatRole {
32    /// The lowercase wire name shared by the OpenAI dialect and the
33    /// canonical facade.
34    pub fn as_str(&self) -> &'static str {
35        match self {
36            ChatRole::System => "system",
37            ChatRole::User => "user",
38            ChatRole::Assistant => "assistant",
39            ChatRole::Tool => "tool",
40        }
41    }
42}

44/// One part of a message's content. Multimodal messages carry parts;
45/// plain-text messages carry exactly one [`ContentPart::Text`].
46#[derive(Debug, Clone, PartialEq)]
47pub enum ContentPart {
48    /// A text fragment.
49    Text { text: String },
50    /// An image. OpenAI clients send `url` (which may be a `data:`
51    /// URI); Gemini uses inline base64 `data` + `media_type`. Both
52    /// fields are carried so a `data:` URI decomposes for Gemini and
53    /// base64 re-composes for OpenAI.
54    Image {
55        /// The URL form (OpenAI `image_url.url`); may be a `data:` URI.
56        url: Option<String>,
57        /// MIME type when known (Gemini `inline_data.mime_type`, or the
58        /// decoded `data:` URI prefix).
59        media_type: Option<String>,
60        /// Base64 image bytes when known (Gemini `inline_data.data`, or
61        /// the decoded `data:` URI payload).
62        data_b64: Option<String>,
63    },
64}
65
66impl ContentPart {
67    /// The text of a Text part; None for images.
68    pub fn as_text(&self) -> Option<&str> {
69        match self {
70            ContentPart::Text { text } => Some(text),
71            ContentPart::Image { .. } => None,
72        }
73    }
74
75    /// An Image part from an OpenAI `image_url.url` value: when the URL
76    /// is a `data:<mime>;base64,<payload>` URI it is DECOMPOSED (the
77    /// media type and base64 payload fill the fields the Anthropic and
78    /// Gemini translations need); a remote URL stays url-only (those
79    /// dialects cannot fetch it — see the adapters' docs).
80    pub fn image_from_openai_url(url: &str) -> ContentPart {
81        if let Some((mime, data)) = split_data_uri(url) {
82            ContentPart::Image {
83                url: Some(url.to_string()),
84                media_type: Some(mime),
85                data_b64: Some(data),
86            }
87        } else {
88            ContentPart::Image {
89                url: Some(url.to_string()),
90                media_type: None,
91                data_b64: None,
92            }
93        }
94    }
95}

97/// Split a `data:<mime>;base64,<payload>` URI into its parts. None for
98/// anything else (remote URLs, malformed data URIs).
99pub(crate) fn split_data_uri(url: &str) -> Option<(String, String)> {
100    let rest = url.strip_prefix("data:")?;
101    let (mime, payload) = rest.split_once(',')?;
102    let mime = mime.strip_suffix(";base64")?.to_string();
103    if mime.is_empty() || payload.is_empty() {
104        return None;
105    }
106    Some((mime, payload.to_string()))
107}

109/// A tool call issued by the assistant. `arguments` is the JSON-encoded
110/// arguments object as a STRING (the OpenAI convention); the Anthropic
111/// and Gemini adapters parse/serialize around their native object
112/// forms.
113#[derive(Debug, Clone, PartialEq)]
114pub struct ToolCall {
115    pub id: String,
116    pub name: String,
117    pub arguments: String,
118}
119
120impl ToolCall {
121    /// Parse `arguments` as a JSON object; an empty or non-object value
122    /// yields an empty map (the OpenAI spec allows the string "null"
123    /// fragments mid-stream; non-object arguments are not expressible
124    /// in the other dialects anyway).
125    pub fn arguments_object(&self) -> BTreeMap<String, Value> {
126        self.arguments
127            .trim()
128            .parse::<Value>()
129            .ok()
130            .and_then(|v| v.as_object().cloned())
131            .map(|map| map.into_iter().collect::<BTreeMap<String, Value>>())
132            .unwrap_or_default()
133    }
134}

136/// One chat message in the canonical shape.
137#[derive(Debug, Clone, PartialEq)]
138pub struct ChatMessage {
139    pub role: ChatRole,
140    pub content: Vec<ContentPart>,
141    /// Optional participant name (OpenAI `name`).
142    pub name: Option<String>,
143    /// Tool calls issued by an assistant message.
144    pub tool_calls: Vec<ToolCall>,
145    /// For Tool-role messages: the call being answered.
146    pub tool_call_id: Option<String>,
147}
148
149impl ChatMessage {
150    /// A plain text message.
151    pub fn text(role: ChatRole, text: impl Into<String>) -> Self {
152        ChatMessage {
153            role,
154            content: vec![ContentPart::Text { text: text.into() }],
155            name: None,
156            tool_calls: Vec::new(),
157            tool_call_id: None,
158        }
159    }
160
161    /// Concatenated text of all Text parts ("" when the message is
162    /// image-only).
163    pub fn text_content(&self) -> String {
164        self.content
165            .iter()
166            .filter_map(|p| p.as_text())
167            .collect::<Vec<_>>()
168            .join("")
169    }
170}

197/// AI-06 (#196): Structured-output request format. Normalized across
198/// provider dialects so the gateway can translate `response_format`
199/// from OpenAI to Anthropic and Gemini.
200#[derive(Debug, Clone, PartialEq)]
201pub enum ResponseFormat {
202    /// Plain text output (no structure). OpenAI `{"type": "text"}`.
203    Text,
204    /// JSON object mode (no schema). OpenAI `{"type": "json_object"}`.
205    /// Anthropic translates this to a tool with a permissive schema;
206    /// Gemini translates to `responseMimeType: "application/json"`.
207    JsonObject,
208    /// JSON Schema mode. OpenAI
209    /// `{"type": "json_schema", "json_schema": {"name": ..., "schema": ...}}`.
210    /// Anthropic translates to a tool with the schema as
211    /// `input_schema`; Gemini translates to `responseMimeType` +
212    /// `responseSchema`.
213    JsonSchema {
214        /// The schema name (OpenAI `json_schema.name`).
215        name: String,
216        /// The JSON Schema object (OpenAI `json_schema.schema`).
217        schema: Value,
218    },
219}

221/// A canonical chat-completions request: what the facade parser
222/// produces and every adapter consumes.
223#[derive(Debug, Clone, PartialEq)]
224pub struct ChatRequest {
225    /// The model ALIAS as the client named it. Adapters receive the
226    /// mapped `provider_model` separately; this field is only echoed
227    /// back to the client.
228    pub model: String,
229    pub messages: Vec<ChatMessage>,
230    pub tools: Vec<ToolSpec>,
231    pub tool_choice: Option<ToolChoice>,
232    pub temperature: Option<f64>,
233    pub top_p: Option<f64>,
234    pub max_tokens: Option<u64>,
235    pub stop: Option<Vec<String>>,
236    /// Whether the client asked for SSE streaming. DW-075 rejects
237    /// streaming at the GATEWAY (400) until the zero-buffer pass-through
238    /// lands (DW-077); adapters still translate delta shapes so the
239    /// streaming pipeline composes later without adapter changes.
240    pub stream: bool,
241    /// OpenAI `stream_options.include_usage` (streaming only).
242    pub stream_options_include_usage: bool,
243    /// AI-06 (#196): structured-output format. Parsed from OpenAI
244    /// `response_format` and translated by each adapter.
245    pub response_format: Option<ResponseFormat>,
246    /// AI-14 (#202): server-side prompt template reference. When set,
247    /// the gateway resolves the named prompt template (from
248    /// `ai.experiments.prompts`), substitutes `prompt_variables` into
249    /// `{{var}}` placeholders, and prepends the result as a system
250    /// message before any existing system message. Format:
251    /// `"prompt_name"` (uses active version) or
252    /// `"prompt_name/version_name"`.
253    pub prompt: Option<String>,
254    /// AI-14 (#202): variables for prompt template substitution. Each
255    /// key maps to a `{{key}}` placeholder in the template body.
256    pub prompt_variables: BTreeMap<String, String>,
257    /// Dialect-specific parameters the canonical surface does not model
258    /// (e.g. `seed`, `presence_penalty`). Carried verbatim; only the
259    /// OpenAI adapter re-emits them.
260    pub other: BTreeMap<String, Value>,
261}

263/// Provider-reported token usage (locked decision: provider-reported
264/// only — the gateway never estimates).
265#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
266pub struct Usage {
267    pub prompt_tokens: Option<u64>,
268    pub completion_tokens: Option<u64>,
269    pub total_tokens: Option<u64>,
270    /// The number of prompt tokens served from the provider's prompt
271    /// cache (DW-AI-03). OpenAI reports this as
272    /// `usage.prompt_tokens_details.cached_tokens`; Anthropic reports
273    /// it as `usage.cache_read_input_tokens` (the adapter normalizes
274    /// both into this field). When present, the cost computation prices
275    /// the cached portion at the model's `cached_input_per_1k_micros`
276    /// rate (when declared) instead of the standard input rate.
277    pub cached_tokens: Option<u64>,
278}
279
280impl Usage {
281    /// Fold a later usage report into this one (streaming providers
282    /// report input tokens in an opening event and output tokens in a
283    /// closing one; each side is filled in as it arrives). When both
284    /// sides are known and the provider never sent a total, the total
285    /// is DERIVED (prompt + completion) — the same rule the
286    /// non-streaming Anthropic parse applies, so a provider's streamed
287    /// and non-streamed totals agree. The `cached_tokens` field takes
288    /// the max of the two reports (a provider may revise the count
289    /// upward as more cache hits are reported).
290    pub fn merge(&mut self, later: Usage) {
291        if later.prompt_tokens.is_some() {
292            self.prompt_tokens = later.prompt_tokens;
293        }
294        if later.completion_tokens.is_some() {
295            self.completion_tokens = later.completion_tokens;
296        }
297        if later.total_tokens.is_some() {
298            self.total_tokens = later.total_tokens;
299        }
300        if self.total_tokens.is_none() {
301            if let (Some(p), Some(c)) = (self.prompt_tokens, self.completion_tokens) {
302                self.total_tokens = Some(p + c);
303            }
304        }
305        self.cached_tokens = match (self.cached_tokens, later.cached_tokens) {
306            (Some(a), Some(b)) => Some(a.max(b)),
307            (a, b) => a.or(b),
308        };
309    }
310}

312/// Why generation stopped, normalized across dialects.
313#[derive(Debug, Clone, PartialEq, Eq)]
314pub enum FinishReason {
315    Stop,
316    Length,
317    ToolCalls,
318    ContentFilter,
319    Other(String),
320}
321
322impl FinishReason {
323    /// The OTel GenAI `gen_ai.response.finish_reasons` token (AI-07):
324    /// the lowercase reason string. `Other` carries a provider-specific
325    /// string verbatim (already lowercase-normalized by the adapters).
326    pub fn as_gen_ai(&self) -> &str {
327        match self {
328            FinishReason::Stop => "stop",
329            FinishReason::Length => "length",
330            FinishReason::ToolCalls => "tool_calls",
331            FinishReason::ContentFilter => "content_filter",
332            FinishReason::Other(s) => s.as_str(),
333        }
334    }
335}
