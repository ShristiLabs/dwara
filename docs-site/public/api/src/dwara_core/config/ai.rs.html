1//! The `ai:` configuration block (DW-075): AI provider adapters.
2//!
3//! This is the source shape only — the runtime translation lives in the
4//! AI domain one level up (see the `ai` module's `AiRuntime`), which
5//! consumes these types at compile time. The block is ADDITIVE to the
6//! gateway schema: absent (the default), the gateway has no AI surface
7//! at all; present, it declares the provider pool (`providers:`) and
8//! the model alias table (`models:`) that map an OpenAI-shaped client
9//! request onto a concrete provider call.
10//!
11//! Design notes (DW-075):
12//!
13//! - A provider names an UPSTREAM (`upstreams:` entry), not a raw base
14//!   URL: the provider's transport — endpoint set, TLS trust, connection
15//!   pooling, timeouts, breaker, passive health — is the SAME machinery
16//!   every other upstream gets, and the adapter layer stays pure
17//!   request/response translation with no HTTP client of its own. A
18//!   multi-region provider pool is then just an upstream with several
19//!   endpoints.
20//! - Authentication is a verbatim header (`auth.header` + `auth.value`)
21//!   rather than a per-provider auth enum: providers disagree on the
22//!   convention (OpenAI wants `Authorization: Bearer ...`, Anthropic
23//!   `x-api-key: ...`, Gemini `x-goog-api-key: ...`), and a verbatim
24//!   pair expresses all of them without the schema growing a variant
25//!   per provider. `value` is secret-bearing: inline values are
26//!   redacted in every config echo (DW-045) and `${...}` references
27//!   resolve at config-compile time, failing the generation closed when
28//!   unresolvable.
29//! - `models:` is a map from the MODEL ALIAS the client names in the
30//!   request body to the provider that serves it plus the provider's own
31//!   model identifier. The alias is what the gateway echoes back in the
32//!   response; the provider model never leaks to the client.

43/// The `ai:` top-level block (DW-075). Absent: no AI surface. Present:
44/// validation requires at least one provider and at least one model —
45/// an `ai:` block that can never serve a model is an authoring error,
46/// not a silently inert config.
47#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
48#[serde(deny_unknown_fields)]
49pub struct AiConfig {
50    /// The provider pool. Each entry names an upstream that carries the
51    /// transport and the adapter `kind` that carries the wire translation.
52    #[serde(default, skip_serializing_if = "Vec::is_empty")]
53    pub providers: Vec<AiProvider>,
54    /// Model alias table: the `model` value a client puts in its request
55    /// body maps to the provider that serves it and the provider's own
56    /// model identifier. The alias is echoed back to the client; the
57    /// `provider_model` never leaves the gateway.
58    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
59    pub models: BTreeMap<String, AiModel>,
60    /// Per-model pricing table (DW-079): the key is the PROVIDER MODEL
61    /// identifier (the actual model the provider charges for, e.g.
62    /// `gpt-4o-mini`), NOT the client-facing alias. Costs are integer
63    /// micro-USD per 1 000 tokens (1 000 000 micro-USD = $1.00 — no
64    /// floating-point money). Spend = usage x price, aggregated per
65    /// key/team/model (DW-079). Absent (the default): no pricing, so
66    /// cost attribution records zero cost (the budget cost window and
67    /// the spend store stay live but inert until prices are declared).
68    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
69    pub pricing: BTreeMap<String, AiPricing>,
70    /// Model governance (DW-084): per-team (policy) model allowlists
71    /// and the shadow-audit switch. Absent (the default): no
72    /// governance — every consumer may call every configured alias,
73    /// and no governance audit events are recorded.
74    #[serde(default, skip_serializing_if = "Option::is_none")]
75    pub governance: Option<AiGovernance>,
76    /// Prompt/response logging (DW-081): opt-in capture of the
77    /// canonical prompt and response with PII redaction, sampling,
78    /// and retention. Absent (the default): no capture — privacy-first
79    /// (no prompt or response text is ever stored). When present and
80    /// `enabled: true`, a redaction pass scrubs PII/secrets before
81    /// storage, `sample_rate` controls volume, and `retention_secs`
82    /// ages records out. Per-consumer `ai_logging` overrides respect
83    /// tenant preference (see [`crate::config::Consumer::ai_logging`]).
84    #[serde(default, skip_serializing_if = "Option::is_none")]
85    pub logging: Option<AiLogging>,
86    /// Guardrails (DW-082): prompt-injection heuristics, PII
87    /// detection, banned-content filters, and output schema
88    /// enforcement as a middleware chain on the AI proxy action.
89    /// Absent (the default): no guardrails — every prompt and
90    /// response passes through uninspected. When present, each rule
91    /// inspects the prompt (before the provider call) and/or the
92    /// response (after), and blocks, redacts, or logs per its
93    /// `action`. Policy-scoped rules apply only to consumers
94    /// attaching a listed policy; an empty `policies` list applies
95    /// to all. Schema enforcement uses the unconditional `jsonschema`
96    /// dependency and works in every build.
97    #[serde(default, skip_serializing_if = "Option::is_none")]
98    pub guardrails: Option<AiGuardrails>,
99    /// Semantic cache (DW-083 `ai.semantic_cache`): an
100    /// embedding-similarity cache for AI prompts. A paraphrased
101    /// prompt within the cosine-similarity threshold returns the
102    /// cached response with no provider call and no token spend.
103    /// Uses an external embedding service (OpenAI-compatible
104    /// /v1/embeddings API) and a pure-Rust HNSW ANN index
105    /// (`hnsw_rs`). Feature-gated behind the `semantic_cache` cargo
106    /// feature; without it the config is accepted but the cache is a
107    /// no-op. Absent (the default): no semantic cache.
108    #[serde(default, skip_serializing_if = "Option::is_none")]
109    pub semantic_cache: Option<SemanticCacheConfig>,
110    /// Routing policies (DW-085 `ai.routing_policies`): named
111    /// within-request escalation or latency-vs-cost selection plans,
112    /// keyed by name. A model alias declares `routing_policy: <name>`
113    /// to be served by one; the policy is evaluated per request (a
114    /// FallbackChain policy may call an external classifier service;
115    /// a LatencyCost policy picks deterministically at compile time).
116    /// Absent (the default): no routing policies.
117    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
118    pub routing_policies: BTreeMap<String, AiRoutingPolicy>,
119    /// Prompt experimentation (DW-086 `ai.experiments`): prompt
120    /// versioning, A/B model comparison, regression evals, and
121    /// feedback ingestion. Absent (the default): no experiments.
122    #[serde(default, skip_serializing_if = "Option::is_none")]
123    pub experiments: Option<AiExperiments>,
124    /// MCP gateway (DW-087 `ai.mcp`): turns dwara into an MCP
125    /// (Model Context Protocol) server/router. Configured tools are
126    /// exposed over JSON-RPC 2.0 on a reserved HTTP path (default
127    /// `/mcp`); the gateway proxies tool calls to upstream HTTP
128    /// endpoints with authN/authZ, manages agent sessions in the
129    /// state store, and correlates tool calls in analytics. Absent
130    /// (the default): no MCP surface (the `/mcp` path returns 404).
131    #[serde(default, skip_serializing_if = "Option::is_none")]
132    pub mcp: Option<AiMcpConfig>,
133    /// A2A (agent-to-agent) protocol support (DW-114 `ai.a2a`):
134    /// configures remote A2A agents that the gateway can route chat
135    /// requests to. Each agent names an upstream (the transport) and
136    /// an Agent Card (discovery doc, file path or inline JSON). The
137    /// adapter translates a canonical chat request into an A2A
138    /// task-submit body; the task lifecycle is STUBBED pending spec
139    /// freeze. Compiled into every build (no cargo feature). Absent
140    /// (the default): no A2A surface.
141    #[serde(default, skip_serializing_if = "Option::is_none")]
142    pub a2a: Option<A2aConfig>,
143    /// Unknown-model pricing policy (DW-AI-03
144    /// `ai.unknown_model_policy`): what to do when a provider model
145    /// has no entry in `ai.pricing`. Absent (the default): `allow`
146    /// (cost 0, fail-open — the original behavior). `fail_closed`
147    /// rejects the response with 403 `ai_unknown_model_pricing`;
148    /// `alert` proceeds with cost 0 but logs a WARN and increments

150    #[serde(default, skip_serializing_if = "Option::is_none")]
151    pub unknown_model_policy: Option<UnknownModelPolicy>,

154/// Model governance (DW-084 `ai.governance`): per-team model
155/// allowlists plus a shadow-audit switch. A team is a POLICY name
156/// (the same vocabulary the token budgets use for `scope: policy`):
157/// every consumer attaching a policy with an allowlist entry is
158/// restricted to that allowlist's aliases. The check runs BEFORE
159/// routing, against the client-facing alias (the `model` value in
160/// the request body), so a typo or a renamed alias is blocked at the
161/// edge rather than surfacing as a provider 404. Multiple policies
162/// with allowlists that bind one request intersect (deny-wins): the
163/// model must be in EVERY binding allowlist, matching the authz
164/// deny-anywhere-wins principle. Consumers with no binding allowlist
165/// policy are unrestricted (fail-open, the DW-017 default posture).
166/// `audit: true` records BOTH allowed and denied attempts into the
167/// `ai_governance_events` analytics table for shadow review.
168#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
169#[serde(deny_unknown_fields)]
170pub struct AiGovernance {
171    /// Per-policy (team) model allowlists. The key is the POLICY name
172    /// (the team), the value is the list of allowed client-facing
173    /// model aliases. A consumer attaching a policy listed here may
174    /// only call the aliases in its allowlist; a consumer attaching
175    /// no listed policy is unrestricted. Validation rejects an
176    /// allowlist entry that names an alias absent from `ai.models`
177    /// (an authoring error, not a runtime 404).
178    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
179    pub team_allowlists: BTreeMap<String, Vec<String>>,
180    /// When true, record model usage for shadow audit: BOTH allowed
181    /// calls and denied attempts land in the `ai_governance_events`
182    /// analytics table (the admin `/analytics/governance-audit`
183    /// endpoint reads it). When false (the default), only the
184    /// denial metric fires — no per-event audit rows are written.
185    #[serde(default, skip_serializing_if = "is_false")]
186    pub audit: bool,
187    /// SEC-14 / CFG-14: dry-run mode. When true, the governance engine
188    /// evaluates the team allowlist and records would-be denials
189    /// (`dwara_policy_dry_run_total{phase="ai_governance"}` + a
190    /// structured log) but does NOT reject the request — the call
191    /// proceeds to the model. Useful for rolling out a new allowlist
192    /// without breaking existing callers. Default false (reject on
193    /// mismatch).
194    #[serde(default, skip_serializing_if = "is_false")]
195    pub dry_run: bool,
196}

198/// One AI provider (DW-075 `ai.providers[]`).
199#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
200#[serde(deny_unknown_fields)]
201pub struct AiProvider {
202    /// Provider name; unique within the `ai:` block. Referenced by
203    /// `ai.models[].provider`.
204    pub name: String,
205    /// Wire-protocol adapter kind. Selects the request/response
206    /// translation applied to every call routed to this provider.
207    pub kind: AiProviderKind,
208    /// Name of the `upstreams:` entry that carries this provider's
209    /// transport (endpoints, TLS trust, pooling, timeouts, breaker).
210    /// The adapter builds provider-specific paths; the endpoint's
211    /// authority becomes the dialed host, exactly like a proxy route.
212    pub upstream: String,
213    /// Authentication attached to every call to this provider: a
214    /// verbatim header. Optional — a provider behind an internal network
215    /// (a self-hosted gateway fronting an Ollama-style endpoint, for
216    /// example) may need none.
217    #[serde(default, skip_serializing_if = "Option::is_none")]
218    pub auth: Option<AiProviderAuth>,
219    /// Full request path override for this provider. When set, the
220    /// transport uses this path verbatim instead of the adapter's
221    /// built-in path. This bridges providers whose API path differs
222    /// from the adapter's default (e.g. z.ai's coding-plan endpoint
223    /// `/api/coding/paas/v4/chat/completions` vs the OpenAI adapter's
224    /// `/v1/chat/completions`) without an external reverse proxy.
225    /// When unset, the adapter's per-kind default path is used.
226    /// Applies to both adapter-translated and passthrough requests.
227    #[serde(default, skip_serializing_if = "Option::is_none")]
228    pub path: Option<String>,
229    /// DW-080 (Ent): a credential pool for this provider. When set,
230    /// the gateway rotates across multiple API keys to aggregate
231    /// rate-limit headroom (the LiteLLM pattern). Each entry is an
232    /// `AiProviderAuth` (same shape as `auth`); the pool is used
233    /// INSTEAD of `auth` — validation rejects a provider that sets
234    /// both. At request time the pool picks one credential per
235    /// request (round-robin by default); when a key returns 429, it
236    /// is temporarily quarantined and subsequent requests rotate to
237    /// the next. Pool exhaustion (all keys quarantined) degrades
238    /// gracefully with a 429 + Retry-After, not a panic. Ent-only:
239    /// rejected at validation when the `ent` cargo feature is off.
240    #[serde(default, skip_serializing_if = "Option::is_none")]
241    pub credential_pool: Option<AiCredentialPool>,
242}

247pub enum AiProviderKind {
248    /// OpenAI chat-completions API (`POST /v1/chat/completions`). Also
249    /// the dialect spoken by OpenAI-compatible servers (vLLM, Ollama's
250    /// compatibility endpoint, Groq, DeepSeek, Mistral, Cohere, and
251    /// others) — point the upstream at one of those and this adapter
252    /// speaks to it unchanged (AI-01).
253    Openai,
254    /// Anthropic messages API (`POST /v1/messages`).

257    /// (`POST /v1beta/models/{model}:generateContent`).
258    Gemini,
259    /// DW-114: A2A (agent-to-agent) protocol. An A2A provider is a
260    /// configured `ai.a2a.agents[]` entry: the adapter translates a
261    /// canonical chat request into an A2A task-submit body and parses
262    /// the task response back. The task lifecycle state machine is
263    /// implemented against the A2A Protocol Specification draft (legal
264    /// and illegal transitions are enforced). Compiled into every
265    /// build (no cargo feature).
266    A2a,
267    /// AI-01: Azure OpenAI. Uses the Azure deployment URL format
268    /// (`/openai/deployments/{deployment}/chat/completions?api-version=...`)

270    /// is OpenAI-compatible (the adapter reuses the OpenAI body
271    /// translation and overrides the path + headers).
272    AzureOpenai,

274    /// (`/model/{model_id}/invoke`) with AWS SigV4 request signing.

300/// Verbatim authentication header for a provider (DW-075
301/// `ai.providers[].auth`). `value` is SECRET-BEARING: an inline value is
302/// replaced by the redaction placeholder in every config echo, and a
303/// `${...}` reference is resolved at config-compile time (DW-045).
304#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
305#[serde(deny_unknown_fields)]
306pub struct AiProviderAuth {
307    /// Header name (e.g. `Authorization`, `x-api-key`,
308    /// `x-goog-api-key`). Must be a valid HTTP header name.
309    pub header: String,
310    /// Header value, verbatim (e.g. `Bearer ${OPENAI_API_KEY}` —
311    /// reference resolution covers the whole value, so the `Bearer `
312    /// prefix belongs INSIDE the referenced variable or the literal).
313    /// Inline values are redacted in config echoes; never logged.
314    pub value: String,
315}

317/// DW-080 (Ent): a credential pool for a provider — multiple API keys
318/// rotated across to aggregate rate-limit headroom. Each entry is an
319/// `AiProviderAuth` (same header+value shape as the singular `auth`
320/// field). The pool is used INSTEAD of `auth`; validation rejects a
321/// provider that sets both.
322#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
323#[serde(deny_unknown_fields)]
324pub struct AiCredentialPool {
325    /// One or more credential entries. Each entry has the same shape
326    /// as `ai.providers[].auth` (a header name + secret-bearing value).
327    /// At least two entries are required (a single-entry pool is
328    /// equivalent to `auth` — validation rejects it).
329    pub credentials: Vec<AiProviderAuth>,
330    /// Rotation strategy (DW-080). `round_robin` (the default) cycles
331    /// through credentials in config order; `weighted` picks
332    /// proportionally to each entry's `weight` (deterministic per
333    /// request id, same as canary splits). When a credential is
334    /// quarantined (429 from the provider), the pool skips it and
335    /// picks the next available one.
336    #[serde(default, skip_serializing_if = "is_default_pool_strategy")]
337    pub strategy: AiPoolStrategy,
338    /// Quarantine duration in seconds for a credential that received a
339    /// 429 from the provider. During quarantine, the pool skips the
340    /// credential; after the window expires, it re-enters rotation.
341    /// Default: 60 seconds. A provider's `Retry-After` header, when
342    /// present, overrides this per-429 (capped at 600 seconds to
343    /// prevent a hostile provider from pinning a key out indefinitely).
344    #[serde(
345        default = "default_quarantine_secs",
346        skip_serializing_if = "is_default_quarantine_secs"
347    )]
348    pub quarantine_secs: u64,
349}

377/// One model alias mapping (DW-075 `ai.models{alias}`).
378#[derive(Debug, Clone, PartialEq, Serialize, Deserialize, JsonSchema)]
379#[serde(deny_unknown_fields)]
380pub struct AiModel {
381    /// Name of the `ai.providers[]` entry that serves this alias.
382    pub provider: String,
383    /// The provider's own model identifier (e.g. `gpt-4o-mini`,
384    /// `claude-sonnet-4-5`, `gemini-2.5-flash`), sent to the provider
385    /// in place of the alias.
386    pub provider_model: String,
387    /// Ordered failover chain (DW-076): when the primary target
388    /// answers 429 or a 5xx, or its transport errors, the gateway
389    /// retries the next entry — a DIFFERENT provider/model pair, never
390    /// a re-send to the provider that just failed (its upstream's
391    /// breaker owns same-provider retries). The client sees one
392    /// response. Absent (the default): no failover, the primary's
393    /// answer is final. Cannot be combined with `canary` on the same
394    /// alias (validation rejects the pairing): failover is an
395    /// availability chain for a single logical model; a canary split
396    /// is deliberately serving multiple versions at once — mixing the
397    /// two would retry a canary request onto the stable version and
398    /// silently undo the experiment.
399    #[serde(default, skip_serializing_if = "Vec::is_empty")]
400    pub failover: Vec<AiModelTarget>,
401    /// Weighted canary split (DW-076): traffic for this alias
402    /// distributes across the listed versions by a deterministic
403    /// weighted hash of the request id (the same slot semantics as
404    /// traffic splitting: ratios hold per request, and re-sending a
405    /// request with the same id lands on the same version). Each
406    /// version names exactly one provider/model pair and carries no
407    /// failover of its own. Absent (the default): no split. Cannot be
408    /// combined with `failover` on the same alias.
409    #[serde(default, skip_serializing_if = "Vec::is_empty")]
410    pub canary: Vec<AiCanaryVersion>,
411    /// A routing policy name (DW-085): when set, this alias is served
412    /// by the named entry in `ai.routing_policies` instead of a plain
413    /// failover chain or canary split. The policy is evaluated per
414    /// request (a FallbackChain policy may call an external classifier
415    /// service to decide cheap-vs-expensive; a LatencyCost policy
416    /// picks deterministically). Cannot be combined with `failover`
417    /// or `canary` on the same alias (validation enforces mutual
418    /// exclusivity): a policy alias composes over OTHER aliases'
419    /// routing plans, so it has no primary provider/model pair of its
420    /// own.
421    #[serde(default, skip_serializing_if = "Option::is_none")]
422    pub routing_policy: Option<String>,
423    /// An A/B test name (DW-086): when set, this alias is served by
424    /// the named entry in `ai.experiments.ab_tests` instead of a
425    /// plain failover chain, canary split, or routing policy. The
426    /// experiment is evaluated per request (a deterministic weighted
427    /// pick by request id selects a variant). Cannot be combined
428    /// with `failover`, `canary`, or `routing_policy` on the same
429    /// alias (validation enforces mutual exclusivity): an experiment
430    /// alias composes over OTHER aliases' routing plans, so it has
431    /// no primary provider/model pair of its own.
432    #[serde(default, skip_serializing_if = "Option::is_none")]
433    pub ab_test: Option<String>,
434    /// Auto-canary analysis (DW-091): when set, a background
435    /// controller compares the canary (second version) vs the
436    /// baseline (first version) on error rate or latency and adjusts
437    /// the canary split weights automatically. Only valid when the
438    /// canary has exactly 2 versions (baseline + canary). Weight
439    /// changes are transient (Generation swap) and revert on config
440    /// reload.
441    #[serde(default, skip_serializing_if = "Option::is_none")]
442    pub canary_analysis: Option<crate::config::CanaryAnalysis>,
443}

476/// Per-model token pricing (DW-079 `ai.pricing.<provider_model>`).
477/// Costs are integer MICRO-USD per 1 000 tokens (1 000 000 micro-USD =
478/// $1.00 — no floating-point money). The key in the parent map is the
479/// PROVIDER MODEL identifier (the actual model the provider charges
480/// for), not the client-facing alias. Spend is computed as
481/// `input_tokens * input_per_1k_micros / 1000 + output_tokens *
482/// output_per_1k_micros / 1000` (integer micro-USD, saturating).
