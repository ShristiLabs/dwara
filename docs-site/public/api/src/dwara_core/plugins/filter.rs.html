1//! The native filter trait and its shared outcome/response types (DW-119).
2//!
3//! [`NativeFilter`] is a dyn-compatible trait mirroring the proxy-wasm
4//! host's phase contract. A compiled-in Rust filter implements it and
5//! registers a factory with [`super::NativeRegistry`]; the unified
6//! [`super::PluginChain`] dispatches to it at the phases the filter's
7//! config declares, exactly as it dispatches to a WASM plugin instance.
8//!
9//! [`LocalResponse`] lives here (in the `plugins` domain) because it is
10//! the shared short-circuit shape both native filters and WASM plugins
11//! produce. The `wasm` host re-exports its own `wasm::host::LocalResponse`
12//! (defined before this domain existed); the two are structurally
13//! identical and converted at the dispatch boundary by the `wasm`
14//! adapter. Keeping the canonical type in the lower `plugins` domain
15//! avoids an upward dependency from `plugins` to `wasm`.
16
17use std::fmt;
18
19/// A local response short-circuiting the request pipeline.
20///
21/// Produced by a native filter via [`FilterOutcome::LocalResponse`] or
22/// by a WASM plugin via `proxy_send_http_response`. The proxy returns
23/// this response immediately, skipping all subsequent phases and the
24/// upstream. Structurally identical to `wasm::host::LocalResponse`; the
25/// `wasm` adapter converts between the two at the dispatch boundary.
26#[derive(Clone, Debug, PartialEq, Eq)]
27pub struct LocalResponse {
28    /// HTTP status code (e.g. 403, 200).
29    pub status: u16,
30    /// Response headers as (name, value) pairs.
31    pub headers: Vec<(String, String)>,
32    /// Response body bytes.
33    pub body: Vec<u8>,
34}
35
36/// The outcome of a native filter phase callback.
37///
38/// Mirrors `wasm::runner::PhaseOutcome` so a native filter and a WASM
39/// plugin are interchangeable in the unified chain:
40///
41/// - [`FilterOutcome::Continue`] -- proceed to the next plugin/phase.
42///   The (possibly modified) headers/body are threaded through.
43/// - [`FilterOutcome::LocalResponse`] -- short-circuit with a local
44///   response; the proxy returns it immediately.
45/// - [`FilterOutcome::Error`] -- the filter failed (the proxy returns a
46///   500, mirroring a WASM trap). The message is logged and never
47///   leaked to the client.
48#[derive(Clone, Debug)]
49pub enum FilterOutcome {
50    /// Continue processing. The headers/body passed in (possibly
51    /// modified by the filter) are returned to the chain.
52    Continue {
53        /// The (possibly modified) headers (request_headers /
54        /// response_headers phases) or empty for body phases.
55        headers: Vec<(String, String)>,
56        /// The (possibly modified) body (request_body / response_body
57        /// phases) or empty for header phases.
58        body: Vec<u8>,
59    },
60    /// Short-circuit with a local response.
61    LocalResponse(LocalResponse),
62    /// The filter errored. The proxy returns a 500 (mirroring a WASM
63    /// trap); the message is logged, never sent to the client.
64    Error(String),
65}
66
67/// A compile-in native filter (DW-119).
68///
69/// A dyn-compatible trait mirroring the proxy-wasm host's phase
70/// callbacks. Each method receives the current headers/body by value
71/// and returns a [`FilterOutcome`]. The methods are synchronous,
72/// matching the WASM runner's synchronous phase methods -- the proxy
73/// calls them synchronously per phase. A filter that does not hook a
74/// phase returns [`FilterOutcome::Continue`] with the input unchanged.
75///
76/// Implementations are registered by name via
77/// [`super::NativeRegistry::register`] with a
78/// [`super::NativeFilterFactory`]; config selects a native filter with
79/// `native: <name>` on a [`crate::config::PluginConfig`].
80pub trait NativeFilter: Send + Sync {
81    /// `request_headers` phase -- after route resolution, before authn.
82    /// Receives the request headers; returns the outcome and the
83    /// (possibly modified) headers.
84    fn on_request_headers(&mut self, headers: Vec<(String, String)>) -> FilterOutcome {
85        FilterOutcome::Continue {
86            headers,
87            body: Vec::new(),
88        }
89    }
90
91    /// `request_body` phase -- after authn/authz/rate-limit, before
92    /// upstream. Receives the request body; returns the outcome and the
93    /// (possibly modified) body.
94    fn on_request_body(&mut self, body: Vec<u8>) -> FilterOutcome {
95        FilterOutcome::Continue {
96            headers: Vec::new(),
97            body,
98        }
99    }
100
101    /// `response_headers` phase -- after the upstream responds, before
102    /// masking. Receives the response headers; returns the outcome and
103    /// the (possibly modified) headers.
104    fn on_response_headers(&mut self, headers: Vec<(String, String)>) -> FilterOutcome {
105        FilterOutcome::Continue {
106            headers,
107            body: Vec::new(),
108        }
109    }
110
111    /// `response_body` phase -- after masking, before compression.
112    /// Receives the response body; returns the outcome and the
113    /// (possibly modified) body.
114    fn on_response_body(&mut self, body: Vec<u8>) -> FilterOutcome {
115        FilterOutcome::Continue {
116            headers: Vec::new(),
117            body,
118        }
119    }
120}
121
122impl fmt::Debug for dyn NativeFilter {
123    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
124        f.write_str("NativeFilter")
125    }
126}
