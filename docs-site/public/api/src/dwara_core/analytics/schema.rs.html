1//! Analytics store schema (DW-043).
2//!
3//! The analytics database is a SEPARATE SQLite file from the state
4//! store (DW-018): different lifecycle (retention deletes vs identity
5//! upserts), different write pattern (high-churn batched appends vs
6//! request-path point lookups), and an independent bounded-disk story
7//! (incremental vacuum reclaims retention deletes here without ever
8//! compacting the identity store). It owns its own `PRAGMA user_version`
9//! namespace — a state-store database and an analytics database can
10//! never be confused for one another, and the state store's
11//! forward-only migration contract stays untouched by analytics churn.
12//!
13//! Schema v1 (this file is migration 001; v2 adds the DW-120
14//! `export_runs` ledger — see [`SCHEMA_V2`]):
15//!
16//! - `meta` — rollup cursors (see [`super::rollup`]): the exclusive
17//!   upper bound each granularity has durably rolled through, written
18//!   in the SAME transaction as the rollup rows it covers, so a crash
19//!   between rollups never double-counts a window.
20//! - `raw` — one row per completed request, exactly the redacted
21//!   [`AccessRecord`](crate::observability::AccessRecord) field set
22//!   plus a JSON object of custom dimensions. Raw retention is SHORT
23//!   (default 24 h): raw exists to seed rollups and answer
24//!   custom-dimension ad hoc queries; the durable history is the
25//!   rollup tables.
26//! - `rollup_fixed` — pre-aggregated counters per (granularity,
27//!   window, fixed dimension tuple). Latency lives as 13 fixed
28//!   per-bucket counts ([`MS_BUCKETS`] bounds) so any set of windows
29//!   merges by summation and percentiles are estimable without
30//!   storing per-request samples.
31//! - `rollup_dim` — the custom-dimension twin: one row per
32//!   (granularity, window, dimension name, value), aggregated in Rust
33//!   (the source is a JSON column; keeping the aggregation out of SQL
34//!   avoids a JSON1 feature dependency in the bundled SQLite).
35
36/// Latency bucket UPPER bounds in milliseconds (inclusive). A request
37/// lands in exactly one bucket: the first bound it does not exceed,
38/// else the overflow bucket (index 12). The bounds are the
39/// observability histogram's shape at millisecond granularity —
40/// sub-millisecond precision is noise at gateway scale, 10 s covers
41/// every classified timeout. Counts are stored NON-cumulative
42/// (per-bucket); cumulative form is computed at read time.
43pub const MS_BUCKETS: [f64; 12] = [
44    1.0, 2.0, 5.0, 10.0, 25.0, 50.0, 100.0, 250.0, 500.0, 1000.0, 2500.0, 5000.0,
45];
46
47/// Total stored bucket counts per rollup row: 12 bounded + overflow.
48pub const BUCKET_COLS: usize = MS_BUCKETS.len() + 1;
49
50/// Bucket index for one duration (the overflow bucket for anything
51/// past the last bound).
52pub fn bucket_of(duration_ms: f64) -> usize {
53    MS_BUCKETS
54        .iter()
55        .position(|b| duration_ms <= *b)
56        .unwrap_or(MS_BUCKETS.len())
57}
58
59/// Percentile estimate from non-cumulative bucket counts: the first
60/// bound whose cumulative count reaches `p * total` (else the last
61/// bound — everything past it is reported as 10 s, the honest ceiling).
62pub fn percentile(buckets: &[i64], p: f64) -> f64 {
63    let total: i64 = buckets.iter().sum();
64    if total == 0 {
65        return 0.0;
66    }
67    let target = (p * total as f64).ceil() as i64;
68    let mut cum = 0i64;
69    for (i, b) in buckets.iter().enumerate() {
70        cum += b;
71        if cum >= target {
72            return if i < MS_BUCKETS.len() {
73                MS_BUCKETS[i]
74            } else {
75                // Overflow bucket: report the last bound as the floor.
76                MS_BUCKETS[MS_BUCKETS.len() - 1]
77            };
78        }
79    }
80    MS_BUCKETS[MS_BUCKETS.len() - 1]
81}

153/// Schema v2 (DW-120): the `export_runs` ledger. One row per
154/// scheduled-or-manual usage-statement export attempt per window
155/// (primary key; a re-export REPLACES the row — the file output is

388/// Apply migrations to a fresh-or-existing analytics connection. A
389/// database at a NEWER version than this build is a hard error (the
390/// state store's forward-only rule, same rationale).
391pub fn migrate(conn: &rusqlite::Connection) -> Result<(), rusqlite::Error> {
392    let version: u32 = conn.query_row("PRAGMA user_version", [], |r| r.get(0))?;
393    if version > LATEST_SCHEMA_VERSION {
394        return Err(rusqlite::Error::ToSqlConversionFailure(Box::new(
395            std::io::Error::other(format!(
396                "analytics database schema version {version} is newer than this \
397                 build's {LATEST_SCHEMA_VERSION}; forward-only: upgrade the binary"
398            )),
399        )));
400    }
401    // Forward-only sequential migrations: each version's batch is
402    // idempotent (IF NOT EXISTS) so a crash between batches re-runs
403    // safely on the next open.
404    if version < 1 {
405        conn.execute_batch(SCHEMA_V1)?;
406        conn.pragma_update(None, "user_version", 1)?;
407    }
408    if version < 2 {
409        conn.execute_batch(SCHEMA_V2)?;
410        conn.pragma_update(None, "user_version", 2)?;
411    }
412    if version < 3 {
413        conn.execute_batch(SCHEMA_V3)?;
414        conn.pragma_update(None, "user_version", 3)?;
415    }
416    if version < 4 {
417        conn.execute_batch(SCHEMA_V4)?;
418        conn.pragma_update(None, "user_version", 4)?;
419    }
420    if version < 5 {
421        conn.execute_batch(SCHEMA_V5)?;
422        conn.pragma_update(None, "user_version", 5)?;
423    }
424    if version < 6 {
425        conn.execute_batch(SCHEMA_V6)?;
426        conn.pragma_update(None, "user_version", 6)?;
427    }
428    if version < 7 {
429        conn.execute_batch(SCHEMA_V7)?;
430        conn.pragma_update(None, "user_version", 7)?;
431    }
432    if version < 8 {
433        conn.execute_batch(SCHEMA_V8)?;
434        conn.pragma_update(None, "user_version", 8)?;
435    }
436    if version < 9 {
437        conn.execute_batch(SCHEMA_V9)?;
438        conn.pragma_update(None, "user_version", 9)?;
439    }
440    if version < 10 {
441        conn.execute_batch(SCHEMA_V10)?;
442        conn.pragma_update(None, "user_version", 10)?;
443    }
444    if version < 11 {
445        conn.execute_batch(super::partition::SCHEMA_V11)?;
446        // SCALE-09 (#188): create the initial raw_all view (just
447        // `raw` — partition tables are added by `refresh_raw_all_view`
448        // as they are created). This ensures databases that migrate
449        // to v11 have the view even if they never call
450        // `refresh_raw_all_view` directly (e.g., test databases).
451        conn.execute_batch(
452            "DROP VIEW IF EXISTS raw_all;
453             CREATE VIEW raw_all AS SELECT * FROM raw;",
454        )?;
455        conn.pragma_update(None, "user_version", 11)?;
456    }
457    if version < 12 {
458        // PERF-12 (#209): add byte counters to the raw access table.
459        conn.execute_batch(SCHEMA_V11)?;
460        conn.pragma_update(None, "user_version", 12)?;
461    }
462    Ok(())
