1//! Analytics bounded context (DW-043): the embedded analytics store.
2//!
3//! Observability answers "is it healthy right now"; analytics answers
4//! "what happened over time, to whom, and why". This domain owns the
5//! DURABLE side of that question: a local SQLite database (separate
6//! file from the state store, DW-018 — different lifecycle, different
7//! write pattern, independent bounded-disk story) holding raw access
8//! records for a short retention window and additive rollup tables at
9//! 1m/5m/1h/1d granularities with per-granularity retention, plus the
10//! query surface over them (dashboard series, Top-N, structured
11//! query) served by the admin API, and the scheduled usage-report
12//! exports (DW-120, [`exports`]) that turn the same rollups into
13//! durable per-consumer statements.
14//!
15//! # Write path (never blocks the dataplane)
16//!
17//! The dataplane's `DataPlane` (see `dataplane::proxy`) holds an
18//! [`EmbeddedAnalytics`] (set once at startup when the config carries
19//! an `analytics` block) and calls [`EmbeddedAnalytics::record`] at
20//! request completion — a `try_send` onto a bounded channel. When the
21//! channel is full the record is DROPPED (counted, logged throttled):
22//! the analytics pipeline must never slow or block a request, the same
23//! fire-and-forget posture DW-121's record stream
24//! (`events::stream`, a sibling pipeline that ships each record to an
25//! external sink instead of the local store) holds. A background
26//! writer drains the channel in batched transactions.
27//!
28//! # The sink seam
29//!
30//! The M1 extension contract
31//! `extensions::analytics::AnalyticsSink` is the OSS/Ent seam (feature
32//! analysis 11.4): the embedded store implements it (over the richer
33//! per-request fields DW-043 added to `extensions::analytics::Event`).
34//! DW-121's raw-record firehose is NOT an implementation of this
35//! contract — it streams the completion-time access record (which
36//! carries `request_id` and the redacted path the extension event
37//! deliberately omits) through its own sink seam in `events::stream`.
38//! The federated analytics pipeline (DW-095, SCALE-07 #186) IS an
39//! implementation of this contract: `cp_dp::analytics::FederatedAnalyticsSink`
40//! batches events on the edge and ships them to the controller over
41//! gRPC; the controller's `EmbeddedCollector` forwards them to the
42//! aggregate `EmbeddedAnalytics` store, tagging each event with the
43//! originating `edge_id` for fleet-wide query filtering.
44//!
45//! # Custom dimensions
46//!
47//! Config-declared dimensions extracted from request HEADERS at
48//! completion-time capture (e.g. `x-plan` -> dimension `plan`): they
49//! ride the [`AccessRecord`] into
50//! raw rows (a JSON object column) and aggregate into their own
51//! narrow rollup table. They are analytics-only — deliberately NOT
52//! added to the access log, whose field list is redacted by
53//! construction and stays that way.

155/// The custom-dimensions JSON column for a (name, value) pair list.
156fn dims_json(pairs: &[(String, String)]) -> String {
157    if pairs.is_empty() {
158        return "{}".to_string();
159    }
160    let map: serde_json::Map<String, serde_json::Value> = pairs
161        .iter()
162        .map(|(k, v)| (k.clone(), serde_json::Value::String(v.clone())))
163        .collect();
164    serde_json::Value::Object(map).to_string()
165}

504/// Live in-process sketches (DW-092): sub-second-freshness per-route
505/// rolling window with counts, errors, and capped latency samples.
506/// Maintained inside [`EmbeddedAnalytics`] and updated synchronously
507/// on every `record()` call — never blocks the dataplane (atomics for
508/// counts, a short mutex over a capped vector for samples). When the
509/// window expires, the completed window's aggregates are handed to
510/// the insights engine (if attached) for forecasting and anomaly
511/// detection.
512pub struct LiveSketches {
513    /// Per-route current-window data, behind a RwLock so the snapshot
514    /// read path (admin) does not block the record write path
515    /// (dataplane).
516    routes: RwLock<HashMap<String, RouteSketch>>,
517    /// The current window's start time (wall-clock ms). Atomic so the
518    /// rotation check on the hot path is lock-free.
519    window_start_ms: AtomicU64,
520    /// The window size in milliseconds (= the freshness target).
521    window_size_ms: u64,
522    /// The insights engine fed on window rotation (None when insights
523    /// are disabled).
524    insights: RwLock<Option<Arc<insights::InsightsEngine>>>,
525}
526
527impl LiveSketches {
528    /// Construct a new live-sketches store with the given window size.
529    pub fn new(window_size_ms: u64) -> Arc<Self> {
530        Arc::new(LiveSketches {
531            routes: RwLock::new(HashMap::new()),
532            window_start_ms: AtomicU64::new(0),
533            window_size_ms,
534            insights: RwLock::new(None),
535        })
536    }
537
538    /// Attach the insights engine so window rotations feed it. Called
539    /// once during [`EmbeddedAnalytics::open`] wiring.
540    pub fn set_insights(&self, engine: Arc<insights::InsightsEngine>) {
541        *self.insights.write().unwrap() = Some(engine);
542    }
543
544    /// The current window's start time (wall-clock ms).
545    pub fn window_start(&self) -> u64 {
546        self.window_start_ms.load(Ordering::Relaxed)
547    }
548
549    /// Record one request completion into the current window. Rotates
550    /// the window (handing the completed aggregates to the insights
551    /// engine) when `now_ms` has passed the window boundary.
552    pub fn record(&self, route: &str, status: u16, duration_ms: f64, now_ms: u64) {
553        // Rotate the window if it has expired. The rotation is under
554        // the write lock so a concurrent snapshot sees a consistent
555        // window boundary.
556        let start = self.window_start_ms.load(Ordering::Relaxed);
557        if start == 0 || now_ms >= start + self.window_size_ms {
558            self.rotate(now_ms);
559        }
560        // Look up or create the route sketch under a short write lock.
561        // The map is keyed by route name; a missing entry is inserted.
562        let map = self.routes.read().unwrap();
563        if let Some(sketch) = map.get(route) {
564            sketch.requests.fetch_add(1, Ordering::Relaxed);
565            if status >= 500 {
566                sketch.errors.fetch_add(1, Ordering::Relaxed);
567            }
568            sketch.latency.lock().unwrap().add(duration_ms);
569            return;
570        }
571        drop(map);
572        let mut map = self.routes.write().unwrap();
573        let sketch = map
574            .entry(route.to_string())
575            .or_insert_with(RouteSketch::new);
576        sketch.requests.fetch_add(1, Ordering::Relaxed);
577        if status >= 500 {
578            sketch.errors.fetch_add(1, Ordering::Relaxed);
579        }
580        sketch.latency.lock().unwrap().add(duration_ms);
581    }
582
583    /// Rotate the window: snapshot the current aggregates, hand them
584    /// to the insights engine (if attached), then reset the per-route
585    /// sketches for the new window. The new window starts at `now_ms`
586    /// (aligned to the rotation, not the expiry — a missed rotation
587    /// catches up to the present).
588    fn rotate(&self, now_ms: u64) {
589        let mut map = self.routes.write().unwrap();
590        let old_start = self.window_start_ms.swap(now_ms, Ordering::Relaxed);
591        // Feed the completed window to the insights engine (if any).
592        if old_start != 0 {
593            let mut total_requests: u64 = 0;
594            let mut total_errors: u64 = 0;
595            let mut latency_sum: f64 = 0.0;
596            let mut latency_count: u64 = 0;
597            for sketch in map.values() {
598                let req = sketch.requests.load(Ordering::Relaxed);
599                let err = sketch.errors.load(Ordering::Relaxed);
600                total_requests += req;
601                total_errors += err;
602                let mean = sketch.latency.lock().unwrap().mean();
603                latency_sum += mean * req as f64;
604                latency_count += req;
605            }
606            let avg_latency = if latency_count > 0 {
607                latency_sum / latency_count as f64
608            } else {
609                0.0
610            };
611            let window = insights::BaselineWindow {
612                ts_ms: old_start as i64,
613                requests: total_requests,
614                errors: total_errors,
615                avg_latency_ms: avg_latency,
616            };
617            if let Some(engine) = self.insights.read().unwrap().as_ref() {
618                engine.observe(window);
619            }
620        }
621        // Reset the per-route sketches for the new window. Clearing
622        // the map is simplest and avoids retaining stale routes; the
623        // first record of the new window re-creates each route's
624        // sketch.
625        map.clear();
626    }
627
628    /// Snapshot the current window's per-route aggregates. Computes
629    /// p50/p95/p99 from the retained latency samples (sort-and-pick,
630    /// O(n log n) with n capped at 1000). This is an admin-path
631    /// operation, never on the request hot path.
632    pub fn snapshot(&self, now_ms: u64) -> LiveSnapshot {
633        // Rotate if the window has expired so a snapshot reflects a
634        // current (non-stale) window.
635        let start = self.window_start_ms.load(Ordering::Relaxed);
636        if start == 0 || now_ms >= start + self.window_size_ms {
637            self.rotate(now_ms);
638        }
639        let start = self.window_start_ms.load(Ordering::Relaxed);
640        let map = self.routes.read().unwrap();
641        let mut routes: Vec<LiveRouteSnapshot> = map
642            .iter()
643            .map(|(route, sketch)| {
644                let requests = sketch.requests.load(Ordering::Relaxed);
645                let errors = sketch.errors.load(Ordering::Relaxed);
646                let (p50, p95, p99, avg) = {
647                    let sketch = sketch.latency.lock().unwrap();
648                    (
649                        sketch.quantile(0.50),
650                        sketch.quantile(0.95),
651                        sketch.quantile(0.99),
652                        sketch.mean(),
653                    )
654                };
655                LiveRouteSnapshot {
656                    route: route.clone(),
657                    requests,
658                    errors,
659                    error_rate: if requests == 0 {
660                        0.0
661                    } else {
662                        errors as f64 / requests as f64
663                    },
664                    p50_ms: p50,
665                    p95_ms: p95,
666                    p99_ms: p99,
667                    avg_ms: avg,
668                }
669            })
670            .collect();
671        routes.sort_by(|a, b| a.route.cmp(&b.route));
672        LiveSnapshot {
673            window_start_ms: start,
674            window_end_ms: start + self.window_size_ms,
675            routes,
676        }
677    }

1792    /// One rollup + retention pass (cursor-guarded; safe to run at any
1793    /// time, from the ticker or shutdown).
1794    fn maintain(&self) {
1795        let conn = self.conn.lock().unwrap();
1796        let now = now_ms();
1797        // SCALE-09 (#188): rotate the raw table into a daily partition
1798        // if it has rows from a previous day. This bounds the raw
1799        // table to the current day and makes retention O(1) (DROP
1800        // TABLE) for old partitions.
1801        if let Err(e) = partition::maybe_rotate(&conn, now) {
1802            tracing::warn!(
1803                code = "analytics_partition_rotate_failed",
1804                "raw partition rotation failed: {e}"
1805            );
1806        }
1807        let mut rolled = 0usize;
1808        match rollup::roll_raw_to_1m(&conn, now, ROLLUP_GRACE_MS) {
1809            Ok(n) => rolled = n,
1810            Err(e) => tracing::warn!(
1811                code = "analytics_rollup_failed",
1812                "raw->1m rollup failed: {e}"
1813            ),
1814        }
1815        for stage in 0..rollup::GRANULARITIES_MS.len() - 1 {
1816            if let Err(e) = rollup::roll_cascade(&conn, stage, now, ROLLUP_GRACE_MS) {
1817                tracing::warn!(
1818                    code = "analytics_cascade_failed",
1819                    "cascade stage {stage} failed: {e}"
1820                );
1821            }
1822        }
1823        if let Err(e) = rollup::sweep_retention(
1824            &conn,
1825            self.retention_ms[0],
1826            self.retention_ms[1..].try_into().unwrap(),
1827            now,
1828            256,
1829        ) {
1830            tracing::warn!(
1831                code = "analytics_retention_failed",
1832                "retention sweep failed: {e}"
1833            );
1834        } else if rolled > 0 {
1835            tracing::debug!(
1836                code = "analytics_rollup",
1837                windows = rolled,
1838                "rollup pass complete"
1839            );
1840        }
1841        // SCALE-09 (#188): drop expired daily partition tables (O(1)
1842        // DROP TABLE vs O(n) DELETE rows).
1843        if let Err(e) = partition::drop_expired_partitions(&conn, self.retention_ms[0], now) {
1844            tracing::warn!(
1845                code = "analytics_partition_drop_failed",
1846                "expired partition drop failed: {e}"
1847            );
1848        }
1849        // DW-081: prompt log retention sweep. Records older than the
1850        // configured retention window are deleted. A retention of 0
1851        // means no sweep (logging is off or unconfigured).
1852        let retention_ms = self.prompt_log_retention_ms.load(Ordering::Relaxed);
1853        if retention_ms > 0 {
1854            let cutoff = now.saturating_sub(retention_ms);
1855            if let Err(e) = conn.execute("DELETE FROM ai_prompt_logs WHERE ts_ms < ?1", [cutoff]) {

1857                    code = "analytics_prompt_log_retention_failed",
1858                    "ai_prompt_logs retention sweep failed: {e}"
1859                );
1860            }
1861        }

2026    /// Offer one raw record to the writer channel; drop-and-count on a
2027    /// full channel (the never-block policy shared by the hot path and
2028    /// the extension-seam impl below). REL-10 (#221): when sampled
2029    /// degradation is configured, probabilistically drops records when
2030    /// the channel fill ratio is above `high_watermark` to avoid a
2031    /// hard drop cliff.
2032    fn offer(&self, raw: RawRecord) {
2033        // REL-10 (#221): sampled degradation. Check the channel fill
2034        // ratio and probabilistically drop records above the high
2035        // watermark. Hysteresis: once sampling is active, it stays
2036        // active until the fill ratio drops below `low_watermark`.
2037        if let Some(cfg) = self.sampled_degradation {
2038            let in_flight = self.in_flight.load(Ordering::Relaxed) as f64;
2039            let fill_ratio = in_flight / CHANNEL_CAP as f64;
2040            let mut active = self.sampling_active.load(Ordering::Relaxed);
2041            // Update hysteresis state.
2042            if !active && fill_ratio > cfg.high_watermark {
2043                let _ = self.sampling_active.compare_exchange(
2044                    false,
2045                    true,
2046                    Ordering::Relaxed,
2047                    Ordering::Relaxed,
2048                );
2049                active = true;
2050            } else if active && fill_ratio < cfg.low_watermark {
2051                let _ = self.sampling_active.compare_exchange(
2052                    true,
2053                    false,
2054                    Ordering::Relaxed,
2055                    Ordering::Relaxed,
2056                );
2057                active = false;
2058            }
2059            if active {
2060                // Sample: keep only `keep_rate` fraction of records.
2061                // Use a fast thread-local PRNG to avoid lock contention.
2062                use std::cell::Cell;
2063                thread_local! {
2064                    static SAMPLE_RNG: Cell<u64> = const { Cell::new(0x9E3779B97F4A7C15) };
2065                }
2066                let keep = SAMPLE_RNG.with(|rng| {
2067                    let s = rng.get();
2068                    // xorshift64 (never returns 0 after the first step).
2069                    let mut x = s.wrapping_add(0x9E3779B97F4A7C15) | 1;
2070                    x ^= x << 13;
2071                    x ^= x >> 7;
2072                    x ^= x << 17;
2073                    rng.set(x);
2074                    x
2075                });
2076                // Map to [0, 1) and compare against keep_rate.
2077                let unit = (keep >> 11) as f64 / (1u64 << 53) as f64;
2078                if unit >= cfg.keep_rate {
2079                    // Drop this record (sampled).
2080                    let n = self.sampled_dropped.fetch_add(1, Ordering::Relaxed);
2081                    if n.is_multiple_of(4096) {
2082                        tracing::warn!(
2083                            code = "analytics_sampled_drop",
2084                            total_sampled_dropped = n + 1,
2085                            fill_ratio,
2086                            keep_rate = cfg.keep_rate,
2087                            "analytics channel above high watermark; sampling records \
2088                             (never blocking the dataplane)"
2089                        );
2090                    }
2091                    return;
2092                }
2093            }
2094        }
2095        if self.tx.try_send(raw).is_err() {
2096            let n = self.dropped.fetch_add(1, Ordering::Relaxed);
2097            if n.is_multiple_of(4096) {
2098                tracing::warn!(
2099                    code = "analytics_channel_full",
2100                    total_dropped = n + 1,
2101                    "analytics channel full; dropping records (never blocking the dataplane)"
2102                );
2103            }
2104        } else {
2105            // Track in-flight for the fill-ratio check.
2106            self.in_flight.fetch_add(1, Ordering::Relaxed);
2107        }
2108    }
