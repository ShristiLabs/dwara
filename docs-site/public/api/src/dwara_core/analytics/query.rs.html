2//! reports, and the structured query endpoint's translator.
3//!
4//! Every query reads ONLY the rollup tables (the dashboard contract is
5//! "week of traffic under 100 ms" — that is rollup-shaped work, not
6//! raw-scan work). All dynamic values bind as SQL parameters; the ONLY
7//! strings interpolated into SQL are schema identifiers chosen from
8//! closed sets ([`DIM_COLUMNS`]) — the structured query endpoint never

29/// One point of a dashboard series.
30#[derive(Debug, Serialize)]
31pub struct SeriesPoint {
32    /// Window start (ms since epoch).

34    pub requests: i64,
35    pub errors: i64,
36    pub error_rate: f64,

41    pub p95_ms: f64,
42    pub p99_ms: f64,
43    /// The group-by key values, when the series is grouped (parallel

49/// Equality filters on dimension columns (dashboard drill-down).
50#[derive(Debug, Default, Clone)]
51pub struct Filters {
52    pub listener: Option<String>,
53    pub route: Option<String>,
54    pub upstream: Option<String>,
55    pub consumer: Option<String>,
56    pub method: Option<String>,
57    pub status_class: Option<String>,
58}
59
60impl Filters {
61    /// WHERE fragments (leading ` AND ...` each) plus their bind
62    /// values, in the order the fragments appear.
63    fn clauses(&self) -> (String, Vec<Box<dyn rusqlite::ToSql>>) {
64        let mut sql = String::new();
65        let mut params: Vec<Box<dyn rusqlite::ToSql>> = Vec::new();
66        for (col, val) in [
67            ("listener", &self.listener),
68            ("route", &self.route),
69            ("upstream", &self.upstream),
70            ("consumer", &self.consumer),
71            ("method", &self.method),
72            ("status_class", &self.status_class),
73        ] {
74            if let Some(v) = val {
75                sql.push_str(&format!(" AND {col} = ?"));
76                params.push(Box::new(v.clone()));
77            }
78        }
79        (sql, params)
80    }
81}

83/// Shared aggregate read state: 5 metric sums plus 13 bucket sums.
84struct Agg {
85    requests: i64,
86    errors: i64,
87    rate_limited: i64,
88    shed: i64,
89    duration_sum_ms: f64,
90    buckets: Vec<i64>,
91}
92
93/// Shared aggregate read: 5 metric sums then 13 bucket sums starting
94/// at column `start`. NULL-tolerant: an UNGROUPED aggregate over an
95/// empty range still yields one all-NULL row (there is no GROUP BY to
96/// eliminate it), and the documented totals shape is zeros, not a
97/// type error (found by DW-120's totals statement; grouped reads are
98/// never NULL — a group exists only where rows do).
99fn read_agg(row: &rusqlite::Row<'_>, start: usize) -> rusqlite::Result<Agg> {
100    let mut i = start;
101    let requests = row.get::<_, Option<i64>>(i)?.unwrap_or(0);
102    i += 1;
103    let errors = row.get::<_, Option<i64>>(i)?.unwrap_or(0);
104    i += 1;
105    let rate_limited = row.get::<_, Option<i64>>(i)?.unwrap_or(0);
106    i += 1;
107    let shed = row.get::<_, Option<i64>>(i)?.unwrap_or(0);
108    i += 1;
109    let duration_sum_ms = row.get::<_, Option<f64>>(i)?.unwrap_or(0.0);
110    i += 1;
111    let mut buckets = Vec::with_capacity(BUCKET_COLS);
112    for _ in 0..BUCKET_COLS {
113        buckets.push(row.get::<_, Option<i64>>(i)?.unwrap_or(0));
114        i += 1;
115    }
116    Ok(Agg {
117        requests,
118        errors,
119        rate_limited,
120        shed,
121        duration_sum_ms,
122        buckets,
123    })
124}
125
126fn metrics_of(a: &Agg) -> (f64, f64, f64, f64, f64, f64, f64) {
127    let n = a.requests.max(0);
128    (
129        if n > 0 {
130            a.errors as f64 / n as f64
131        } else {
132            0.0
133        },
134        a.rate_limited as f64,
135        a.shed as f64,
136        if n > 0 {
137            a.duration_sum_ms / n as f64
138        } else {
139            0.0
140        },
141        percentile(&a.buckets, 0.50),
142        percentile(&a.buckets, 0.95),
143        percentile(&a.buckets, 0.99),
144    )
145}

147/// Dashboard series: per-window metric points at a granularity
148/// (0=1m..3=1d), optionally split by ONE dimension and filtered — the
149/// drill-down axes of the dashboard data API.
150pub fn dashboard(
151    conn: &Connection,
152    from_ms: i64,
153    to_ms: i64,
154    gran: usize,
155    group_by: Option<&str>,
156    filters: &Filters,
157) -> rusqlite::Result<Vec<SeriesPoint>> {
158    let group_col = group_by.and_then(|g| DIM_COLUMNS.iter().find(|c| **c == g).copied());
159    let key_sel = group_col.map(|c| format!("{c}, ")).unwrap_or_default();
160    let group_key = group_col.unwrap_or("window_start");
161    let (filter_sql, bind) = filters.clauses();
162    let sql = format!(
163        "SELECT window_start, {key_sel}
164                SUM(requests), SUM(errors), SUM(rate_limited), SUM(shed),
165                SUM(duration_sum_ms),
166                SUM(b0), SUM(b1), SUM(b2), SUM(b3), SUM(b4), SUM(b5),
167                SUM(b6), SUM(b7), SUM(b8), SUM(b9), SUM(b10), SUM(b11),
168                SUM(b12)
169         FROM rollup_fixed
170         WHERE gran = ? AND window_start >= ? AND window_start < ?{filter_sql}
171         GROUP BY window_start, {group_key}
172         ORDER BY window_start ASC"
173    );
174    let mut all: Vec<Box<dyn rusqlite::ToSql>> =
175        vec![Box::new(gran as i64), Box::new(from_ms), Box::new(to_ms)];
176    all.extend(bind);
177    let mut stmt = conn.prepare(&sql)?;
178    let mut rows = stmt.query(rusqlite::params_from_iter(all.iter().map(|p| p.as_ref())))?;
179    let key_len = usize::from(group_col.is_some());
180    let mut out = Vec::new();
181    while let Some(row) = rows.next()? {
182        let window_start: i64 = row.get(0)?;
183        let key = if key_len == 1 {
184            Some(row.get::<_, String>(1)?)
185        } else {
186            None
187        };
188        let agg = read_agg(row, 1 + key_len)?;
189        let (error_rate, rate_limited, shed, avg_ms, p50, p95, p99) = metrics_of(&agg);
190        out.push(SeriesPoint {
191            window_start,
192            requests: agg.requests,
193            errors: agg.errors,
194            error_rate,
195            rate_limited: rate_limited as i64,
196            shed: shed as i64,
197            avg_ms,
198            p50_ms: p50,
199            p95_ms: p95,
200            p99_ms: p99,
201            key,
202        });
203    }
204    Ok(out)
205}

207/// Top-N report kinds (the five frozen in the feature analysis).
208#[derive(Debug, Clone, Copy, PartialEq, Eq)]
209pub enum TopKind {
210    Consumers,
211    Routes,
212    Slowest,
213    ErrorProne,
214    RateLimited,
215}
216
217impl TopKind {
218    /// Parse the endpoint's `kind` parameter. The closed set.
219    pub fn parse(s: &str) -> Option<Self> {
220        match s {
221            "consumers" => Some(TopKind::Consumers),
222            "routes" => Some(TopKind::Routes),
223            "slowest" => Some(TopKind::Slowest),
224            "error_prone" => Some(TopKind::ErrorProne),
225            "rate_limited" => Some(TopKind::RateLimited),
226            _ => None,
227        }
228    }
229    fn as_str(&self) -> &'static str {
230        match self {
231            TopKind::Consumers => "consumers",
232            TopKind::Routes => "routes",
233            TopKind::Slowest => "slowest",
234            TopKind::ErrorProne => "error_prone",
235            TopKind::RateLimited => "rate_limited",
236        }
237    }
238}

253/// Top-N over a time range, summed across the FINEST granularity's
254/// windows (summing finer windows loses nothing — the whole schema is
255/// additive).
256pub fn top(
257    conn: &Connection,
258    kind: TopKind,
259    from_ms: i64,
260    to_ms: i64,
261    limit: usize,
262) -> rusqlite::Result<Vec<TopEntry>> {
263    let group = match kind {
264        TopKind::Consumers | TopKind::RateLimited => "consumer",
265        TopKind::Routes | TopKind::Slowest | TopKind::ErrorProne => "route",
266    };
267    let order = match kind {
268        TopKind::Consumers | TopKind::Routes => "SUM(requests) DESC",
269        TopKind::Slowest => "avg_ms DESC",
270        TopKind::ErrorProne => "err_rate DESC, SUM(requests) DESC",
271        TopKind::RateLimited => "SUM(rate_limited) DESC",
272    };
273    let sql = format!(
274        "SELECT {group} AS name,
275                SUM(requests), SUM(errors), SUM(rate_limited), SUM(shed),
276                SUM(duration_sum_ms),
277                SUM(b0), SUM(b1), SUM(b2), SUM(b3), SUM(b4), SUM(b5),
278                SUM(b6), SUM(b7), SUM(b8), SUM(b9), SUM(b10), SUM(b11),
279                SUM(b12),
280                (SUM(errors) * 1.0 / MAX(SUM(requests), 1)) AS err_rate,
281                (SUM(duration_sum_ms) / MAX(SUM(requests), 1)) AS avg_ms
282         FROM rollup_fixed
283         WHERE gran = 0 AND window_start >= ?1 AND window_start < ?2
284         GROUP BY {group}
285         ORDER BY {order}
286         LIMIT ?3"
287    );
288    let mut stmt = conn.prepare(&sql)?;
289    let mut rows = stmt.query(rusqlite::params![from_ms, to_ms, limit as i64])?;
290    let mut out = Vec::new();
291    while let Some(row) = rows.next()? {
292        let name: String = row.get(0)?;
293        let agg = read_agg(row, 1)?;
294        let requests = agg.requests.max(0);
295        let error_rate = if requests > 0 {
296            agg.errors as f64 / requests as f64
297        } else {
298            0.0
299        };
300        out.push(TopEntry {
301            kind: kind.as_str(),
302            name,
303            requests,
304            errors: agg.errors,
305            error_rate,
306            avg_ms: if requests > 0 {
307                agg.duration_sum_ms / requests as f64
308            } else {
309                0.0
310            },
311            p95_ms: percentile(&agg.buckets, 0.95),
312            rate_limited: agg.rate_limited,
313        });
314    }
315    Ok(out)
316}

318/// The structured query endpoint's request body (DW-043): a closed
319/// grammar translated to SQL — never SQL text from the caller.
320///
321/// SCALE-10 (#190): `dim_group_by` opens ad-hoc grouping over a
322/// captured custom dimension (bounded cardinality). When set, the
323/// query reads from `rollup_dim` instead of `rollup_fixed`, grouping
324/// by the dimension's values.
325#[derive(Debug, Deserialize)]
326pub struct StructuredQuery {
327    pub from_ms: i64,
328    pub to_ms: i64,
329    /// Granularity index 0..=3 (1m/5m/1h/1d).
330    pub gran: usize,
331    /// Dimension columns to group by (closed set; empty = one totals
332    /// row).
333    #[serde(default)]
334    pub group_by: Vec<String>,
335    /// SCALE-12 (#190): custom dimension name to group by. When set,
336    /// the query reads from `rollup_dim` grouped by the dimension's
337    /// values. Bounded to the configured custom dimensions'
338    /// cardinality (the rollup_dim table's primary key).
339    #[serde(default, skip_serializing_if = "Option::is_none")]
340    pub dim_group_by: Option<String>,
341    /// Equality filters on dimension columns.
342    #[serde(default)]
343    pub filters: FiltersBody,
344    /// SCALE-12 (#190): equality filters on custom dimensions. The
345    /// key is the dimension name, the value is the required value.
346    /// Applied as a semi-join against `rollup_dim`.
347    #[serde(default)]
348    pub dim_filters: std::collections::BTreeMap<String, String>,
349    /// Maximum returned rows (default 1000, hard cap 10 000).
350    #[serde(default)]
351    pub limit: Option<usize>,
352}

354/// The filter map of a [`StructuredQuery`]: column -> required value.
355#[derive(Debug, Default, Deserialize)]
356pub struct FiltersBody {
357    pub listener: Option<String>,
358    pub route: Option<String>,
359    pub upstream: Option<String>,
360    pub consumer: Option<String>,
361    pub method: Option<String>,
362    pub status_class: Option<String>,
363}

365/// Why a structured query was rejected before SQL was built.
366#[derive(Debug)]
367pub enum QueryError {
368    UnknownGroupBy(String),
369    BadGranularity(usize),
370    BadRange,
371    /// SCALE-12 (#190): the dim_group_by name is empty or exceeds the
372    /// maximum length (64 chars). Dimension names are config-bounded;
373    /// this is a safety cap against abuse.
374    BadDimGroupBy(String),
375}
376
377impl std::fmt::Display for QueryError {
378    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
379        match self {
380            QueryError::UnknownGroupBy(g) => write!(
381                f,
382                "unknown group_by '{g}' (one of: {})",
383                DIM_COLUMNS.join(", ")
384            ),
385            QueryError::BadGranularity(g) => {
386                write!(f, "gran must be 0..=3 (1m/5m/1h/1d), got {g}")
387            }
388            QueryError::BadRange => write!(f, "from_ms must be < to_ms"),
389            QueryError::BadDimGroupBy(g) => {
390                write!(f, "dim_group_by '{g}' is empty or exceeds 64 characters")
391            }
392        }
393    }
394}
395
396impl std::error::Error for QueryError {}
397
398/// Maximum length of a custom dimension name for dim_group_by.
399const MAX_DIM_NAME_LEN: usize = 64;
400
401impl StructuredQuery {
402    /// Reject anything outside the closed grammar.
403    pub fn validate(&self) -> Result<(), QueryError> {
404        if self.gran >= GRANULARITIES_MS.len() {
405            return Err(QueryError::BadGranularity(self.gran));
406        }
407        if self.from_ms >= self.to_ms {
408            return Err(QueryError::BadRange);
409        }
410        for g in &self.group_by {
411            if !DIM_COLUMNS.contains(&g.as_str()) {
412                return Err(QueryError::UnknownGroupBy(g.clone()));
413            }
414        }
415        if let Some(d) = &self.dim_group_by {
416            if d.is_empty() || d.len() > MAX_DIM_NAME_LEN {
417                return Err(QueryError::BadDimGroupBy(d.clone()));
418            }
419        }
420        for k in self.dim_filters.keys() {
421            if k.is_empty() || k.len() > MAX_DIM_NAME_LEN {
422                return Err(QueryError::BadDimGroupBy(k.clone()));
423            }
424        }
425        Ok(())
426    }
427}

445/// Execute a validated structured query. Range-aggregate shape: the
446/// window column is NOT grouped (per-window series is the dashboard
447/// endpoint's job); rows order by request volume descending.
448///
449/// SCALE-12 (#190): when `dim_group_by` is set, the query reads from
450/// `rollup_dim` instead of `rollup_fixed`, grouping by the custom
451/// dimension's values. When `dim_filters` are set, they are applied
452/// as semi-join subqueries against `rollup_dim`.
453pub fn structured(conn: &Connection, q: &StructuredQuery) -> rusqlite::Result<Vec<QueryRow>> {
454    // SCALE-12 (#190): custom dimension grouping path.
455    if let Some(dim_name) = &q.dim_group_by {
456        return structured_dim(conn, q, dim_name);
457    }
458    let key_len = q.group_by.len();
459    let key_sel = if key_len == 0 {
460        String::new()
461    } else {
462        format!("{}, ", q.group_by.join(", "))
463    };
464    let filters = Filters {
465        listener: q.filters.listener.clone(),
466        route: q.filters.route.clone(),
467        upstream: q.filters.upstream.clone(),
468        consumer: q.filters.consumer.clone(),
469        method: q.filters.method.clone(),
470        status_class: q.filters.status_class.clone(),
471    };
472    let (filter_sql, bind) = filters.clauses();
473    // SCALE-12 (#190): dim_filters semi-join against rollup_dim.
474    let (dim_filter_sql, dim_bind) = dim_filter_clauses(&q.dim_filters);
475    let sql = format!(
476        "SELECT {key_sel}
477                SUM(requests), SUM(errors), SUM(rate_limited), SUM(shed),
478                SUM(duration_sum_ms),
479                SUM(b0), SUM(b1), SUM(b2), SUM(b3), SUM(b4), SUM(b5),
480                SUM(b6), SUM(b7), SUM(b8), SUM(b9), SUM(b10), SUM(b11),
481                SUM(b12)
482         FROM rollup_fixed
483         WHERE gran = ? AND window_start >= ? AND window_start < ?{filter_sql}{dim_filter_sql}
484         {group_all}
485         ORDER BY SUM(requests) DESC
486         LIMIT ?",
487        group_all = if key_len == 0 {
488            // No group keys: a plain aggregate over the range — ONE
489            // totals row (the documented shape). A literal `GROUP BY
490            // 1` would resolve positionally to the leading SUM(...) —
491            // an invalid aggregate-in-GROUP-BY (found by DW-120's
492            // totals statement).
493            String::new()
494        } else {
495            format!("GROUP BY {}", q.group_by.join(", "))
496        }
497    );
498    let limit = q.limit.unwrap_or(1000).min(10_000);
499    let mut all: Vec<Box<dyn rusqlite::ToSql>> = vec![
500        Box::new(q.gran as i64),
501        Box::new(q.from_ms),
502        Box::new(q.to_ms),
503    ];
504    all.extend(bind);
505    all.extend(dim_bind);
506    all.push(Box::new(limit as i64));
507    let mut stmt = conn.prepare(&sql)?;
508    let mut rows = stmt.query(rusqlite::params_from_iter(all.iter().map(|p| p.as_ref())))?;
509    let mut out = Vec::new();
510    while let Some(row) = rows.next()? {
511        let mut key = Vec::with_capacity(key_len);
512        for i in 0..key_len {
513            key.push(row.get::<_, String>(i)?);
514        }
515        let agg = read_agg(row, key_len)?;
516        let (error_rate, rate_limited, shed, avg_ms, p50, p95, p99) = metrics_of(&agg);
517        out.push(QueryRow {
518            key,
519            requests: agg.requests,
520            errors: agg.errors,
521            error_rate,
522            rate_limited: rate_limited as i64,
523            shed: shed as i64,
524            avg_ms,
525            p50_ms: p50,
526            p95_ms: p95,
527            p99_ms: p99,
528        });
529    }
530    Ok(out)
531}

533/// SCALE-12 (#190): Build WHERE clauses for custom dimension filters.
534/// Each filter becomes a semi-join: the window_start must appear in
535/// the set of windows where the dimension equals the required value.
536/// Returns (SQL fragment, bind values).
537fn dim_filter_clauses(
538    dim_filters: &std::collections::BTreeMap<String, String>,
539) -> (String, Vec<Box<dyn rusqlite::ToSql>>) {
540    if dim_filters.is_empty() {
541        return (String::new(), Vec::new());
542    }
543    let mut sql = String::new();
544    let mut params: Vec<Box<dyn rusqlite::ToSql>> = Vec::new();
545    for (dim, value) in dim_filters {
546        // Semi-join: window_start must appear in the set of windows
547        // where this dimension has this value. Uses EXISTS for clarity.
548        sql.push_str(&format!(
549            " AND EXISTS (SELECT 1 FROM rollup_dim d{idx} \
550             WHERE d{idx}.gran = rollup_fixed.gran \
551             AND d{idx}.window_start = rollup_fixed.window_start \
552             AND d{idx}.dim = ? AND d{idx}.value = ?)",
553            idx = params.len() / 2
554        ));
555        params.push(Box::new(dim.clone()));
556        params.push(Box::new(value.clone()));
557    }
558    (sql, params)
559}

561/// SCALE-12 (#190): Execute a structured query grouped by a custom
562/// dimension. Reads from `rollup_dim` instead of `rollup_fixed`,
563/// grouping by the dimension's values. The `rollup_dim` table does
564/// not carry `rate_limited` or `shed` columns (custom dimensions
565/// aggregate every request carrying the dimension, not split by fixed
566/// dimensions), so those fields are zero in the result.
567fn structured_dim(
568    conn: &Connection,
569    q: &StructuredQuery,
570    dim_name: &str,
571) -> rusqlite::Result<Vec<QueryRow>> {
572    let limit = q.limit.unwrap_or(1000).min(10_000);
573    let (filter_sql, bind) = dim_filter_clauses(&q.dim_filters);
574    // The rollup_dim table does not have rate_limited or shed columns.
575    let sql = format!(
576        "SELECT value,
577                SUM(requests), SUM(errors), 0, 0,
578                SUM(duration_sum_ms),
579                SUM(b0), SUM(b1), SUM(b2), SUM(b3), SUM(b4), SUM(b5),
580                SUM(b6), SUM(b7), SUM(b8), SUM(b9), SUM(b10), SUM(b11),
581                SUM(b12)
582         FROM rollup_dim
583         WHERE gran = ? AND window_start >= ? AND window_start < ?
584         AND dim = ?{filter_sql}
585         GROUP BY value
586         ORDER BY SUM(requests) DESC
587         LIMIT ?"
588    );
589    let mut all: Vec<Box<dyn rusqlite::ToSql>> = vec![
590        Box::new(q.gran as i64),
591        Box::new(q.from_ms),
592        Box::new(q.to_ms),
593        Box::new(dim_name.to_string()),
594    ];
595    all.extend(bind);
596    all.push(Box::new(limit as i64));
597    let mut stmt = conn.prepare(&sql)?;
598    let mut rows = stmt.query(rusqlite::params_from_iter(all.iter().map(|p| p.as_ref())))?;
599    let mut out = Vec::new();
600    while let Some(row) = rows.next()? {
601        let key = vec![row.get::<_, String>(0)?];
602        let agg = read_agg(row, 1)?;
603        let (error_rate, _rate_limited, _shed, avg_ms, p50, p95, p99) = metrics_of(&agg);
604        out.push(QueryRow {
605            key,
606            requests: agg.requests,
607            errors: agg.errors,
608            error_rate,
609            rate_limited: 0,
610            shed: 0,
611            avg_ms,
612            p50_ms: p50,

701/// Execute a validated spend query (DW-079): aggregate the `ai_spend`
702/// table over a billing window, grouped by the requested dimensions.
703/// Rows order by cost descending (the billing-relevant axis). Reads
704/// the RAW `ai_spend` rows directly — spend is per-request, not
705/// rolled up (the volume is orders of magnitude below the access
706/// record path, and billing windows are short).
707pub fn spend_summary(conn: &Connection, q: &SpendQuery) -> rusqlite::Result<Vec<SpendRow>> {
708    let key_len = q.group_by.len();
709    let key_sel = if key_len == 0 {
710        String::new()
711    } else {
712        format!("{}, ", q.group_by.join(", "))
713    };
714    let group_all = if key_len == 0 {
715        String::new()
716    } else {
717        format!("GROUP BY {}", q.group_by.join(", "))
718    };
719    let sql = format!(
720        "SELECT {key_sel}
721                SUM(prompt_tokens), SUM(completion_tokens), SUM(total_tokens),
722                SUM(cost_micros), COUNT(*)
723         FROM ai_spend
724         WHERE ts_ms >= ? AND ts_ms < ?
725         {group_all}
726         ORDER BY SUM(cost_micros) DESC
727         LIMIT ?"
728    );
729    let limit = q.limit.unwrap_or(10_000).min(10_000);
730    let mut stmt = conn.prepare(&sql)?;
731    let mut rows = stmt.query(rusqlite::params![q.from_ms, q.to_ms, limit as i64])?;
732    let mut out = Vec::new();
733    while let Some(row) = rows.next()? {
734        let mut key = Vec::with_capacity(key_len);
735        for i in 0..key_len {
736            key.push(row.get::<_, String>(i)?);
737        }
738        let base = key_len;
739        out.push(SpendRow {
740            key,
741            prompt_tokens: row.get::<_, i64>(base)?,
742            completion_tokens: row.get::<_, i64>(base + 1)?,
743            total_tokens: row.get::<_, i64>(base + 2)?,
744            cost_micros: row.get::<_, i64>(base + 3)?,
745            request_count: row.get::<_, i64>(base + 4)?,
746        });
747    }
748    Ok(out)
749}

1234    }

1238    let mut rows = stmt.query(param_refs.as_slice())?;
1239    let mut out = Vec::new();
1240    while let Some(row) = rows.next()? {
