1//! The AI proxy action (DW-075): the dataplane half of the
2//! provider-adapter pack — extended by DW-076 with routing and
3//! failover.
4//!
5//! Runs when a route's action is `ai`: parse the client's OpenAI
6//! chat-completions body, resolve the model alias through the
7//! generation's [`AiRuntime`](crate::ai::AiRuntime), translate via the
8//! provider's [`ProviderAdapter`],
9//! place the call through the provider's UPSTREAM (pooling, TLS,
10//! timeouts, breaker, health — the standard machinery), and translate
11//! the response back. All failures answer in the OpenAI error shape so
12//! standard client SDKs parse them; the gateway's request id rides
13//! along inside `error.request_id`.
14//!
15//! # Routing and failover (DW-076)
16//!
17//! `AiRuntime::route` turns the alias into an ordered candidate list:
18//! `[primary, failover...]` for a chained alias, or exactly one
19//! deterministically-picked canary version for a split alias (see
20//! `ai::routing`). The action walks the candidates and advances on
21//! RETRYABLE outcomes — a 429, a 5xx, a transport error, or a
22//! dialect-specific translation rejection — and treats everything
23//! else as final. Failover is invisible to the client by
24//! construction: a provider response is read and translated WHOLE
25//! before any byte reaches the client, so a failing candidate emits
26//! nothing client-visible. Same-provider retries are deliberately not
27//! the chain's job — the provider's upstream breaker owns those; the
28//! chain only ever moves to a DIFFERENT provider/model pair. Usage and
29//! request metrics attribute to the provider and canary version that
30//! actually served; the access record follows the serving provider.
31//!
32//! Body bounds: an AI request must be read whole to translate it (the
33//! zero-buffering rule yields here by necessity, opt-in by choosing
34//! the action), and both directions are capped INCREMENTALLY — the
35//! read aborts as soon as the cap is crossed, never after buffering
36//! the whole body — at [`MAX_AI_REQUEST_BYTES`] inbound and
37//! [`MAX_AI_PROVIDER_RESPONSE_BYTES`] from the provider, so a hostile
38//! or runaway peer cannot turn the translation into an unbounded
39//! buffer.
40//!
41//! Streaming (DW-077): `stream: true` requests stream back to the
42//! client as `text/event-stream`, translated frame-by-frame with ZERO
43//! added buffering ([`AiStreamBody`] + [`StreamTranslator`]): each
44//! complete provider frame becomes client frames in the same poll.
45//! Token counts are provider-reported only (locked decision) and
46//! accumulate mid-stream into one terminal usage chunk; the gateway
47//! owns the `data: [DONE]` terminator. The failover chain applies
48//! until the streaming response is returned (the commit point); after
49//! the first forwarded frame a provider abort ends the stream cleanly
50//! with an error chunk — already-forwarded content stands.

69/// Inbound AI request body cap: 16 MiB, generous against long-context
70/// requests while bounding the translation buffer.
71pub const MAX_AI_REQUEST_BYTES: usize = 16 * 1024 * 1024;
72
73/// Provider response body cap: 32 MiB (non-streaming completions of
74/// extreme length). Provider ERROR bodies are capped tighter at
75/// [`MAX_AI_ERROR_BYTES`] — they are never legitimately large.
76pub const MAX_AI_PROVIDER_RESPONSE_BYTES: usize = 32 * 1024 * 1024;
77
78/// Provider error body cap: 1 MiB.
79pub const MAX_AI_ERROR_BYTES: usize = 1024 * 1024;

81/// AI-02: the provider path for each non-chat endpoint. The passthrough
82/// forwards the request body as-is to the provider's upstream at this
83/// path (no adapter translation). The path is OpenAI-compatible (the
84/// most common target); providers with different paths can override
85/// the path per-provider via `ai.providers[].path`.
86fn passthrough_path(endpoint: crate::config::AiEndpoint) -> &'static str {
87    match endpoint {
88        crate::config::AiEndpoint::Embeddings => "/v1/embeddings",
89        crate::config::AiEndpoint::Images => "/v1/images/generations",
90        crate::config::AiEndpoint::Audio => "/v1/audio/speech",
91        crate::config::AiEndpoint::Moderation => "/v1/moderations",
92        crate::config::AiEndpoint::Chat => "/v1/chat/completions",
93        // AI-04 (#194): Batch API uses /v1/batches for create/status.
94        // The result file endpoint (/v1/batches/{id}/output) is handled
95        // separately in serve_ai_batch_results.
96        crate::config::AiEndpoint::Batch => "/v1/batches",
97    }
98}

108/// AI-02: serve a non-chat AI endpoint (embeddings, images, audio,
109/// moderation) as a passthrough. The request body is forwarded to the
110/// provider's upstream as-is (no adapter translation), and the response
111/// is returned as-is. Model governance (allowlist check) still applies
112/// using the `model` field from the request body. Budget enforcement
113/// and semantic caching do NOT apply (they are chat-specific).
114#[allow(clippy::too_many_arguments)]
115async fn serve_ai_passthrough(
116    json_body: serde_json::Value,
117    route_name: &str,
118    gen: &Arc<Generation>,
119    dp: &Arc<DataPlane>,
120    rid: &str,
121    rec: &mut crate::observability::AccessRecord,
122    identity: Option<&crate::security::authn::Identity>,
123    endpoint: crate::config::AiEndpoint,
124) -> Response<ProxyBody> {
125    // Extract the model alias from the request body for governance and
126    // routing. The `model` field is present in all OpenAI endpoint
127    // request shapes (embeddings, images, audio, moderation). When
128    // absent, the request is forwarded without governance/routing (a
129    // passthrough to the first provider's upstream).
130    let model_alias = json_body
131        .get("model")
132        .and_then(serde_json::Value::as_str)
133        .unwrap_or("");
134
135    // DW-084: model governance pre-route check (same as chat). The
136    // requested model alias is checked against the consumer's binding
137    // team allowlists BEFORE routing.
138    let gateway = gen.snapshot.gateway();
139    let governance = dp.ai_governance();
140    if !governance.is_empty() {
141        let consumer = identity.map(|id| id.consumer_name.as_str());
142        let route_cfg = gateway.routes.iter().find(|r| r.name == route_name);
143        let route_policies: &[String] = route_cfg.map(|r| r.policies.as_slice()).unwrap_or(&[]);
144        let service_policies: &[String] = route_cfg
145            .map(|r| crate::ai::budget::service_policies_of(gateway, &r.service))
146            .unwrap_or(&[]);
147        let listener_policies = crate::ai::budget::listener_policies_of(gateway, &rec.listener);
148        let consumer_policies: &[String] = consumer
149            .map(|c| crate::ai::budget::consumer_policies_of(gateway, c))
150            .unwrap_or(&[]);
151        let verdict = governance.check(
152            consumer,
153            consumer_policies,
154            route_policies,
155            service_policies,
156            listener_policies,
157            &[],
158            model_alias,
159        );
160        if let crate::ai::governance::GovernanceVerdict::Deny { reason, .. } = &verdict {
161            dp.observability_arc().record_ai_governance_denied(reason);
162            return ai_error_response(
163                StatusCode::FORBIDDEN,
164                &format!("model '{}' is denied by policy: {reason}", model_alias),
165                "invalid_request_error",
166                Some("model_denied_by_policy"),
167                rid,
168            );
169        }
170    }
171
172    // Route the model alias to find the provider (same runtime as
173    // chat). When the alias is empty or not found, return 404.
174    let Some(runtime) = gen.ai() else {
175        return ai_error_response(
176            StatusCode::INTERNAL_SERVER_ERROR,
177            "no ai block configured",
178            "api_error",
179            Some("internal_error"),
180            rid,
181        );
182    };
183    let (candidates, _, _) = runtime
184        .route_with_policy_and_canary_index(model_alias, rid, "")
185        .await;
186    if candidates.is_empty() {
187        return ai_error_response(
188            StatusCode::NOT_FOUND,
189            &format!("the model '{}' does not exist", model_alias),
190            "invalid_request_error",
191            Some("model_not_found"),
192            rid,
193        );
194    }
195
196    // Forward to the first candidate (no failover for passthrough —
197    // the response is returned as-is, so retry logic would need to
198    // buffer the entire response, which defeats the passthrough
199    // design).
200    let target = &candidates[0];
201    let version = target.version.as_deref().unwrap_or("default");
202    let Some(provider) = runtime.provider(&target.provider) else {
203        return ai_error_response(
204            StatusCode::INTERNAL_SERVER_ERROR,
205            "no provider could serve the model",
206            "api_error",
207            Some("provider_unreachable"),
208            rid,
209        );
210    };
211    let Some(handle) = gen.registry().get(&provider.upstream) else {
212        return ai_error_response(
213            StatusCode::INTERNAL_SERVER_ERROR,
214            "no provider could serve the model",
215            "api_error",
216            Some("provider_unreachable"),
217            rid,
218        );
219    };
220    rec.upstream = Some(provider.upstream.clone());
221    rec.attempts = 1;
222
223    // Build the outbound request: the passthrough path + the original
224    // body + the provider's auth headers.
225    let path = resolve_path(provider, passthrough_path(endpoint));
226    let mut outbound = http::Request::builder()
227        .method(http::Method::POST)
228        .uri(path);
229    outbound = outbound.header("content-type", "application/json");
230    let auth_pairs: Vec<(String, String)> = if let Some(pool) = &provider.credential_pool {
231        match pool.pick(rid) {
232            Some(entry) => vec![(entry.header.clone(), entry.value.clone())],
233            None => provider.auth_headers.clone(),
234        }
235    } else {
236        provider.auth_headers.clone()
237    };
238    for (name, value) in &auth_pairs {
239        if let (Ok(n), Ok(v)) = (
240            hyper::header::HeaderName::from_bytes(name.as_bytes()),
241            hyper::header::HeaderValue::from_str(value),
242        ) {
243            outbound = outbound.header(n, v);
244        }
245    }
246    let body_bytes = serde_json::to_vec(&json_body).unwrap_or_default();
247    let outbound = match outbound.body(Full::new(Bytes::from(body_bytes))) {
248        Ok(r) => r,
249        Err(_) => {
250            return ai_error_response(
251                StatusCode::INTERNAL_SERVER_ERROR,
252                "failed to build provider request",
253                "api_error",
254                Some("internal_error"),
255                rid,
256            );
257        }
258    };
259
260    let obs = dp.observability_arc();
261    let started = std::time::Instant::now();
262    let upstream_resp = match handle.send(outbound).await {
263        Ok(r) => r,
264        Err(e) => {
265            obs.record_ai_request(&provider.name, route_name, "transport_error", version);
266            obs.record_ai_request_duration(
267                &provider.name,
268                route_name,
269                started.elapsed().as_secs_f64(),
270            );
271            tracing::warn!(
272                code = "ai_provider_unreachable",
273                request_id = %rid,
274                provider = %provider.name,
275                upstream = %provider.upstream,
276                "ai passthrough call failed: {e}"
277            );
278            return ai_error_response(
279                StatusCode::BAD_GATEWAY,
280                "no provider could serve the model",
281                "api_error",
282                Some("provider_unreachable"),
283                rid,
284            );
285        }
286    };
287
288    let status = upstream_resp.status();
289    let up_headers = upstream_resp.headers().clone();
290    let up_body = upstream_resp.into_body();
291    obs.record_ai_request(
292        &provider.name,
293        route_name,
294        if status.is_success() {
295            "success"
296        } else {
297            "error"
298        },
299        version,
300    );
301    obs.record_ai_request_duration(&provider.name, route_name, started.elapsed().as_secs_f64());
302
303    if !status.is_success() {
304        let err_bytes =
305            match bounded_collect(up_body, MAX_AI_ERROR_BYTES, "provider error", rid).await {
306                Ok(b) => b,
307                Err(resp) => return resp,
308            };
309        let err_json: serde_json::Value = serde_json::from_slice(&err_bytes).unwrap_or(
310            serde_json::json!({"error": {"message": String::from_utf8_lossy(&err_bytes).to_string()}}),
311        );
312        return response_with_json(status, &err_json, &up_headers);
313    }
314
315    // Success: return the response body as-is (no translation).
316    let ok_bytes =
317        match bounded_collect(up_body, MAX_AI_PROVIDER_RESPONSE_BYTES, "response", rid).await {
318            Ok(b) => b,
319            Err(resp) => return resp,
320        };
321    Response::builder()
322        .status(status)
323        .header("content-type", "application/json")
324        .body(ProxyBody::Full(Full::new(ok_bytes)))
325        .expect("passthrough response is valid")
326}

537/// Serve one `ai` route action (called from `dispatch_action` in the
538/// proxy module). `rec` is the request's access record: the provider's
539/// upstream name is attributed there so analytics and the access log
540/// see which provider served the call.
541///
542/// AI-02: the `endpoint` parameter selects which AI endpoint the route
543/// serves. `Chat` uses the full adapter translation pipeline (DW-075).
544/// The other endpoints (`Embeddings`, `Images`, `Audio`, `Moderation`)
545/// are proxied as a passthrough: the request body is forwarded to the
546/// provider's upstream as-is, and the response is returned as-is.
547/// Model governance (allowlist check) still applies using the `model`
548/// field from the request body.
549#[allow(clippy::too_many_arguments)]
550pub(super) async fn serve_ai<B>(
551    req: Request<B>,
552    route_name: &str,
553    gen: &Arc<Generation>,
554    dp: &Arc<DataPlane>,
555    rid: &str,
556    rec: &mut crate::observability::AccessRecord,
557    identity: Option<&crate::security::authn::Identity>,
558    listener_name: &str,
559    endpoint: crate::config::AiEndpoint,
560    dialect: crate::config::AiIngressDialect,
561) -> Response<ProxyBody>
562where
563    B: Body<Data = Bytes> + Send + 'static,
564    B::Error: Into<Box<dyn std::error::Error + Send + Sync>>,
565{
566    // AI token budget pre-check (DW-078): BEFORE the request body is
567    // even read — a holder whose window is exhausted never reaches a
568    // provider. The resolution mirrors the rate-limit chain (consumer
569    // > route > service > listener > global, most-specific budget
570    // governs); unbudgeted holders resolve no guard and skip.
571    let budget = {
572        let engine = dp.ai_budgets();
573        if engine.is_empty() {
574            None
575        } else {
576            let consumer = identity.map(|id| id.consumer_name.as_str());
577            let gateway = gen.snapshot.gateway();
578            let route_cfg = gateway.routes.iter().find(|r| r.name == route_name);
579            let route_policies: &[String] = route_cfg.map(|r| r.policies.as_slice()).unwrap_or(&[]);
580            let service_policies: &[String] = route_cfg
581                .map(|r| crate::ai::budget::service_policies_of(gateway, &r.service))
582                .unwrap_or(&[]);
583            let consumer_policies: &[String] = consumer
584                .map(|c| crate::ai::budget::consumer_policies_of(gateway, c))
585                .unwrap_or(&[]);
586            let listener_policies = crate::ai::budget::listener_policies_of(gateway, listener_name);
587            engine.resolve(
588                consumer,
589                consumer_policies,
590                route_policies,
591                service_policies,
592                listener_policies,

1179                0,

1185        }
1186        // Record the assignment to analytics (fire-and-forget).

1189                ts_ms: std::time::SystemTime::now()
1190                    .duration_since(std::time::UNIX_EPOCH)
1191                    .map(|d| d.as_millis() as i64)
1192                    .unwrap_or(0),

1195                variant: variant.name.clone(),

1209            Some("model_not_found"),

1214    // 5. Walk the candidates (DW-076 failover). Retryable outcomes —
1215    // a 429, a 5xx, a transport error, or a per-dialect translation
1216    // rejection — advance to the NEXT candidate; anything else is
1217    // final. Nothing reaches the client until a candidate SUCCEEDS
1218    // (the provider response is read whole before translation), so
1219    // failover is invisible to the client. Same-provider retries are
1220    // not the chain's job: the provider's upstream breaker owns those.
1221    let obs = dp.observability_arc();

1223    let mut last_transport_error = true;
1224    let mut fallback_response: Response<ProxyBody> = ai_error_response(
1225        StatusCode::BAD_GATEWAY,
1226        "no provider could serve the model",
1227        "api_error",
1228        Some("provider_unreachable"),
1229        rid,
1230    );

1233        let version = target.version.as_deref().unwrap_or("default");
1234        let Some(provider) = runtime.provider(&target.provider) else {
1235            tracing::error!(
1236                code = "ai_provider_missing",
1237                request_id = %rid,
1238                provider = %target.provider,
1239                "routed provider is not in the compiled table (validate-vs-build race)"
1240            );

1242            obs.record_ai_request_duration(
1243                &target.provider,
1244                route_name,
1245                ai_started.elapsed().as_secs_f64(),
1246            );

1251        // (re-recorded per candidate so the span reflects the provider
1252        // that actually served; on failover the last candidate wins).

1260            Err(e) => {

1274                    StatusCode::BAD_REQUEST,

1278                    rid,
1279                );

1287                code = "ai_provider_upstream_missing",
1288                request_id = %rid,
1289                provider = %provider.name,

1295                &provider.name,
1296                route_name,

1350                        "all provider API keys are rate-limited; retry later",

1369                hyper::header::HeaderName::from_bytes(name.as_bytes()),

1374                }

1387            Err(_) => {

1390                    &provider.name,
1391                    route_name,
1392                    ai_started.elapsed().as_secs_f64(),
1393                );
1394                continue;
1395            }
1396        };

1401        let attempt_started = std::time::Instant::now();

1407            Err(e) => {

1410                    &provider.name,
1411                    route_name,
1412                    ai_started.elapsed().as_secs_f64(),
1413                );
1414                tracing::warn!(
1415                    code = "ai_provider_unreachable",
1416                    request_id = %rid,
1417                    provider = %provider.name,
1418                    upstream = %provider.upstream,
1419                    "ai provider call failed: {e}"
1420                );
1421                last_transport_error = true;
1422                continue;
1423            }

1468                route_name,
1469                ai_started.elapsed().as_secs_f64(),

1504                }

1506                // none succeed, the LAST provider's answer is the one
1507                // the client sees (closest to the truth of the outage).

1509                    code = "ai_provider_failover",
1510                    request_id = %rid,
1511                    provider = %provider.name,
1512                    status = status.as_u16(),

1514                );
1515                fallback_response = resp;
1516                last_transport_error = false;
1517                continue;
1518            }
1519            // Other 4xx (bad request, bad key) is deterministic —
1520            // retrying another provider would only re-diagnose it.
1521            return resp;
1522        }

1526        // proactive cooldown.
1527        if let (Some(pool), Some(idx)) = (&provider.credential_pool, pool_entry_index) {
1528            let health = parse_rate_limit_headers(&up_parts.headers, status.as_u16());

1530        }

1548                    .snapshot
1549                    .gateway()

1579                            Some("ai_unknown_model_pricing"),
1580                            rid,
1581                        );
1582                    }

1588                            route = %route_name,
1589                            provider = %provider.name,
1590                            model = %target.provider_model,

1595                }
1596            }

1600            // stream's own duration is dwara_ai_stream_duration_seconds).
1601            obs.record_ai_request_duration(
1602                &provider.name,
1603                route_name,

1620                .map(|id| id.consumer_name.clone())

1623                .map(|id| consumer_type_str(id.consumer_type).to_string())

1689                Ok(b) => b,
1690                Err(resp) => {
1691                    // An over-cap or unreadable SUCCESS body is
1692                    // provider-specific (a runaway completion), exactly
1693                    // like a malformed one below: record it, stash the
1694                    // 502, and let the next candidate answer. (The

1698                    obs.record_ai_request(&provider.name, route_name, "translation_error", version);

1700                        &provider.name,
1701                        route_name,
1702                        ai_started.elapsed().as_secs_f64(),
1703                    );
1704                    tracing::warn!(
1705                        code = "ai_provider_body_over_cap",
1706                        request_id = %rid,

1714            };

1716            Ok(v) => v,
1717            Err(e) => {

1720                    &provider.name,

1731                // candidate may be healthy.

1733                    StatusCode::BAD_GATEWAY,
1734                    "the model provider returned a malformed response",
1735                    "api_error",
1736                    Some("provider_malformed_response"),
1737                    rid,
1738                );

1741            }

1744            Ok(r) => r,
1745            Err(e) => {

1747                obs.record_ai_request_duration(
1748                    &provider.name,
1749                    route_name,
1750                    ai_started.elapsed().as_secs_f64(),
1751                );

1753                    code = "ai_provider_body_untranslatable",
1754                    request_id = %rid,
1755                    provider = %provider.name,
1756                    "provider 200 response could not be translated: {e}"
1757                );

1759                    StatusCode::BAD_GATEWAY,
1760                    "the model provider returned a response the gateway could not translate",
1761                    "api_error",
1762                    Some("provider_untranslatable_response"),
1763                    rid,
1764                );
1765                last_transport_error = false;
1766                continue;
1767            }

1770        // span (provider-reported only; None leaves the field empty).

1772            .choices
1773            .iter()
1774            .map(|c| c.finish_reason.as_gen_ai())
1775            .collect();

1777            &gen_ai_span,
1778            chat_resp.model.as_deref(),

1780            chat_resp.usage.and_then(|u| u.completion_tokens),

1787            // version (DW-076; the input DW-079 cost metering reads).

1789                &provider.name,
1790                usage.prompt_tokens.unwrap_or(0),
1791                usage.completion_tokens.unwrap_or(0),
1792                version,
1793            );
1794            // AI-07: per-request token distribution histogram.
1795            obs.record_ai_tokens_per_request(
1796                &provider.name,
1797                &target.provider_model,
1798                usage.prompt_tokens.unwrap_or(0),
1799                usage.completion_tokens.unwrap_or(0),
1800            );

1802            // pricing table (swapped on reload, so a pricing change
1803            // applies to the next request with no restart). Unknown
1804            // model -> None (the unknown-model policy is applied
1805            // below, DW-AI-03).
1806            let pricing = dp.ai_pricing();

1808            let cost = cost_opt.unwrap_or(0);

1810                obs.record_ai_cost(&provider.name, &target.provider_model, cost);

1814            // the call — check-then-spend; the crossing (if any) is
1815            // already priced into the next pre-check.
1816            if let Some(guard) = &budget {
1817                guard.spend(now_epoch_s(), usage, cost);
1818            }

1820            // store for billing reconciliation. The team field is the
1821            // policy-scoped budget's key (the policy name) when a
1822            // team budget binds, else empty.
1823            if let Some(analytics) = dp.analytics() {

1825                    .map(|id| id.consumer_name.as_str())
1826                    .unwrap_or("anonymous");
1827                let team = budget
1828                    .as_ref()
1829                    .map(|g| g.team_key().to_string())
1830                    .unwrap_or_default();

1832                let completion = usage.completion_tokens.unwrap_or(0);
1833                let total = usage
1834                    .total_tokens
1835                    .unwrap_or_else(|| prompt.saturating_add(completion));
1836                analytics.offer_ai_spend(crate::analytics::AiSpendRecord {
1837                    ts_ms: now_ms(),
1838                    consumer: consumer_name.to_string(),
1839                    consumer_type: identity
1840                        .map(|i| consumer_type_str(i.consumer_type))
1841                        .unwrap_or("user")
1842                        .to_string(),
1843                    team,
1844                    provider: provider.name.clone(),
1845                    model: target.provider_model.clone(),
1846                    version: version.to_string(),
1847                    prompt_tokens: prompt,
1848                    completion_tokens: completion,
1849                    total_tokens: total,
1850                    cost_micros: cost,
1851                });
1852            }
1853        }

1855        // computation, if the serving provider_model has no pricing
1856        // entry, apply the configured policy. `fail_closed` rejects
1857        // the response with 403 `ai_unknown_model_pricing` (the
1858        // provider call already happened, but the unpriced model
1859        // cannot serve traffic silently); `alert` logs a WARN and

1865            let policy = gen
1866                .snapshot
1867                .gateway()
1868                .ai
1869                .as_ref()
1870                .and_then(|c| c.unknown_model_policy)
1871                .unwrap_or_default();

1873                UnknownModelPolicy::FailClosed => {
1874                    obs.record_ai_request(
1875                        &provider.name,
1876                        route_name,
1877                        "unknown_model_pricing",
1878                        version,
1879                    );
1880                    tracing::warn!(
1881                        code = "ai_unknown_model_pricing",
1882                        request_id = %rid,
1883                        route = %route_name,
1884                        provider = %provider.name,
1885                        model = %target.provider_model,
1886                        "rejecting response: model has no pricing entry (fail_closed policy)"
1887                    );
1888                    return ai_error_response(
1889                        StatusCode::FORBIDDEN,
1890                        "the model has no pricing entry and the unknown-model policy is fail_closed",
1891                        "invalid_request_error",
1892                        Some("ai_unknown_model_pricing"),
1893                        rid,
1894                    );
1895                }
1896                UnknownModelPolicy::Alert => {
1897                    obs.record_ai_unknown_pricing(&target.provider_model);
1898                    tracing::warn!(
1899                        code = "ai_unknown_model_pricing",
1900                        request_id = %rid,
1901                        route = %route_name,
1902                        provider = %provider.name,
1903                        model = %target.provider_model,
1904                        "model has no pricing entry; proceeding with cost 0 (alert policy)"
1905                    );
1906                }
1907                UnknownModelPolicy::Allow => {}
1908            }
1909        }
1910        obs.record_ai_request(&provider.name, route_name, "success", version);
1911        // AI-07: end-to-end AI request latency (non-streaming success).
1912        obs.record_ai_request_duration(
1913            &provider.name,
1914            route_name,
1915            ai_started.elapsed().as_secs_f64(),
1916        );
1917        // DW-082: response guardrails. AFTER the response is parsed
1918        // and BEFORE it is returned to the client. A `block` action
1919        // returns a 400 (`response_schema_violation` for schema kind,
1920        // `guardrail_blocked` otherwise); a `log` action records and
1921        // continues. Non-streaming only (streaming responses cannot

1929            let service_policies: &[String] = route_cfg
1930                .map(|r| crate::ai::budget::service_policies_of(gateway, &r.service))
1931                .unwrap_or(&[]);
1932            let consumer_policies: &[String] = consumer
1933                .map(|c| crate::ai::budget::consumer_policies_of(gateway, c))
1934                .unwrap_or(&[]);

1937                consumer_policies,
1938                route_policies,
1939                service_policies,

1942                &chat_resp,
1943            );
1944            if let crate::ai::guardrails::GuardrailResult::Block { rule_name, reason } =
1945                &guardrail_result
1946            {

1948                    "schema"
1949                } else {
1950                    "response"
1951                };

1953                    .record_ai_guardrail_blocked(rule_name, kind_label);

1955                    code = "guardrail_blocked",
1956                    request_id = %rid,
1957                    route = %route_name,
1958                    rule = %rule_name,
1959                    reason = %reason,
1960                    "AI guardrail blocked the response before returning to the client"
1961                );
1962                let error_code = if reason == "response_schema_violation" {
1963                    "response_schema_violation"
1964                } else {
1965                    "guardrail_blocked"
1966                };
1967                let body = openai_compat::error_body(
1968                    &format!("the response was blocked by guardrail rule '{rule_name}': {reason}"),
1969                    "invalid_request_error",
1970                    Some(error_code),
1971                    rid,
1972                );
1973                return Response::builder()
1974                    .status(StatusCode::BAD_REQUEST)
1975                    .header(hyper::header::CONTENT_TYPE, "application/json")

1978            }
1979        }
1980        let body = openai_compat::response_to_openai(&chat_resp, &chat_req.model, rid);
1981        // DW-081: prompt/response logging capture hook (non-streaming).
1982        // After the response is parsed and BEFORE it is sent, check
1983        // whether logging is enabled (global + per-consumer override),
1984        // apply sampling, redact, and offer to analytics. The capture
1985        // is fire-and-forget (never blocks the response path).
1986        capture_prompt_log(
1987            dp,
1988            gen,
1989            rid,
1990            route_name,
1991            &provider.name,
1992            &target.provider_model,
1993            version,
1994            identity,
1995            &json_body,
1996            &body,
1997            false,
1998        );
1999        tracing::info!(
2000            code = "ai_request_served",
2001            request_id = %rid,
2002            route = %route_name,
2003            provider = %provider.name,
2004            model = %chat_req.model,
2005            version = %version,
2006            attempts = attempts,
2007            upstream = %provider.upstream,
2008            "ai chat completion translated and served"
2009        );
2010        // DW-083: store the response in the semantic cache
2011        // (non-streaming only). Fire-and-forget — the embedding call
2012        // and HNSW insert happen in a spawned task so they never
2013        // block the response path. The prompt text is the joined
2014        // message text (the same shape the lookup used); the body is
2015        // the OpenAI-shaped response JSON.
2016        if can_cache {
2017            if let Some(sem_cache) = &sem_cache {
2018                let sem_cache = Arc::clone(sem_cache);
2019                let prompt_text = chat_req
2020                    .messages
2021                    .iter()
2022                    .map(|m| m.text_content())
2023                    .collect::<Vec<_>>()
2024                    .join("\n");
2025                let body_clone = body.clone();
2026                let model = chat_req.model.clone();
2027                tokio::spawn(async move {
2028                    sem_cache.store(&prompt_text, &body_clone, &model).await;
2029                });
2030            }
2031        }
2032        return response_with_json(StatusCode::OK, &body, &up_parts.headers);
2033    }
2034
2035    // 6. Every candidate failed. The last stashed provider answer is
2036    // the most truthful response; a pure-transport exhaustion keeps
2037    // the generic 502.
2038    rec.attempts = attempts.max(1);
2039    if attempts > 1 {
2040        tracing::warn!(
2041            code = "ai_failover_exhausted",
2042            request_id = %rid,
2043            model = %chat_req.model,
2044            attempts = attempts,
2045            transport_only = last_transport_error,
2046            "all providers failed for the model"
2047        );
2048    }
2049    fallback_response
2050}

2161/// Collect a body up to `cap` bytes, aborting the read the moment the
2162/// cap is crossed (the producer is dropped mid-stream — an over-cap
2163/// body is never buffered whole). Over-cap answers 413 inbound / 502
2164/// provider-side, in the AI error shape.
2165async fn bounded_collect<B>(
2166    body: B,
2167    cap: usize,
2168    what: &str,
2169    rid: &str,
2170) -> Result<Bytes, Response<ProxyBody>>
2171where
2172    B: Body<Data = Bytes> + Send + 'static,
2173    B::Error: Into<Box<dyn std::error::Error + Send + Sync>>,
2174{
2175    let mut buf: Vec<u8> = Vec::new();
2176    let mut body = pin!(body);
2177    while let Some(frame) = body.as_mut().frame().await {
2178        let frame = frame.map_err(|e| {
2179            let e = e.into();
2180            ai_error_response(
2181                StatusCode::BAD_REQUEST,
2182                &format!("the {what} body could not be read: {e}"),
2183                if what == "request" {
2184                    "invalid_request_error"
2185                } else {
2186                    "api_error"
2187                },
2188                None,
2189                rid,
2190            )
2191        })?;
2192        if let Some(data) = frame.data_ref() {
2193            if buf.len() + data.len() > cap {
2194                return Err(ai_error_response(
2195                    if what == "request" {
2196                        StatusCode::PAYLOAD_TOO_LARGE
2197                    } else {
2198                        StatusCode::BAD_GATEWAY
2199                    },
2200                    &format!("the {what} body exceeds the gateway's {cap}-byte limit"),
2201                    if what == "request" {
2202                        "invalid_request_error"
2203                    } else {
2204                        "api_error"
2205                    },
2206                    Some("body_too_large"),
2207                    rid,
2208                ));
2209            }
2210            buf.extend_from_slice(data);
2211        }
2212    }
2213    Ok(Bytes::from(buf))
2214}

2273

2278    };

2280    // OpenAI: x-ratelimit-remaining-requests, x-ratelimit-remaining-tokens
2281    // Anthropic: anthropic-ratelimit-requests-remaining, anthropic-ratelimit-tokens-remaining
2282    // Gemini: x-ratelimit-remaining (generic)
2283    // We try all known header names and take the first that parses.
2284    let remaining_request_headers = [
2285        "x-ratelimit-remaining-requests",
2286        "anthropic-ratelimit-requests-remaining",
2287        "x-ratelimit-remaining",
2288    ];

2290        "x-ratelimit-remaining-tokens",
2291        "anthropic-ratelimit-tokens-remaining",
2292    ];

2294        "x-ratelimit-reset-requests",
2295        "anthropic-ratelimit-requests-reset",
2296        "x-ratelimit-reset",
2297    ];

2300        if let Some(v) = headers.get(*name).and_then(|h| h.to_str().ok()) {
2301            if let Ok(n) = v.trim().parse::<u64>() {
2302                health.remaining_requests = Some(n);
2303                break;
2304            }
2305            // Some providers send a duration like "1s" or "6m0s"
2306            if let Some(n) = parse_duration_seconds(v) {
2307                health.remaining_requests = Some(n);
2308                break;
2309            }
2310        }

2313    for name in &remaining_token_headers {
2314        if let Some(v) = headers.get(*name).and_then(|h| h.to_str().ok()) {
2315            if let Ok(n) = v.trim().parse::<u64>() {
2316                health.remaining_tokens = Some(n);
2317                break;
2318            }
2319        }
2320    }
2321
2322    for name in &reset_headers {
2323        if let Some(v) = headers.get(*name).and_then(|h| h.to_str().ok()) {
2324            if let Some(secs) = parse_duration_seconds(v) {
2325                health.reset_at = Some(Instant::now() + Duration::from_secs(secs));
2326                break;
2327            }
2328            // Try as integer seconds
2329            if let Ok(secs) = v.trim().parse::<u64>() {
2330                health.reset_at = Some(Instant::now() + Duration::from_secs(secs));
2331                break;
2332            }
2333        }
2334    }
2335
2336    health
2337}

2339/// Parse a duration string like "1s", "6m0s", "1h2m3s" into seconds.
2340fn parse_duration_seconds(s: &str) -> Option<u64> {
2341    let s = s.trim();
2342    if s.is_empty() {
2343        return None;
2344    }
2345    // Try plain integer first
2346    if let Ok(n) = s.parse::<u64>() {
2347        return Some(n);
2348    }
2349    // Parse h/m/s suffixes
2350    let mut total: u64 = 0;
2351    let mut num = String::new();
2352    for ch in s.chars() {
2353        if ch.is_ascii_digit() {
2354            num.push(ch);
2355        } else {
2356            let n: u64 = num.parse().ok()?;
2357            num.clear();
2358            match ch {
2359                'h' => total += n * 3600,
2360                'm' => total += n * 60,
2361                's' => total += n,
2362                _ => return None,
2363            }
2364        }
2365    }
2366    // Trailing number without suffix = seconds
2367    if !num.is_empty() {
2368        let n: u64 = num.parse().ok()?;
2369        total += n;
2370    }
2371    Some(total)
2372}

2597    /// Spend the provider-reported usage GROWTH since the last spend
2598    /// (DW-078). The translator's usage is an ACCUMULATOR that grows
2599    /// as the provider reports (input tokens early, output late), so
2600    /// only the delta above the watermark is NEW spend — each
2601    /// reported token is counted exactly once regardless of how many
2602    /// chunks its report spanned. Returns whether the token window
2603    /// crossed its limit on THIS spend (the mid-stream cutoff signal).
2604    fn budget_spend_delta(&mut self) -> bool {
2605        let Some(guard) = &self.budget else {
2606            return false;
2607        };
2608        let usage = self.translator.usage();
2609        let total = usage.total_tokens.unwrap_or_else(|| {
2610            usage
2611                .prompt_tokens
2612                .unwrap_or(0)
2613                .saturating_add(usage.completion_tokens.unwrap_or(0))
2614        });
2615        let tokens = total.saturating_sub(self.budget_spent_tokens);
2616        let cost_total = self
2617            .pricing
2618            .cost_micros(&self.provider_model, usage, false)
2619            .unwrap_or(0);
2620        let cost = cost_total.saturating_sub(self.budget_spent_cost_micros);
2621        if tokens == 0 && cost == 0 {
2622            return false;
2623        }
2624        // Watermarks advance to the accumulated totals (max, so a
2625        // provider that revises a count downward never re-spends the
2626        // difference later).
2627        self.budget_spent_tokens = total.max(self.budget_spent_tokens);
2628        self.budget_spent_cost_micros = cost_total.max(self.budget_spent_cost_micros);
2629        guard.spend(
2630            now_epoch_s(),
2631            Usage {
2632                total_tokens: Some(tokens),
2633                ..Usage::default()
2634            },
2635            cost,
2636        )

2640    /// batch: the batch's NEW usage is spent against the holder's
2641    /// window, and a crossing (only detectable when the dialect

2643    /// tokens) cuts the stream.

2645        if self.budget.is_none() {
2646            return;
2647        }
2648        let crossed = self.budget_spend_delta();

2652            // cancelled the moment its body is dropped, however slowly
2653            // (or never) the client drains the cutoff tail below — the
2654            // cancel must not wait on client behavior. The tail still
2655            // reaches the client: the cutoff event, then [DONE].
2656            self.inner = None;
2657            self.obs.record_ai_budget_denied("tokens");
2658            tracing::warn!(
2659                code = "ai_budget_exceeded_midstream",
2660                request_id = %self.rid,
2661                "token budget crossed mid-stream; cutting off and cancelling the provider stream"

2664                .push_back(crate::ai::budget::BudgetGuard::cutoff_frame(&self.rid));

2666        }

2683        if let Some(rule_name) = self.guardrails.check_stream_chunk(&batch_text) {
2684            self.guardrail_cut_off = true;
2685            // Drop the provider body NOW (eager upstream cancel).
2686            self.inner = None;
2687            self.obs.record_ai_guardrail_blocked(&rule_name, "banned");

2689                code = "guardrail_blocked_midstream",

2693            );

2695            let error_payload = serde_json::json!({
2696                "error": {
2697                    "code": "guardrail_blocked",
2698                    "message": "the response was blocked by a banned-content guardrail",
2699                    "request_id": self.rid,
2700                }
2701            });
2702            self.tail.push_back(format!("data: {}\n\n", error_payload));
2703            self.tail.push_back("data: [DONE]\n\n".to_string());
2704        }

2780        self.obs.record_ai_stream_end(

2782            self.gauges.started.elapsed().as_secs_f64(),

2800            // AI-07: per-request token distribution histogram.

2806            );
2807        }
2808        // AI-07: record the response-side gen_ai.* attributes on the

2810        // provider model (streaming providers do not echo a model in a

2816            Some(&self.provider_model),
2817            usage.prompt_tokens,
2818            usage.completion_tokens,

2822        );

2824        // store (exactly once per stream). The terminal usage is the
2825        // accumulated provider-reported total; cost is priced through
2826        // the dataplane's pricing table.
2827        if !self.spend_recorded {
2828            self.spend_recorded = true;
2829            let prompt = usage.prompt_tokens.unwrap_or(0);
2830            let completion = usage.completion_tokens.unwrap_or(0);
2831            let total = usage
2832                .total_tokens
2833                .unwrap_or_else(|| prompt.saturating_add(completion));
2834            let cost = self
2835                .pricing
2836                .cost_micros(&self.provider_model, usage, false)
2837                .unwrap_or(0);
2838            if cost > 0 {
2839                self.obs
2840                    .record_ai_cost(&self.gauges.provider, &self.provider_model, cost);
2841            }
2842            if let Some(analytics) = &self.analytics {
2843                analytics.offer_ai_spend(crate::analytics::AiSpendRecord {
2844                    ts_ms: now_ms(),

2846                    consumer_type: self.consumer_type.clone(),
2847                    team: self.team.clone(),
2848                    provider: self.gauges.provider.clone(),
2849                    model: self.provider_model.clone(),
2850                    version: self.gauges.version.clone(),
2851                    prompt_tokens: prompt,
2852                    completion_tokens: completion,
2853                    total_tokens: total,
2854                    cost_micros: cost,
2855                });
2856            }
2857        }
2858        // DW-081: streaming prompt/response capture. The prompt is
2859        // stored redacted (redacted at build time); the response is
2860        // marked as streamed (the full content is not reassembled —
2861        // buffering the entire stream would contradict the zero-buffer
2862        // design). The context was built at stream start only when
2863        // logging is enabled for this consumer and the sampling
2864        // decision was positive.
2865        if let Some(ctx) = &mut self.prompt_log_ctx {
2866            if !ctx.recorded {
2867                ctx.recorded = true;
2868                if let Some(analytics) = &self.analytics {
2869                    analytics.offer_ai_prompt_log(crate::analytics::AiPromptLogRecord {
2870                        ts_ms: now_ms(),
2871                        request_id: self.rid.clone(),

2891impl hyper::body::Body for AiStreamBody {
2892    type Data = Bytes;
2893    type Error = super::proxy::ProxyBodyError;
2894
2895    fn poll_frame(
2896        self: Pin<&mut Self>,
2897        cx: &mut Context<'_>,
2898    ) -> Poll<Option<Result<hyper::body::Frame<Bytes>, super::proxy::ProxyBodyError>>> {
2899        let this = self.get_mut();
2900        loop {
2901            // Drain synthesized terminal frames first: one data frame
2902            // per poll (flushes reach the client per frame).
2903            if let Some(text) = this.tail.pop_front() {
2904                if this.tail.is_empty() && this.translator.is_ended() {
2905                    // The last terminal frame: close after it is
2906                    // handed over. The NEXT poll returns None.
2907                    this.tee_collect(std::slice::from_ref(&text));
2908                    this.close();
2909                }
2910                return Poll::Ready(Some(Ok(hyper::body::Frame::data(Bytes::from(text)))));
2911            }
2912            if this.translator.is_ended() {
2913                return Poll::Ready(None);
2914            }
2915            // The provider body is gone once the budget cutoff dropped
2916            // it (eager upstream cancel): the synthesized tail above is
2917            // the stream's entire remainder.
2918            let Some(inner) = this.inner.as_mut() else {
2919                return Poll::Ready(None);
2920            };
2921            match Pin::new(inner).poll_frame(cx) {
2922                Poll::Ready(Some(Ok(frame))) => {
2923                    let Some(data) = frame.data_ref().cloned() else {
2924                        continue; // trailers / non-data frames pass by
2925                    };
2926                    let (frames, chunks, first) = this.translator.feed(data.as_ref(), this.adapter);
2927                    this.note_chunks(chunks, first);
2928                    this.budget_tick();
2929                    if this.translator.is_ended() {
2930                        this.tail.extend(this.translator.finish());
2931                    }
2932                    if frames.is_empty() {
2933                        // Bytes only continued a partial frame; keep
2934                        // polling (or drain the tail next loop).
2935                        continue;
2936                    }
2937                    let mut joined = String::new();
2938                    for f in &frames {
2939                        joined.push_str(f);
2940                    }
2941                    // DW-082: banned-content guardrail check on the
2942                    // forwarded batch text. A match cuts the stream
2943                    // off (the tail is synthesized above); the
2944                    // already-joined batch is still forwarded (the
2945                    // match is in THIS batch — the cutoff prevents
2946                    // FURTHER content, not retroactive suppression).
2947                    this.guardrail_tick(&joined);
2948                    if this.guardrail_cut_off {
2949                        // The guardrail fired: forward this batch
2950                        // (already-joined) then the tail (error +
2951                        // [DONE]) drains on subsequent polls. The
2952                        // provider body was dropped by guardrail_tick.
2953                        this.tee_collect(&frames);
2954                        return Poll::Ready(Some(Ok(hyper::body::Frame::data(Bytes::from(
2955                            joined,
2956                        )))));
2957                    }
2958                    this.tee_collect(&frames);
2959                    return Poll::Ready(Some(Ok(hyper::body::Frame::data(Bytes::from(joined)))));
2960                }
2961                Poll::Ready(Some(Err(e))) => {
2962                    // Mid-stream provider abort: already-forwarded
2963                    // content stands; the client stream ends cleanly
2964                    // with an error chunk and the terminator.
2965                    tracing::warn!(
2966                        code = "ai_stream_aborted",
2967                        provider = %this.gauges.provider,
2968                        "provider stream aborted mid-flight: {e}"
2969                    );
2970                    this.tail.extend(
2971                        this.translator
2972                            .abort_tail("the model provider closed the stream"),
2973                    );
2974                    continue;
2975                }
2976                Poll::Ready(None) => {
2977                    // Clean end: flush any unterminated frame, then
2978                    // the terminal usage chunk and the terminator.
2979                    let (frames, chunks, first) = this.translator.flush_partial(this.adapter);
2980                    this.note_chunks(chunks, first);
2981                    let mut joined = String::new();
2982                    for f in frames {
2983                        joined.push_str(&f);
2984                    }
2985                    this.tail.extend(this.translator.finish());
2986                    if joined.is_empty() {
2987                        continue;
2988                    }
2989                    return Poll::Ready(Some(Ok(hyper::body::Frame::data(Bytes::from(joined)))));
2990                }
2991                Poll::Pending => return Poll::Pending,
2992            }
2993        }
2994    }
